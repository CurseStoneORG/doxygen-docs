23 // Recipe — a single crafting recipe. Shaped recipes have a 3x3 grid
24 // pattern; shapeless recipes have a list of ingredients. Result is
25 // always an ItemStack.
26 //
27 // Las recetas se registran en CraftingManager::registerBuiltinRecipes(),
28 // en codigo. Cada tipo se construye con su helper:
29 // - RECIPE_SHAPED shaped(w, h, grilla 3x3, resultado)
30 // - RECIPE_SHAPELESS shapeless(ingredientes, resultado)
31 // - RECIPE_FURNACE furnace(entrada, meta, resultado)
32 //
33 // El meta de cada ingrediente es -1 cuando la receta acepta cualquier
34 // meta ("any damage", Nukkit item->hasAnyDamageValue()).
35 #pragma once
36 #include "item/ItemStack.h"
37 #include <vector>
38 #include <string>
39 #include <array>
40 #include <cstdint>
41 namespace cursestone::inventory {
42 enum RecipeType { RECIPE_SHAPED = 0, RECIPE_SHAPELESS = 1, RECIPE_FURNACE = 2 };
43 class Recipe {
44 public:
45  Recipe(RecipeType type, cursestone::item::ItemStack result)
46  : type_(type), result_(result) {}
47  RecipeType type() const { return type_; }
48  const cursestone::item::ItemStack& result() const { return result_; }
49  // Shaped recipe: 3x3 grid of item ids (0 = no requirement),
50  // zero-padded alrededor del patrón Width x Height real.
51  void setGrid(const std::array<uint16_t, 9>& g) { grid_ = g; }
52  const std::array<uint16_t, 9>& grid() const { return grid_; }
53  // Meta/damage requerido por celda (-1 = cualquier damage). Índices
54  // alineados 1:1 con grid().
55  void setGridMeta(const std::array<int16_t, 9>& m) { grid_meta_ = m; }
56  const std::array<int16_t, 9>& gridMeta() const { return grid_meta_; }
57  // Shapeless ingredients (list of item ids).
58  void setIngredients(const std::vector<uint16_t>& is) { ingredients_ = is; }
59  const std::vector<uint16_t>& ingredients() const { return ingredients_; }
60  // Meta por ingrediente shapeless (-1 = cualquier damage). Alineado
61  // 1:1 con ingredients(); si viene corto se trata como -1.
62  void setIngredientMeta(const std::vector<int16_t>& ms) { ingredient_meta_ = ms; }
63  const std::vector<int16_t>& ingredientMeta() const { return ingredient_meta_; }
64  // Bounding box del patrón shaped tal como se registro (1..3). Para
65  // matchShaped basta grid(); width/height se guardan para el
66  // CraftingDataPacket (que serializa el patrón SIN padding,
67  // Width x Height celdas exactas).
68  void setSize(int w, int h) { width_ = w; height_ = h; }
69  int width() const { return width_; }
70  int height() const { return height_; }
71  // Furnace input damage (-1 = cualquier damage, ENTRY_FURNACE en el
72  // wire; 0..n = damage exacto, ENTRY_FURNACE_DATA).
73  void setFurnaceInputMeta(int16_t m) { furnace_input_meta_ = m; }
74  int16_t furnaceInputMeta() const { return furnace_input_meta_; }
75  void setUuid(uint64_t u) { uuid_ = u; }
76  uint64_t uuid() const { return uuid_; }
77  // Los 16 bytes crudos del UUID de receta que viaja en
78  // CraftingDataPacket y vuelve en CraftingEventPacket (mismo orden
79  // que PMMP UUID::toBinary). El cliente lo ecoa tal cual, así que el
80  // match contra recetas se hace por estos bytes, no por uuid().
81  void setUuidBytes(const std::array<uint8_t, 16>& b) { uuid_bytes_ = b; }
82  const std::array<uint8_t, 16>& uuidBytes() const { return uuid_bytes_; }
83 private:
84  RecipeType type_;
85  cursestone::item::ItemStack result_;
86  std::array<uint16_t, 9> grid_{};
87  std::array<int16_t, 9> grid_meta_{};
88  std::vector<uint16_t> ingredients_;
89  std::vector<int16_t> ingredient_meta_;
90  int width_ = 0;
91  int height_ = 0;
92  int16_t furnace_input_meta_ = -1;
93  uint64_t uuid_ = 0;
94  std::array<uint8_t, 16> uuid_bytes_{};
95 };
96 } // namespace
