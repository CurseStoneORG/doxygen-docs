37 // Split string on a single-char delimiter (whitespace-aware).
38 inline std::vector<std::string> split(const std::string& s, char delim) {
39  std::vector<std::string> out;
40  std::string tok;
41  for (char c : s) {
42  if (c == delim) {
43  if (!tok.empty()) {
44  out.push_back(tok);
45  tok.clear();
46  }
47  } else {
48  tok += c;
49  }
50  }
51  if (!tok.empty())
52  out.push_back(tok);
53  return out;
54 }

56 inline std::string trim(std::string s) {
57  auto l = s.find_first_not_of(" \t\r\n");
58  auto r = s.find_last_not_of(" \t\r\n");
59  return (l == std::string::npos) ? "" : s.substr(l, r - l + 1);
60 }
