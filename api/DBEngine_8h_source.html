24 // DBEngine — a self-contained, real-format LevelDB engine.
25 //
26 // Creates and maintains the same file set the real MCPE client produces
27 // inside `<world>/db`: CURRENT, LOCK, MANIFEST-<n>, <n>.log (WAL) and
28 // <n>.ldb (tables) — no external library involved. Worlds written by the
29 // real device (snappy tables) and by PocketMine (zlib tables) open here,
30 // and the files this engine writes are ordinary LevelDB tables any reader
31 // of the format understands.
32 //
33 // Layout it maintains:
34 // db/CURRENT -> "MANIFEST-000002\n"
35 // db/LOCK -> flock()ed exclusively while open
36 // db/MANIFEST-<n> -> log-format file holding the full VersionEdit
37 // db/<n>.log -> WAL of batches not yet flushed to a table
38 // db/<n>.ldb -> SSTables (level 0 from flushes, level 1 from
39 // the periodic full compaction)
40 //
41 // Durability contract (what a server actually needs):
42 // * every accepted batch is in the WAL before put_batch() returns
43 // (process-crash safe),
44 // * flush() turns the memtable into a table + rotates the WAL (the
45 // manifest is rewritten and CURRENT swapped atomically before the old
46 // WAL is deleted, so a crash in between replays, never loses),
47 // * full compaction merges every table when level-0 count crosses the
48 // trigger, keeping the file set bounded.
49 //
50 // Threading: one internal mutex. The provider funnels writes through its
51 // background worker; reads may come from the main thread concurrently.

75 // A cached read-only view of one live table. Tables are immutable once
76 // written, so the image and the parsed reader are shared safely across
77 // threads WITHOUT the engine mutex: point lookups and scans only take
78 // mtx_ briefly to fetch the shared view (and, on a miss, to publish the
79 // freshly loaded one). POSIX maps the file read-only (page-cache backed,
80 // no RSS amplification); platforms without mmap cache the bytes read
81 // once into the heap buffer.
82 struct OpenTable {
83  uint64_t number = 0;
84  size_t size = 0;
85  const uint8_t* image = nullptr; // mapped view when mapped, else heap.data()
86  std::vector<uint8_t> heap; // populated when the file is not mapped
87  bool mapped = false;
88  db::TableReader reader;
89 };

98  // Opens (or creates) the database rooted at `db_dir`. Fails when the

110  }

112  // One write batch. nullopt value = deletion tombstone. All ops land

119  std::optional<std::string> get(const std::string& user_key) const;

121  // Calls sink(user_key, value) for every live key starting with

128  // Approximate bytes waiting in the memtable (flush trigger input).

134  // Memtable -> new level-0 table + WAL rotation + manifest rewrite.

139  return dir_;

142  private:
143  // --- manifest & files (all called with mtx_ held unless noted) ------
144  bool write_manifest_locked(std::string* err);
145  bool replay_logs_locked(uint64_t from_number, uint64_t* max_seq, std::string* err);
146  bool flush_locked(std::string* err);
147  bool compact_locked(std::string* err); // full merge of every table

154  // --- cached table views --------------------------------------------
155  // NOT called with mtx_ held for the load path: a cache miss maps/
156  // reads the file with the mutex RELEASED (the whole point — the old
157  // code read whole tables under mtx_, so every chunk load serialized
158  // behind flushes and compactions). Returns a shared view that stays
159  // valid even after the file is pruned from the cache or deleted by a
160  // later compaction (the caller holds its own reference).
161  std::shared_ptr<const OpenTable> get_table(uint64_t number) const;
162  // One point-lookup attempt over the memtable + candidate tables (see
163  // get() for the retry-on-generation-change wrapper).
164  bool get_once(const std::string& user_key, const std::string& target,
165  uint64_t* gen, std::optional<std::string>* out, bool* retry) const;
166  // Drops cached views of tables that no longer belong to the live file
167  // set (after a compaction) and enforces the cache byte cap. Called
168  // with mtx_ held; entries still referenced by a reader stay alive via
169  // shared_ptr until the reader is done with them.
170  void prune_table_cache_locked() const;
171 
172  // --- state (guarded by mtx_) ----------------------------------------
173  mutable std::mutex mtx_;
174  bool opened_ = false;
175  std::string dir_;
176 
177  // Memtable: internal keys (with sequence + type) sorted newest-first
178  // per user key. Value "" + kTypeDeletion trailer = tombstone.
179  std::map<std::string, std::string, InternalKeyLess> memtable_;
180  size_t memtable_bytes_ = 0;
181 
182  std::vector<db::FileMeta> files_; // all live tables, any level
183  uint64_t log_number_ = 0;
184  uint64_t next_file_number_ = 1;
185  uint64_t last_sequence_ = 0;
186 
187  FILE* log_file_ = nullptr;
188  uint64_t wal_bytes_ = 0; // bytes already written to the current WAL
189  int lock_fd_ = -1;
190  uint64_t manifest_number_ = 0;
191  size_t memtable_batch_count_ = 0; // batches since last flush
192 
193  // Cached immutable table views (see OpenTable). Mutated only under
194  // mtx_ (insert/prune — mutable because get() is a const interface
195  // that lazily fills the cache); the views themselves are used
196  // lock-free.
197  mutable std::map<uint64_t, std::shared_ptr<const OpenTable>> open_tables_;
198  mutable size_t open_tables_bytes_ = 0;
199  // Bumped on every files_ change (flush publish, compaction publish).
200  // Point lookups snapshot the candidate list + generation and RETRY
201  // when the generation moved underneath them: a compaction that runs
202  // mid-lookup relocates the data into the merged table, and a stale
203  // candidate list alone could report a missing key for data that is
204  // very much alive (a read returning a pre-write value is legitimate
205  // snapshot semantics; a read missing a committed key is not).
206  mutable uint64_t files_generation_{0};
207  // Generous cap: a 2500-chunk MCPE db is ~60 MB of tables. Entries are
208  // evicted FIFO (oldest unreferenced first) once this is exceeded.
209  static constexpr size_t kTableCacheMaxBytes = 512u << 20;
210 };
