24 // VillagePalette — the biome-independent block vocabulary of the village
25 // templates plus the per-biome resolver that turns it into legacy blocks.
26 //
27 // The CurseStone-Structures village exports speak modern Java block
28 // names, and — like vanilla's StructureVillagePieces — the SAME template
29 // must build with a DIFFERENT material set depending on the village
30 // biome: a plains village is cobblestone and oak wood with gravel/paths,
31 // a desert village is the sandstone family with sand where the plains
32 // one has grass. This file is that mapping, the equivalent of vanilla's
33 // getBiomeSpecificBlockState(), hand-tuned to what MCPE 0.15.x has.
34 //
35 // Where a block kind has no direct desert equivalent in 0.15 the
36 // substitution is documented inline: farmland and grass become sand (a
37 // desert farm falls back to the sand that vanilla pushes around its
38 // crops — 0.15 has no desert crops), gravel path becomes sand, and the
39 // sandstone stairs vanilla would use fall back to cobble stairs, the
40 // same substitution the desert temple makes.
41 //
42 // Facings inside the specs are the JAVA compass (1 east, 2 west, 3
43 // south, 4 north, 5 up); resolve() receives the rotation the template
44 // is being stamped with and turns the facing with it, so doors still
45 // face their opening after a 90-degree turn.

56 // ---- The vocabulary -------------------------------------------------------
57 enum SpecKind : uint8_t {
58  SPEC_AIR = 0,
59  SPEC_COBBLE, // cobblestone walls/frames; desert: smooth sandstone
60  SPEC_LOG, // log pillars/beams; desert: smooth sandstone pillar
61  SPEC_PLANKS, // plank floors/walls; desert: cut sandstone
62  SPEC_DOUBLE_SLAB, // double stone slab floor; desert: double sandstone
63  SPEC_SLAB, // stone slab step; desert: sandstone slab
64  SPEC_WOOD_STAIRS, // oak stairs; desert: cobble stairs (no sand stairs in 0.15)
65  SPEC_COBBLE_STAIRS, // cobble stairs, both biomes
66  SPEC_FENCE, // oak fence, both biomes
67  SPEC_DOOR, // wooden door, both biomes
68  SPEC_PLATE, // wooden pressure plate
69  SPEC_STONE_PLATE, // stone pressure plate
70  SPEC_TORCH, // standing torch (facing 5)
71  SPEC_WALL_TORCH, // wall torch
72  SPEC_LADDER, // ladder
73  SPEC_GLASS_PANE, // glass pane, both biomes
74  SPEC_BOOKSHELF, // bookshelf, both biomes
75  SPEC_CRAFTING, // crafting table
76  SPEC_CHEST, // chest (facing = which way the front points)
77  SPEC_FURNACE, // furnace (facing = which way the front points)
78  SPEC_IRON_BARS, // iron bars, both biomes
79  SPEC_WOOL, // wool (misc = legacy colour)
80  SPEC_GRASS, // grass block; desert: sand
81  SPEC_DIRT, // dirt; desert: sand
82  SPEC_FARMLAND, // farmland (aux = moisture); desert: sand
83  SPEC_WATER, // still water (misc = level)
84  SPEC_LAVA, // lava (misc = level)
85 };

87 // ---- The per-biome material sets ------------------------------------------
88 //
89 // Which material family a village in a given biome builds with. SAVANNA
90 // uses the plains set: vanilla 1.9-era savanna villages reuse the plains
91 // pieces with acacia accents, and acacia wood/planks only exist as LOG
92 // data 1 / 125 data 1 — close enough to oak to not be worth splitting
93 // the palette for, while the shapes and layout stay vanilla.
94 inline bool isDesertVillageBiome(uint8_t biomeId) {
95  return biomeId == (uint8_t)cursestone::world::generator::biome::BIOME_DESERT;
96 }
97 
98 // Legacy data for the JAVA compass values (1 east, 2 west, 3 south,
99 // 4 north), per block family:
100 //
101 // * stairs (legacy 53/67): 0 east, 1 west, 2 south, 3 north + 4 top.
102 // Vanilla legacy stairs face TOWARD the player: the export's value
103 // is already that, so it transfers 1:1.
104 // * doors: 0 east, 1 south, 2 west, 3 north (bits 0-1), + 4 hinge
105 // right, + 8 open. The upper half stores nothing directional.
106 // * wall torch: 1 west, 2 east, 3 south, 4 north — the same map the
107 // desert temple used (verified there against the client).
108 // * chest/furnace/ladder (side-mounted): each family has its OWN
109 // table — a ladder's data is the side its face touches (2 north,
110 // 3 south, 4 west, 5 east), while a chest/furnace's data is its
111 // front direction in 0..3 (0 north, 1 south, 2 west, 3 east).
112 // * standing torch: 5.
113 constexpr uint8_t kStairsFacing[5] = {0, 0, 1, 2, 3}; // idx = java facing
114 constexpr uint8_t kDoorFacing[5] = {0, 0, 2, 1, 3};
115 constexpr uint8_t kWallTorchFacing[5]= {0, 2, 1, 3, 4};
116 // Ladder: legacy data is the side the ladder's face touches
117 // (2 north, 3 south, 4 west, 5 east).
118 constexpr uint8_t kLadderFacing[5] = {0, 5, 4, 3, 2};
119 // Chest/furnace: legacy data is the FRONT direction in 0..3
120 // (0 north, 1 south, 2 west, 3 east) — a different table from the
121 // ladder's, and writing a ladder value here would spawn a double chest.
122 constexpr uint8_t kFrontFacing[5] = {0, 3, 2, 1, 0};
123 
124 // The spec facings are the JAVA compass (1 east, 2 west, 3 south, 4
125 // north), so they index the tables directly — masking with 3 here would
126 // silently turn north (4) into east (0).
127 inline uint8_t stairsData(const BlockSpec& s) {
128  return (uint8_t)(kStairsFacing[s.facing] | (s.aux & 1 ? 4 : 0));
129 }
130 inline uint8_t doorData(const BlockSpec& s) {
131  const bool upper = (s.aux & 1) != 0;
132  const uint8_t bits = (uint8_t)(kDoorFacing[s.facing] |
133  (s.misc & 1 ? 4 : 0)); // open
134  return upper ? (uint8_t)(bits | 8) : bits;
135 }

137 // The resolver Structure::placeInto calls per cell. `user` carries the
138 // village biome id (0 = not a village biome, resolves as plains).
139 inline BlockState resolve(const BlockSpec& s, int rot, void* user) {
140  const uint8_t biomeId = user ? *(const uint8_t*)user : 0;
141  const bool desert = isDesertVillageBiome(biomeId);
142  // Facings rotate WITH the template (rotation-invariant kinds just
143  // pass through; the door's hinge/open bits ride along untouched).
144  const uint8_t facing =
145  (s.facing <= 4 && (rot & 3) != 0) ? rotateFacing(s.facing, rot)
146  : s.facing;
147  const BlockSpec t{s.kind, facing, s.aux, s.misc};
148 
149  switch (s.kind) {
150  case SPEC_AIR: return {0, 0};
151  case SPEC_COBBLE: return desert ? BlockState{(uint8_t)BlockId::SANDSTONE, 0}
152  : BlockState{(uint8_t)BlockId::COBBLESTONE, 0};
153  case SPEC_LOG: return desert ? BlockState{(uint8_t)BlockId::SANDSTONE, 0}
154  : BlockState{(uint8_t)BlockId::LOG, 0};
155  case SPEC_PLANKS: return desert ? BlockState{(uint8_t)BlockId::SANDSTONE, 2}
156  : BlockState{(uint8_t)BlockId::PLANKS, 0};
157  case SPEC_DOUBLE_SLAB: return desert ? BlockState{43, 1}
158  : BlockState{43, 0};
159  case SPEC_SLAB: return desert ? BlockState{(uint8_t)BlockId::STONE_SLAB,
160  (uint8_t)(1 | (t.aux & 1 ? 8 : 0))}
161  : BlockState{(uint8_t)BlockId::STONE_SLAB,
162  (uint8_t)(3 | (t.aux & 1 ? 8 : 0))};
163  case SPEC_WOOD_STAIRS: return desert ? BlockState{(uint8_t)BlockId::COBBLE_STAIRS, stairsData(t)}
164  : BlockState{(uint8_t)BlockId::WOOD_STAIRS, stairsData(t)};
165  case SPEC_COBBLE_STAIRS:return {(uint8_t)BlockId::COBBLE_STAIRS, stairsData(t)};
166  case SPEC_FENCE: return {(uint8_t)BlockId::FENCE, 0};
167  case SPEC_DOOR: return {(uint8_t)BlockId::WOODEN_DOOR, doorData(t)};
168  case SPEC_PLATE: return {(uint8_t)BlockId::WOOD_PRESSURE_PLATE, 0};
169  case SPEC_STONE_PLATE: return {(uint8_t)BlockId::STONE_PRESSURE_PLATE, 0};
170  case SPEC_TORCH: return {(uint8_t)BlockId::TORCH, 5};
171  case SPEC_WALL_TORCH: return {(uint8_t)BlockId::TORCH, kWallTorchFacing[t.facing]};
172  case SPEC_LADDER: return {(uint8_t)BlockId::LADDER, kLadderFacing[t.facing]};
173  case SPEC_GLASS_PANE: return {(uint8_t)BlockId::GLASS_PANE, 0};
174  case SPEC_BOOKSHELF: return {(uint8_t)BlockId::BOOKSHELF, 0};
175  case SPEC_CRAFTING: return {(uint8_t)BlockId::WORKBENCH, 0};
176  case SPEC_CHEST: return {(uint8_t)BlockId::CHEST, kFrontFacing[t.facing]};
177  case SPEC_FURNACE: return {(uint8_t)BlockId::FURNACE, kFrontFacing[t.facing]};
178  case SPEC_IRON_BARS: return {(uint8_t)BlockId::IRON_BARS, 0};
179  case SPEC_WOOL: return {(uint8_t)BlockId::WOOL, t.misc};
180  case SPEC_GRASS: return desert ? BlockState{(uint8_t)BlockId::SAND, 0}
181  : BlockState{(uint8_t)BlockId::GRASS, 0};
182  case SPEC_DIRT: return desert ? BlockState{(uint8_t)BlockId::SAND, 0}
183  : BlockState{(uint8_t)BlockId::DIRT, 0};
184  case SPEC_FARMLAND: return desert ? BlockState{(uint8_t)BlockId::SAND, 0}
185  : BlockState{(uint8_t)BlockId::FARMLAND, t.aux};
186  case SPEC_WATER: return {(uint8_t)BlockId::WATER, t.misc};
187  case SPEC_LAVA: return {(uint8_t)BlockId::LAVA, t.misc};
188  default: return {0, 0};
189  }
