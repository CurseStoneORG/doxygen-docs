23 // Living — base class for entities with health, armor, and potion
24 // effects. Mobs (Hostile, Passive), players, and humans derive
25 // from this. AI logic is *not* included.

31 #include <vector>

35 // Result of a damage()/damageByEntity() call:
36 // CANCELLED — no damage happened (already dead, a plugin cancelled the
37 // event, or the victim is immune — e.g. creative players).
38 // HURT — damage applied, the entity is still alive.
39 // KILLED — damage applied and this was the lethal blow (onDeath ran).
40 // Callers use this to decide whether the hit must be synced to clients:
41 // on CANCELLED not a single packet goes out (Infinity: a cancelled
42 // EntityDamageEvent produces no EntityEventPacket, no health sync and no
43 // knockback).
44 enum class DamageApply {
45  CANCELLED = 0,
46  HURT = 1,
47  KILLED = 2,
48 };

50 class Living : public Entity {
51  public:
52  using Entity::Entity;
53  float health() const {
54  return health_;
55  }
56  void setHealth(float h) {
57  health_ = h;
58  }
59  float maxHealth() const {
60  return max_health_;
61  }
62  void setMaxHealth(float m) {
63  max_health_ = m;
64  }
65 
66  // Damage the entity with a generic/no-attacker cause (CUSTOM). Dispara
67  // EntityDamageEvent antes de aplicar el dano; un listener puede
68  // reescribir la cantidad o cancelar el evento.
69  virtual DamageApply damage(float amount);
70 
71  // Igual que damage(), pero con un atacante conocido — dispara
72  // EntityDamageByEntityEvent en vez de EntityDamageEvent. Usado
73  // por el pipeline de combate cuerpo a cuerpo (ver
74  // Player::handleInteract / INTERACT_PACKET) y por las flechas
75  // (World::tickArrows, cause=PROJECTILE).
76  virtual DamageApply
77  damageByEntity(float amount, Entity* damager,
78  event::entity::DamageCause cause = event::entity::DamageCause::ENTITY_ATTACK);
79 
80  // mirrors PMMP Entity::attack(EntityDamageEvent) — punto de
81  // entrada generico sin atacante conocido (p.ej.
82  // BlockFire::onEntityCollide). Delega en damage().
83  void attack(float amount) override {
84  damage(amount);
85  }
86  // Damage with an explicit cause (Infinity-style): fires
87  // EntityDamageEvent with that cause so listeners can cancel/reduce
88  // per-cause (armor on FALL, fire resistance on LAVA, etc.). Used by
89  // the ambient damage pipeline: fall damage, lava, fire, cactus,
90  // drowning, starvation.
91  virtual DamageApply damage(float amount, event::entity::DamageCause cause);
92 
93  // Heal the entity by amount (capped at maxHealth).
94  virtual void heal(float amount);
95  // Dead flag (used by sweep removal).
96  bool isDead() const {
97  return health_ <= 0.0f;
98  }
99  void addEffect(std::shared_ptr<Effect> e) {
100  effects_.push_back(std::move(e));
101  }
102  const std::vector<std::shared_ptr<Effect>>& effects() const {
103  return effects_;
104  }
105  void tick(uint64_t current_tick) override;
106 
107  protected:
108  // Llamado por damage()/damageByEntity() cada vez que la salud baja
109  // por dano SIN morir (cuando muere se llama onDeath()). Player lo
110  // sobreescribe para sincronizar la vida al cliente AL INSTANTE
111  // (barra de corazones fluida) y mandar la animacion de golpe, igual
112  // que Infinity (Player::setHealth + EntityEventPacket::HURT_ANIMATION).
113  virtual void onHealthChanged() {
114  }
115 
116  // Llamado por damage()/damageByEntity() la primera vez que la
117  // salud cae a <= 0. No-op por defecto (mobs genericos no
118  // necesitan un evento de muerte todavia); Player lo sobreescribe
119  // para disparar PlayerDeathEvent.
120  virtual void onDeath() {
121  }
122 
123  // Aplica `amount` a health_ (clamp a 0) y llama onDeath() si esta
124  // fue la transicion viva->muerta. Compartido por damage() y
125  // damageByEntity() para no duplicar la logica de clamping/onDeath.
126  bool applyDamage(float amount);
127 
128  float health_ = 20.0f;
129  float max_health_ = 20.0f;
130  std::vector<std::shared_ptr<Effect>> effects_;
131 };
132 } // namespace
