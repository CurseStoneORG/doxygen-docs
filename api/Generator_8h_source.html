23 // Generator — abstract world generator interface.
24 // Mirrors Nukkit's cn.nukkit.level.generator.Generator: a generator is
25 // asked to populate a chunk's terrain, biomes and features. Generators
26 // run on AsyncPool workers (one chunk at a time), so implementations must
27 // be self-contained and stateless w.r.t. the main thread (the seed is
28 // the only state).
29 //
30 // CRITICAL: each generator defines its OWN spawn point via getSpawn().
31 // This mirrors Nukkit where Normal returns (127.5, 128, 127.5), Flat
32 // returns (128, floorLevel, 128), and Nether returns (128, 64, 128).
33 // The World no longer hardcodes a spawn position — it asks the generator.
34 #pragma once
35 #include "world/format/Chunk.h"
36 #include "math/Vector3.h"
37 #include <stdint.h>

39 namespace cursestone::world::generator {

41 class TerrainNoise; // definido en TerrainShape.h

43 // NOTE: la generacion corre 1 chunk por tarea en el worker del AsyncPool,
44 // asi que el ctx vive y muere en un solo hilo: un puntero plano sin
45 // ownership es seguro.
46 struct GenerationContext {
47  int64_t seed; // world seed
48  int32_t chunk_x;
49  int32_t chunk_z;
50  const uint8_t* biomes_in; // optional input biome map (256 bytes), NULL = generate
51  mcp_chunk_t* chunk; // target chunk — generator fills sections + biomes + heightmap
52 
53  // OPT (generacion lenta): ruido compartido entre Normal::generate y
54  // los populators del MISMO chunk. Es un TerrainNoise construido con
55  // ctx.seed y poseido por el que lo crea (Normal::generate, en el
56  // mismo scope que ctx). NULL = el populator construye el suyo
57  // (comportamiento por defecto; los tools viejos con ctx{} ni se
58  // enteran).
59  //
60  // Por que IMPORTA: construir TerrainNoise cuesta ~17 ms (8 capas de
61  // PerlinNoise + la calibracion de 5 campos, 5 x 4096 muestras) y
62  // antes CINCO populators lo construian cada uno por chunk: ~85 ms de
63  // trabajo identico tirado por chunk.
64  void* noise_cache = nullptr; // TerrainNoise* opaco, evita depender del header
65 };
66 
67 class Generator {
68 public:
69  virtual ~Generator() = default;
70  virtual const char* name() const = 0;
71 
72  // Phase 1: terrain + biomes + heightmap
73  virtual void generate(GenerationContext& ctx) = 0;
74 
75  // Phase 2: features (trees, ores, grass, flowers, caves) — may
76  // touch neighbouring chunks; called after terrain_populated=true.
77  virtual void populate(GenerationContext& ctx) = 0;
78 
79  // Returns the world spawn position for this generator type.
80  // Mirrors Nukkit's Generator.getSpawn() — each generator type
81  // defines its own spawn (Normal, Flat, Nether all differ).
82  // The World calls this on first load to set the spawn point.
83  virtual mcp_vector3_t getSpawn() const = 0;
84 };
