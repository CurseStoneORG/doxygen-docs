25 // Physics.h — the server-side collision engine.
26 //
27 // This is the WORLD-FREE half of the physics port: pure geometry over
28 // the client's net.minecraft.world.phys model. It knows nothing about
29 // World, blocks or entities — the caller hands it the collision boxes
30 // once (World::collisionBoxes()) and it answers the two questions the
31 // client's own physics answer:
32 //
33 // sweep() — Entity::move(): where does this box end up when pushed
34 // by a motion vector through a set of obstacles, and did
35 // it land on the ground?
36 // clipRay() — Level::clip()/AABB::clip(): what block does this
37 // segment hit, on which face, and exactly where?
38 //
39 // Keeping these dependency-free is deliberate: tools/phys_test.cpp links
40 // ONLY Physics.cpp + the math headers and checks the numbers, instead of
41 // re-implementing the algorithm next to the real one (the way the
42 // dropped-item integrator used to be duplicated inline in World.cpp).
43 // ============================================================

54 // A block's collision box in world space, tagged with the block
55 // coordinates it came from — those coordinates are what makes a ray hit
56 // usable (breaking/placing/interacting all need the block, not just the
57 // point). Air and every walk-through block simply never show up here.
58 struct BlockBox {
59  int32_t x = 0, y = 0, z = 0;
60  mcp_aabb_t box{};
61 };
62 
63 // Result of one swept move. `motion` is the delta that was actually
64 // applied (each component clipped down to the free gap against whatever
65 // it ran into). The three `clipped_*` flags are the caller's cue to
66 // cancel that motion component, exactly as Entity::move() does.
67 struct MotionResult {
68  mcp_vector3_t motion{};
69  bool on_ground = false;
70  bool clipped_x = false;
71  bool clipped_y = false;
72  bool clipped_z = false;
73 };
74 
75 // Entity::move(box, motion): resolve the displacement against `boxes`
76 // one axis at a time — Y, then X, then Z, which is the order the client
77 // uses — mutating the working box between passes so the X pass already
78 // sees the Y offset (and the Z pass sees both). Behaviour is identical
79 // to the inline loops that used to live in World::tickDroppedItems().
80 //
81 // `on_ground` is Entity::checkGroundState(): true only when the DOWNWARD
82 // motion was clipped, i.e. `clipped_y && motion.y < 0`.
83 MotionResult sweep(mcp_aabb_t box, mcp_vector3_t motion, const std::vector<BlockBox>& boxes);
84 
85 // Nearest block hit along `from` -> `to` (segment, not infinite ray),
86 // with the block's coordinates and the face that was struck. Returns
87 // MCP_HIT_NONE when the segment reaches nothing. Ties are resolved by
88 // distance from `from`; boxes are not sorted by the caller.
89 mcp_hit_result_t clipRay(mcp_vector3_t from, mcp_vector3_t to, const std::vector<BlockBox>& boxes);
