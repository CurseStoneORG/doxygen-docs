23 // AsyncPool — N worker threads that drain a thread-safe task queue.
24 // Chunk generation, file I/O and packet compression post tasks here.
25 // Mirrors pocketmine/scheduler/AsyncPool.
26 //
27 // STABILITY FIXES applied here (see also ServerScheduler.cpp, which
28 // already wraps its own scheduleAsync() lambdas in try/catch — the
29 // fixes below make the pool itself safe for ANY caller, not just
30 // ServerScheduler):
31 // 1. submit() after stop() is now rejected instead of silently
32 // queuing work that will never run (and, before this fix, could
33 // race with workers already past their exit check).
34 // 2. Every unit of work is run inside try/catch. Previously, an
35 // exception thrown by `w()` would escape the worker thread's
36 // entry point. In C++, that calls std::terminate() and kills
37 // the ENTIRE process — not just the pool, not just the calling
38 // player, everything. This matters even though
39 // ServerScheduler::scheduleAsync() already catches internally,
40 // because AsyncPool is a general-purpose primitive: nothing
41 // about `submit()` guarantees every caller will remember to
42 // wrap its own lambda.
43 // 3. `running_` is now consistently read/written under `mtx_`
44 // everywhere (it already mostly was; `stop()` now also drains
45 // any tasks left in the queue instead of abandoning them, so
46 // shutdown never silently loses queued work — each is still run,
47 // just synchronously on the calling thread if workers have
48 // already been asked to stop, which only happens during a
49 // deliberate stop()/destructor call).
50 #pragma once
51 #include <vector>
52 #include <thread>
53 #include <queue>
54 #include <mutex>
55 #include <condition_variable>
56 #include <functional>
57 #include <atomic>
58 #include <exception>
59 #include <string>
60 #include "../utils/MainLogger.hpp"
61 namespace cursestone::scheduler {
62 class AsyncPool {
63 public:
64  explicit AsyncPool(int n_workers = 4) : n_(n_workers > 0 ? n_workers : 1) {
65  for (int i = 0; i < n_; ++i)
66  workers_.emplace_back([this]{ this->workerLoop(); });
67  }
68  ~AsyncPool() { stop(); }
69 
70  using Work = std::function<void()>;
71 
72  // Returns false (and does not queue) if the pool is already
73  // stopping/stopped. Callers that ignore the return value simply
74  // lose that one unit of work, which is safe — it's the same
75  // outcome as scheduleAsync()'s existing "no AsyncPool" fallback
76  // path, just surfaced with a return value instead of happening
77  // implicitly.
78  bool submit(Work w) {
79  std::unique_lock<std::mutex> lk(mtx_);
80  if (!running_) return false;
81  queue_.push(std::move(w));
82  cv_.notify_one();
83  return true;
84  }
85 
86  void stop() {
87  {
88  std::unique_lock<std::mutex> lk(mtx_);
89  if (!running_) return; // already stopped — idempotent
90  running_ = false;
91  }
92  cv_.notify_all();
93  for (auto& t : workers_) if (t.joinable()) t.join();
94 
95  // FIX: drain anything still left in the queue after workers
96  // have exited, instead of destroying the queue (and silently
97  // discarding pending work, e.g. an unflushed chunk write)
98  // along with the pool. Run on the calling thread (normally
99  // the main thread during shutdown) since no workers remain.
100  // Each item still gets the same exception protection as the
101  // normal worker path.
102  std::queue<Work> remaining;
103  {
104  std::unique_lock<std::mutex> lk(mtx_);
105  std::swap(remaining, queue_);
106  }
107  while (!remaining.empty()) {
108  runOne(remaining.front());
109  remaining.pop();
110  }
111  }
112 
113 private:
114  static void runOne(Work& w) {
115  if (!w) return;
116  try {
117  w();
118  } catch (const std::exception& e) {
119  utils::MainLogger::get().warn(
120  std::string("[AsyncPool] Task threw during execution: ") + e.what() +
121  " — task dropped, pool kept running.");
122  } catch (...) {
123  utils::MainLogger::get().warn(
124  "[AsyncPool] Task threw a non-std::exception during execution — task dropped, pool kept running.");
125  }
126  }
127 
128  void workerLoop() {
129  for (;;) {
130  Work w;
131  {
132  std::unique_lock<std::mutex> lk(mtx_);
133  cv_.wait(lk, [&] { return !running_ || !queue_.empty(); });
134  if (!running_ && queue_.empty()) return;
135  if (queue_.empty()) continue; // spurious wake / stop with items still draining elsewhere
136  w = std::move(queue_.front());
137  queue_.pop();
138  }
139  runOne(w);
140  }
141  }
142  int n_;
143  std::vector<std::thread> workers_;
144  std::queue<Work> queue_;
145  std::mutex mtx_;
146  std::condition_variable cv_;
147  bool running_{true}; // always accessed under mtx_
148 };
149 } // namespace
