23 // HandlerList — el conjunto de listeners registrados para UN tipo
24 // concreto de evento, organizados en buckets por EventPriority. Cada
25 // tipo de evento tiene su propio HandlerList, administrado por
26 // EventManager (ver EventManager.h).
27 //
28 // STABILITY CONTRACT (this replaces an earlier version that never got
29 // wired to the rest of the code and had two latent bugs: dispatch()
30 // took the event as `const Event&`, so no listener could ever call
31 // setCancelled(); and EventHandler.h's priority/ignoreCancelled fields
32 // were never consulted anywhere):
33 //
34 // - dispatch() takes a non-const Event& so listeners can both read
35 // and modify the event (cancel it, rewrite a chat message, etc).
36 // - Listeners run in EventPriority order (LOWEST first, MONITOR
37 // last), so higher priorities see what lower ones produced and
38 // MONITOR listeners always see the final state.
39 // - A listener registered with ignoreCancelled=true is skipped as
40 // soon as the event was already cancelled by an earlier listener —
41 // replica RegisteredListener::callEvent() del servidor de referencia.
42 // - A listener that throws is caught, logged and skipped: a broken
43 // handler can never take down the tick loop nor block the rest
44 // (same stability doctrine as the post-fire-cascade scheduler).
45 // - All mutations (register/unregister) are mutex-protected, so it
46 // can be used from any thread. dispatch() snapshots under the lock
47 // and runs callbacks outside of it, so a handler can (un)register
48 // listeners without deadlocking against itself.

52 #include "RegisteredListener.h"
53 #include <array>
54 #include <vector>
55 #include <mutex>

57 #include <functional>

61 class HandlerList {
62  public:
63  // Registra `callback` en la prioridad dada. Devuelve un id de
64  // registro utilizable con unregister(). Devuelve 0 si callback
65  // is empty (nothing to register).
66  uint64_t registerListener(void* owner, EventPriority priority, bool ignoreCancelled,
67  std::function<void(Event&)> callback);
68 
69  // Removes a single registration by id. true if found.
70  bool unregister(uint64_t id);
71 
72  // Elimina todos los registros cuyo `owner` coincida. Devuelve
73  // how many were removed. Used when a Listener is destroyed.
74  size_t unregisterOwner(void* owner);
75 
76  // Elimina absolutamente todos los registros.
77  void unregisterAll();
78 
79  // Ejecuta cada listener registrado para `event`, en orden de
80  // priority, skipping ignoreCancelled handlers once the event is
81  // cancelado. Segura de llamar mientras otros hilos registran o
82  // desregistran listeners concurrentemente.
83  void dispatch(Event& event) const;
84 
85  size_t size() const;
86 
87  private:
88  mutable std::mutex mtx_;
89  std::array<std::vector<RegisteredListener>, kEventPriorityCount> slots_;
90  uint64_t next_id_ = 1;
91 };
