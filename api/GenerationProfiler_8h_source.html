23 // GenerationProfiler — where generation time actually goes.
24 //
25 // Every stage of the chunk pipeline (terrain noise, caves, populate,
26 // light, wire encode, disk load) accumulates nanoseconds and call
27 // counts into lock-free counters; PerformanceManager formats a
28 // per-chunk breakdown out of them. The overhead is two steady_clock
29 // reads plus two atomic adds per scoped stage — noise next to the
30 // milliseconds of real work a chunk costs.
31 //
32 // gen.noise, gen.surface and gen.caves are measured INSIDE
33 // gen.terrain; gen.copy_in/gen.copy_out run inside gen.populate. The
34 // numbers overlap on purpose: read them as "what does each part cost",
35 // not as a partition of gen.total.
36 #pragma once

39 #include <chrono>
40 #include <cstdint>

44 enum class Stage : uint8_t {
45  Total, // whole chunk task: terrain + populate + light
46  TerrainBuild, // terrain-only chunk build (prepareHeights..carve)
47  Noise, // Perlin octave regions inside prepareHeights
48  Surface, // biome surface patches
49  Caves, // LargeCaveFeature carving
50  BiomeFill, // biome layer stack sampling
51  Populate, // features, ores, structures, snow
52  CopyIn, // real chunk -> virtual working copy
53  CopyOut, // virtual volume -> real chunk sections
54  Light, // sky light propagation
55  Encode, // chunk wire transposition + zlib (worker)
56  Snapshot, // chunk wire snapshot (tick thread)
57  DiskLoad, // provider loadChunk (region read + inflate + NBT)
58  Lakes, // water + lava lake placement
59  Structures, // village / mineshaft / scattered temple passes
60  Dungeons, // monster rooms
61  Freeze, // freeze tops scan
62  Decorate, // biome decoration umbrella
63  DecorateOres, // ore vein placement
64  DecorateSurface, // sand / clay / gravel / mushroom blobs
65  DecorateTrees, // trees + tall grass around them
66  DecorateMisc, // flowers, tall grass, sugar cane, pumpkins...
67  Snow, // snow cover pass
68  Count
69 };

71 class GenerationProfiler {
72  public:
73  struct Entry {
74  uint64_t ns = 0;
75  uint64_t calls = 0;
76  };
77  struct Snapshot {
78  Entry stages[(size_t)Stage::Count];
79  };
80 
81  // Timers are on by default; the cost is a few atomic adds per chunk.
82  static bool enabled() {
83  return enabled_.load(std::memory_order_relaxed);
84  }
85  static void setEnabled(bool on) {
86  enabled_.store(on, std::memory_order_relaxed);
87  }
88 
89  static void add(Stage s, uint64_t ns) {
90  if (!enabled())
91  return;
92  auto& e = stages_[(size_t)s];
93  e.ns.fetch_add(ns, std::memory_order_relaxed);
94  e.calls.fetch_add(1, std::memory_order_relaxed);
95  }
96  // Atomically drain the counters: concurrent stages keep adding into
97  // the fresh counters, so a snapshot is "at least everything up to
98  // now" rather than an exact instant — fine for periodic reporting.
99  static Snapshot drain() {
100  Snapshot out;
101  for (size_t i = 0; i < (size_t)Stage::Count; ++i) {
102  out.stages[i].ns = stages_[i].ns.exchange(0, std::memory_order_relaxed);
103  out.stages[i].calls =
104  stages_[i].calls.exchange(0, std::memory_order_relaxed);
105  }
106  return out;
107  }
108  // Non-destructive read for reports that drain elsewhere (or diff two
109  // reads to describe a window without consuming anything).
110  static Snapshot peek() {
111  Snapshot out;
112  for (size_t i = 0; i < (size_t)Stage::Count; ++i) {
113  out.stages[i].ns = stages_[i].ns.load(std::memory_order_relaxed);
114  out.stages[i].calls =
115  stages_[i].calls.load(std::memory_order_relaxed);
116  }
117  return out;
118  }
119 
120  private:
121  struct AtomicEntry {
122  // Zero-initialized by static storage; no NSDMI here so the
123  // stages_ array can be declared inside the class body.
124  std::atomic<uint64_t> ns;
125  std::atomic<uint64_t> calls;
126  };
127  static inline std::atomic<bool> enabled_{true};
128  static inline AtomicEntry stages_[(size_t)Stage::Count];
129 };

131 // Times one scope into one stage. Cheap enough to sprinkle freely.
132 class ScopedStage {
133  public:
134  explicit ScopedStage(Stage s)
135  : stage_(s), start_(GenerationProfiler::enabled()
136  ? std::chrono::steady_clock::now()
137  : TimePoint{}) {}
138  ~ScopedStage() {
139  if (!GenerationProfiler::enabled())
140  return;
141  const auto end = std::chrono::steady_clock::now();
142  GenerationProfiler::add(stage_,
143  (uint64_t)std::chrono::duration_cast<
144  std::chrono::nanoseconds>(end - start_)
145  .count());
146  }
147 
148  private:
149  using TimePoint = std::chrono::steady_clock::time_point;
150  Stage stage_;
151  TimePoint start_;
152 };
