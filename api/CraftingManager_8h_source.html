42  // Registra TODAS las recetas de MCPE 0.15.10 (protocolo 84)
43  // llamando shaped()/shapeless()/furnace() desde
44  // registerBuiltinRecipes(), en el .cpp. Devuelve cuantas quedaron.
45  //
46  // El set completo vive en codigo: no hay ningun archivo de recetas
47  // que cargar. Antes se leia uno externo que no se empaqueta con el
48  // binario, asi que una release limpia arrancaba con 0 recetas y el
49  // crafting quedaba dead. Estilo alineado con
50  // Nukkit, que declara todas las recetas en
51  // su CraftingManager.
52  //
53  // Para agregar o cambiar una receta se edita registerBuiltinRecipes()
54  // en el .cpp; no hay ninguna ruta de carga externa.
55  int registerBuiltinRecipes();
56 
57  // Finds the matching shaped recipe for a 3x3 grid (item ids only).
58  // grid() de la receta tambien esta zero-padded, asi que la
59  // comparacion es directa. Ignora recipes con "any damage" en
60  // alguna celda (para esas usar matchShapedWithMeta).
61  const Recipe* matchShaped(const std::array<uint16_t, 9>& grid) const;
62 
63  // Igual que matchShaped pero validando tambien el meta de cada
64  // celda (grid_meta); las celdas de receta con meta -1 aceptan
65  // cualquier damage del jugador.
66  const Recipe* matchShapedWithMeta(const std::array<uint16_t, 9>& grid,
67  const std::array<uint16_t, 9>& grid_meta) const;
68 
69  // Finds the matching shapeless recipe for an ingredient list.
70  // `ingredients` debe venir ordenado ascendente (ids). Ignora
71  // recipes con "any damage" (usar matchShapelessWithMeta).
72  const Recipe* matchShapeless(const std::vector<uint16_t>& ing) const;
73 
74  // Igual que matchShapeless validando meta por ingrediente
75  // (-1 en la receta = cualquier damage).
76  const Recipe* matchShapelessWithMeta(const std::vector<uint16_t>& ing,
77  const std::vector<uint16_t>& meta) const;
78 
79  // Finds furnace result for a given input id (any damage).
80  const Recipe* matchFurnace(uint16_t input_id) const;
81 
82  // Igual que matchFurnace pero exigiendo damage exacto cuando la
83  // receta lo declara (meta != -1). Usado por TileFurnace, donde el
84  // input tiene meta concreto.
85  const Recipe* matchFurnaceWithMeta(uint16_t input_id, uint16_t meta) const;

87  // Busqueda por los 16 bytes del UUID de receta (binario, mismo
88  // orden que el que escribio CraftingDataPacket). El cliente los

91  const Recipe* getRecipeByUuidBytes(const uint8_t* uuid16) const;

105  private:
106  // Helpers de registro — los usa registerBuiltinRecipes() para
107  // escribir cada receta como una linea legible en vez de construir
108  // los Recipe a mano con los setters.
109  //
110  // shaped(): `grid` son las 9 celdas en orden row-major (3x3
111  // zero-padded). Cada celda es {item_id, meta}; id 0 = celda vacia
112  // y meta -1 = "cualquier damage", que es lo que matchShaped() y
113  // matchShapedWithMeta() esperan. w x h es el bounding box real del
114  // patron: lo demas de la grilla tiene que estar vacio.
115  //
116  // Se toma initializer_list y no std::array<std::pair<...>,9> porque
117  // std::pair no es un aggregate: no elide los braces internos y una
118  // lista de 9 pares a mano necesita un nivel extra de llaves, lo que
119  // haria ilegible el dataset. Con initializer_list cada celda se
120  // escribe plano: {{5,-1},{5,-1},{0,-1}, ...}.
121  void shaped(int w, int h, std::initializer_list<std::pair<uint16_t, int16_t>> grid,
122  cursestone::item::ItemStack result);
123 
124  // shapeless(): lista de ingredientes sin orden en la grilla.
125  void shapeless(std::initializer_list<std::pair<uint16_t, int16_t>> ing,
126  cursestone::item::ItemStack result);
127 
128  // furnace(): input_meta -1 = "cualquier damage" (lo que
129  // matchFurnace() trata como ENTRY_FURNACE en el wire).
130  void furnace(uint16_t input_id, int16_t input_meta, cursestone::item::ItemStack result);
131 
132  // Asigna UUID deterministas (1, 2, 3...) a todas las recetas
133  // registradas y construye el indice por bytes. Se llama una vez al
134  // terminar de registrar, en registerBuiltinRecipes().
135  void buildIndex();
136 
137  std::vector<std::unique_ptr<Recipe>> recipes_;
138 
139  // Indice uuid(16 bytes crudos) -> receta, construido por
140  // buildIndex(). El array de 16 bytes como clave via array<uint8_t,16>
141  // (hashable con un functor propio).
142  struct Uuid16Hash {
143  size_t operator()(const std::array<uint8_t, 16>& k) const {
144  size_t h = 1469598103934665603ull; // FNV offset basis
145  for (uint8_t b : k) {
146  h ^= b;
147  h *= 1099511628211ull;
148  }
149  return h;
150  }
151  };
152  std::unordered_map<std::array<uint8_t, 16>, Recipe*, Uuid16Hash> by_uuid_;
153 };
