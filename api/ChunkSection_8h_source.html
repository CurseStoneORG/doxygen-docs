24 // ChunkSection — standalone helpers for a single 16×16×16 sub-chunk.
25 //
26 // A `mcp_chunk_section_t` is the canonical sub-chunk: 4096 bytes of
27 // block IDs, plus three 2048-byte nibble arrays for block data,
28 // sky light, and block light. Chunk.h includes this header so
29 // Chunk can stack 16 of these on Y.
30 //
31 // Pure C, no global state. Small index/nibble helpers are inlined;
32 // larger allocators live in ChunkSection.c.

43 // Opaque struct — the layout is defined in ChunkSection.c.
44 // `mcp_chunk_section_t` is the public alias; users go through
45 // the accessors below and never touch the fields directly.
46 typedef struct mcp_chunk_section {
47  uint8_t y_index; // 0..15 — which sub-Y this is.
48  uint8_t* blocks; // 4096 bytes (XZY order)
49  uint8_t* data; // 2048 bytes (4-bit nibbles)
50  uint8_t* sky_light; // 2048 bytes (4-bit nibbles)
51  uint8_t* block_light; // 2048 bytes (4-bit nibbles)
52  bool non_air_top; // true if upper half has any non-air block
53 } mcp_chunk_section_t;

55 // XZY coordinate packing: idx = (x&15) | ((z&15)<<4) | ((y&15)<<8).
56 // Range 0..4095 for blocks, 0..2047 for nibble arrays.
57 //
58 // NOTE (CurseStone): the nibble index MUST use the same shift-8 packing
59 // as Chunk.c's nibble_index()/get_nibble()/set_nibble() — byte =
60 // 128*y + 8*z + x/2, low nibble for even x, high nibble for odd x.
61 // An earlier revision used (y<<7) here, which overlaps with the z bits
62 // ((x=1,z=8,y=0) and (x=1,z=0,y=1) both map to 129) and desynchronizes
63 // the whole data array for every y >= 1: blocks written through these
64 // helpers came back with the wrong data (or none) when read through
65 // mcp_chunk_get_block_data, the save path and mcp_chunk_network_serialize.
66 // Visible effect: spruce/leaves/snow data nibbles above the first
67 // section row read back as 0 — e.g. taiga trees with oak trunks.
68 static inline uint32_t mcp_section_block_index(int x, int y, int z) {
69  return (uint32_t)((x & 15) | ((z & 15) << 4) | ((y & 15) << 8));
70 }
71 
72 static inline uint32_t mcp_section_nibble_index(int x, int y, int z) {
73  return (uint32_t)((x & 15) | ((z & 15) << 4) | ((y & 15) << 8));
74 }
75 
76 static inline uint8_t mcp_section_get_nibble(const uint8_t* arr, int x, int y, int z) {
77  if (!arr)
78  return 0;
79  uint32_t i = mcp_section_nibble_index(x, y, z);
80  uint8_t b = arr[i >> 1];
81  return (i & 1) ? (uint8_t)((b >> 4) & 0x0F) : (uint8_t)(b & 0x0F);
82 }
83 
84 static inline void mcp_section_set_nibble(uint8_t* arr, int x, int y, int z, uint8_t v) {
85  if (!arr)
86  return;
87  v &= 0x0F;
88  uint32_t i = mcp_section_nibble_index(x, y, z);
89  uint8_t* b = &arr[i >> 1];
90  if (i & 1) {
91  *b = (uint8_t)((*b & 0x0F) | (v << 4));
92  } else {
93  *b = (uint8_t)((*b & 0xF0) | v);
94  }
95 }
