24 // DoubleChestInventory — the vanilla large chest (InventoryLargeChest):
25 // two 27-slot chest halves addressed as one 54-slot container.
26 //
27 // Exactly like the vanilla reference, this wrapper owns NO storage of
28 // its own. Every slot read/write is forwarded to one of the two backing
29 // ChestInventory halves:
30 // slots 0-26 → left half
31 // slots 27-53 → right half
32 // Keeping the two TileChest halves as the single source of truth is what
33 // makes the rest vanilla too: each half persists its own 27 slots to its
34 // own tile NBT, and breaking one half drops only that half's slots while
35 // the other half keeps its items (no duplication, no re-indexing).
36 #pragma once
37 #include "BaseInventory.h"
38 #include "ChestInventory.h"
39 #include <memory>
