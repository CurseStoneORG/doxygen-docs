24 // LightEngine — block light that behaves like vanilla, on top of a
25 // GLOBAL sky layer.
26 //
27 // Sky light is global everywhere (the Aquamarine populateSkyLight()
28 // semantics: every block reads 15, caves and oceans included — see
29 // mcp_chunk_propagate_sky_light in the chunk layer). Nothing occludes
30 // it, so this engine never writes the sky channel: no decay, no flood,
31 // no relight of it. That removed the dominant per-chunk light cost and
32 // the "dark stains while flying" class of bugs with it.
33 //
34 // What is left to do here is the BLOCK layer (torches, glowstone,
35 // burning furnaces), in two kinds of work:
36 //
37 // * FULL sweep (relightChunk): clear a chunk PLUS one light range
38 // into every neighbour and re-flood the whole box from scratch.
39 // Reserved for chunks whose BLOCKS changed: World::setBlock and
40 // generation spill batches.
41 //
42 // * BORDER entry (runBorderEntry): everything else — a chunk that
43 // just arrived with its interior already lit by the generation
44 // worker. Three cheap monotone steps: borderPull raises the
45 // chunk's cells from its neighbours' current light, pushBorder
46 // raises neighbour cells the chunk can improve, and interiorFlood
47 // spreads light a neighbour deposited mid-chunk (flagged via
48 // light_touched). Values only ever RISE and are capped at 15, so
49 // the pass converges to the same fixed point the full sweep
50 // reaches, without ever re-wiping a healthy neighbour.
51 //
52 // A wall-clock budget in processPending keeps the main-thread tick
53 // flat no matter how fast chunks arrive; flushPending finalizes one
54 // chunk's queued entry on demand (the chunk-send path uses it so a
55 // protocol-84 client never receives a chunk whose border pass has not
56 // run yet — clients keep the light bytes they get).

67 namespace cursestone::world {

69 class World;

71 class LightEngine {
72  public:
73  // Light falls off after 15 steps, so a region 15 blocks wider than the
74  // change on every side can never miss a source further away: the
75  // recomputation is self-contained.
76  static constexpr int32_t kRelightRadius = 15;
77  // A chunk whose blocks changed needs its neighbours relit too, so
78  // the sweep extends one full light range past the chunk border.
79  static constexpr int32_t kChunkRelightRadius = 15;
80 
81  // Chunks relit per server tick at most, so loading a view full of
82  // chunks never stalls the main loop in one giant burst.
83  static constexpr int32_t kPendingBudget = 2;
84  // Wall-clock cap for the per-tick light drain (processPending stops
85  // before STARTING another entry once this many ms are spent; an
86  // entry that already started always completes).
87  static constexpr int32_t kTickBudgetMs = 6;
88 
89  explicit LightEngine(World& lvl) : lvl_(lvl) {
90  }
91 
92  // Recompute both layers in a cube around a block change. The home
93  // chunk is (x >> 4, z >> 4); every other chunk the sweep writes is
94  // flagged (light_touched) and re-queued for a border entry, so
95  // light the sweep deposited in a neighbour never freezes.
96  void relightAround(int32_t x, int32_t y, int32_t z, int32_t radius);
97 
98  // Same sweep, centred on a chunk, used when a chunk's blocks
99  // changed (generation spill batch) so the rewrite covers one full
100  // light range into every neighbour.
101  void relightChunk(int32_t cx, int32_t cz, int32_t radius);
102 
103  // Called from the chunk worker thread when a chunk enters the
104  // manager. Only queues the chunk — doing real work here would block
105  // generation and risk lock inversion with ChunkManager::mtx_.
106  // `full` selects the entry kind: true = blocks changed under this
107  // chunk (spill batch / the tools' re-signal), false = the chunk
108  // merely arrived with worker-lit interior (a BORDER entry).
109  void onChunkAdded(int32_t cx, int32_t cz, bool full = true);
110 
111  // Drain up to `budget` queued chunks, stopping before starting
112  // another entry once `max_ms` (if > 0) is spent. Returns how many
113  // entries were completed.
114  size_t processPending(int32_t budget, int32_t max_ms = 0);
115 
116  // Run the queued entry for exactly this chunk right now (removing
117  // it from the queue). Returns false when nothing is queued for it.
118  // The chunk-send path calls this so a chunk never leaves with its
119  // border pass still pending.
120  bool flushPending(int32_t cx, int32_t cz);
121 
122  bool hasPending() const;
123 
124  // One exact clearing re-flood over the whole box: clear + reseed +
125  // flood, no queueing, no home chunk, no neighbour re-queues. This is
126  // the ground-truth fixed point the convergence judge
127  // (tools/light_oracle.cpp) compares the FULL and BORDER drains
128  // against.
129  void relightBoxExact(int32_t x0, int32_t y0, int32_t z0, int32_t x1, int32_t y1,
130  int32_t z1) {
131  relight(x0, y0, z0, x1, y1, z1, INT32_MIN, 0);
132  }
133 
134  private:
135  struct PendingEntry {
136  int32_t cx, cz;
137  bool full;
138  };
139 
140  World& lvl_;
141  mutable std::mutex mtx_;
142  std::deque<PendingEntry> pending_;
143  // Same members as pending_, as a set: population storms used to call
144  // onChunkAdded for the same chunk once per spilled block, and the
145  // duplicate check scanned the whole deque each time (O(n) per block).
146  // A queued Border entry is UPGRADED to Full when a spilled-block
147  // batch lands on the same chunk before its turn.
148  std::unordered_set<uint64_t> pending_keys_;
149 
150  // Runs one full region recomputation for both layers. When
151  // (homeCx, homeCz) is a real chunk, every OTHER loaded chunk the
152  // box touches is flagged light_touched and re-queued as a border
153  // entry; pass homeCx = INT32_MIN to disable (the oracle's exact
154  // global re-flood).
155  void relight(int32_t x0, int32_t y0, int32_t z0, int32_t x1, int32_t y1, int32_t z1,
156  int32_t homeCx = INT32_MIN, int32_t homeCz = 0);
157 
158  // One BORDER entry: pull neighbours' light in, spread mid-chunk
159  // deposits, push our light out, re-queue neighbours whose cells the
160  // push raised.
161  void runBorderEntry(int32_t cx, int32_t cz);
162  void borderPull(int32_t cx, int32_t cz);
163  void pushBorder(int32_t cx, int32_t cz, std::vector<std::pair<int32_t, int32_t>>* touched);
164  void interiorFlood(int32_t cx, int32_t cz);
165 
166  // Queue helper shared by runBorderEntry / relight's re-queue.
167  void queueEntry(int32_t cx, int32_t cz, bool full);
168 };
