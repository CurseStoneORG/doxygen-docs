24 // WorldProviderManager — registry of on-disk world formats. C++.
25 //
26 // The reference server's WorldProviderManager is a static registry that maps a
27 // format identifier string (e.g. "anvil", "leveldb") to a concrete
28 // WorldProvider class. We mirror that pattern, but the C++ version is
29 // a non-static class so test code can inject a fresh registry per
30 // test.
31 //
32 // Concrete providers register themselves with the manager at process
33 // startup (see anvil/Anvil.cpp's static initializer). The World loader
34 // then asks the manager for a provider by format name; if absent it
35 // falls back to the default ("anvil").
36 #pragma once
37 
38 #include <functional>
39 #include <memory>
40 #include <string>
41 #include <unordered_map>
42 #include <vector>
43 #include "../../CurseStone.h"
44 #include "WorldProvider.h"
45 
46 namespace cursestone::world::format {
47 
48 // Factory callback. Creates a fresh provider instance.
49 using ProviderFactory = std::function<std::unique_ptr<WorldProvider>()>;
50 
51 class WorldProviderManager {
52  public:
53  static WorldProviderManager& getInstance();
54 
55  // Register a provider by canonical name. Names are lower-case
56  // ASCII; the manager stores them as-is and matches case-sensitively.
57  // Returns false if the name is already taken.
58  bool registerProvider(const std::string& name, ProviderFactory factory);
59 
60  // Look up a factory by name. Returns nullptr if unknown.
61  ProviderFactory getFactory(const std::string& name) const;
62 
63  // Convenience: instantiate a provider by name, returning nullptr
64  // on unknown name. The returned pointer is owned by the caller.
65  std::unique_ptr<WorldProvider> create(const std::string& name) const;
66 
67  // List every registered format name (sorted for stable output).
68  std::vector<std::string> listFormats() const;
69 
70  // Default format used when a config doesn't specify one.
71  static const char* defaultFormatName();
72 
73  private:
74  WorldProviderManager() = default;
75  std::unordered_map<std::string, ProviderFactory> registry_;
76 };
