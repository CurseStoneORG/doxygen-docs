28 // through these fields, so state mutations stay on the main thread

120  // ---- Knockback carry-over (decayed by the actuation step) ----
121  // Horizontal shove written when a hit's record knockback is consumed;
122  // the actuation adds it to the sweep, decays it every tick and zeroes
123  // it below the stop epsilon, so a mob keeps sliding briefly and stops.
124  float& motionX() {
125  return motion_x_;
126  }
127  float& motionZ() {
128  return motion_z_;
129  }
131  bool hasKnockback() const {
132  return motion_x_ != 0.0f || motion_z_ != 0.0f;
133  }
136  void decayKnockback(float friction) {
137  motion_x_ *= friction;
138  motion_z_ *= friction;
139  if (motion_x_ > -kKnockbackStop && motion_x_ < kKnockbackStop)
140  motion_x_ = 0.0f;
141  if (motion_z_ > -kKnockbackStop && motion_z_ < kKnockbackStop)
142  motion_z_ = 0.0f;
143  }
