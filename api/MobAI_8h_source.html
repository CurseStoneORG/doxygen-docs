24 // MobBrain — every per-mob dynamic AI state in one place: the owned
25 // goal list, the movement intent the active goals produced, and the
26 // physics carry-over the actuation step integrates. A brain holds NO
27 // pointers into the world; goals and actuation communicate only
28 // through these fields, so state mutations stay on the main thread
29 // (docs/MOB_AI.md).

111  // ---- Knockback carry-over (decayed by the actuation step) ----
112  // Horizontal shove written when a hit's record knockback is consumed;
113  // the actuation adds it to the sweep, decays it every tick and zeroes
114  // it below the stop epsilon, so a mob keeps sliding briefly and stops.
115  float& motionX() {
116  return motion_x_;
117  }
118  float& motionZ() {
119  return motion_z_;
120  }
122  bool hasKnockback() const {
123  return motion_x_ != 0.0f || motion_z_ != 0.0f;
124  }
127  void decayKnockback(float friction) {
128  motion_x_ *= friction;
129  motion_z_ *= friction;
130  if (motion_x_ > -kKnockbackStop && motion_x_ < kKnockbackStop)
131  motion_x_ = 0.0f;
132  if (motion_z_ > -kKnockbackStop && motion_z_ < kKnockbackStop)
133  motion_z_ = 0.0f;
134  }
