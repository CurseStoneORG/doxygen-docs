24 // VillagePieces — the 1.9-era village piece set as code.
25 //
26 // The templates live in VillagePieces.cpp (GENERATED from the
27 // CurseStone-Structures village exports by tools/gen_village_pieces.py);
28 // this header is the stable API the populator talks to: the piece kind
29 // list, the concrete Structure subclass that carries a spec palette,
30 // and the accessor that hands pieces out.
31 //
32 // The piece set is the full vanilla old-village inventory: the well,
33 // four farms (well-adjacent small + large), five house tiers, the
34 // library, the church, the blacksmith, the butcher shop and the street
35 // lamp. VillagePopulator assembles them around the well the way
36 // StructureVillagePieces.Start does — center building, roads on the
37 // four cardinal axes, houses branching off the roads.

44 // One concrete structure carrying a BlockSpec palette. The per-biome
45 // material choice happens in VillagePalette::resolve at placement time;
46 // the template itself is biome-neutral.
47 //
48 // frontFacing is the java compass (1 east, 2 west, 3 south, 4 north)
49 // the piece's ENTRANCE door opens toward once placed — the door data
50 // itself faces that way (the exports store the inward facing, which
51 // gen_village_pieces.py flips). 0 means the export has no outside door
52 // (the tower houses, the farms, the lamp and the well), and such a
53 // piece is rotated freely.
54 class VillagePiece final : public Structure {
55  public:
56  VillagePiece(const char* name, int sizeX, int sizeY, int sizeZ, const BlockSpec* specs,
57  size_t specCount, const char* volume, int frontFacing)
58  : Structure(name, sizeX, sizeY, sizeZ, specs, specCount, volume),
59  frontFacing_(frontFacing) {
60  }
61 
62  int frontFacing() const {
63  return frontFacing_;
64  }
65 
66  private:
67  int frontFacing_;
68 };
