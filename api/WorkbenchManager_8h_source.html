24 // WorkbenchManager.h — comprehensive server diagnostics system.
25 //
26 // 6 modules:
27 // 1. TPS Monitor — tracks real TPS every second
28 // 2. Tick Profiler — times each phase of Server::tick()
29 // 3. Packet Logger — logs every IN/OUT packet
30 // 4. RakLib Stats — atomic counters for send/recv/retransmit
31 // 5. Network Watchdog — separate thread detecting main-thread stalls
32 // 6. Disconnect Crash Dump — captures state when a player disconnects
33 //
34 // Usage:
35 // WorkbenchManager::get().start(); // via /workbench start
36 // WorkbenchManager::get().stop(); // via /workbench stop → exports reports
37 // WorkbenchManager::get().report(); // via /workbench report → snapshot export
38 // WorkbenchManager::get().status(); // via /workbench status → console output
39 //
40 // Hook points (called from Server::tick):
41 // WorkbenchManager::get().onTickStart();
42 // WorkbenchManager::get().onTickPhase(phase_name, duration_ms);
43 // WorkbenchManager::get().onTickEnd(total_ms);
44 // WorkbenchManager::get().onPlayerUpdate(player_name, duration_ms);
45 // WorkbenchManager::get().onPacketIn(ip, port, pid, size);
46 // WorkbenchManager::get().onPacketOut(player_name, pid, size);
47 // WorkbenchManager::get().onDisconnect(player_name, reason, session_ms);
