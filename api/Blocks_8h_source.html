1 // CurseStone | vgen — block table shim.
2 //
3 // The restored vanilla generator references Mojang's Block registry
4 // (`Block::mStone->mID`, `Block::mBlocks[id]->getMaterial()`...). This
5 // header re-declares that registry surface on top of the MCPE 0.15
6 // (protocol 84) block palette, so the ported generator code needed no
7 // changes: every constant keeps its vanilla name and stores the 0.15
8 // numeric id it maps to.
9 //
10 // Downport notes (1.0.0 -> 0.15): the 1.0 BlockState model is emulated
11 // for the handful of states the generator writes (log axis/type, dirt
12 // coarse bit, vine growth, tripwire attach) — each maps onto plain
13 // 0.15 metadata bits.

26 // BlockState shim — 1.0 wrote `block->getBlockState(BlockState::X).set(data, v)`.
27 // On 0.15 every state is plain metadata, so the shim folds the value into
28 // `data` at the state's bit position using a per-(block,state) table.
29 enum class BlockState : int {
30  MappedType, // log type / dirt coarse bit — low bits
31  Direction, // log axis (<<2) or tripwire facing (<<0)
32  VineGrowth, // vine face bits 1/2/4/8 — identity
33  AttachedBit, // tripwire attached (<<2)
34 };

36 class BlockStateRef {
37 public:
38  BlockStateRef(uint8_t shift, uint8_t mask) : mShift(shift), mMask(mask) {}
39  void set(DataID& data, int value) const {
40  data = (uint8_t)((data & ~(mMask << mShift)) | ((value & mMask) << mShift));
41  }
42 
43 private:
44  uint8_t mShift, mMask;
45 };
