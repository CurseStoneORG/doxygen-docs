24 // AnvilChunkCodec.h — encodes/decodes mcp_chunk_t* to/from the Nukkit
25 // Anvil chunk NBT payload (big-endian, ZLIB-compressed separately).
26 //
27 // Ported from Nukkit InfinityEdition's
28 // `cn.nukkit.level.format.anvil.Chunk` (toBinary / fromBinary).
29 //
30 // On-disk chunk NBT layout (after ZLIB inflation):
31 //
32 // Root CompoundTag (name "")
33 // └── "Level" : CompoundTag
34 // ├── "xPos" : Int
35 // ├── "zPos" : Int
36 // ├── "LastUpdate" : Long
37 // ├── "InhabitedTime" : Long
38 // ├── "vxl" : Byte (CurseStone-specific marker)
39 // ├── "TerrainGenerated": Byte (0/1)
40 // ├── "TerrainPopulated": Byte (0/1)
41 // ├── "LightPopulated" : Byte (0/1)
42 // ├── "Sections" : ListTag<CompoundTag>
43 // │ each: { "Y":Byte, "Blocks":ByteArray(4096),
44 // │ "Data":ByteArray(2048),
45 // │ "BlockLight":ByteArray(2048),
46 // │ "SkyLight":ByteArray(2048) }
47 // ├── "BiomeColors" : IntArray(256) (BE ints, RGBA-ish)
48 // ├── "HeightMap" : IntArray(256) (BE ints)
49 // ├── "ExtraData" : ByteArray (var-int count + entries)
50 // ├── "Entities" : ListTag<CompoundTag>
51 // └── "TileEntities" : ListTag<CompoundTag>
52 //
53 // The CurseStone chunk has 16 sections (256-tall). We persist only the
54 // lower 8 sections (Y 0..7, i.e. block Y 0..127) since MCPE 0.15.x is
55 // 128-tall — matching what Nukkit stores in its Anvil format. Upper
56 // sections are dropped on save and zero-initialized on load.

69 // Returns NULL on failure. *out_len receives the byte count.
