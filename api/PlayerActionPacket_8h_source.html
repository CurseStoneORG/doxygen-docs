24 // PlayerActionPacket.h — client→server action notification (start/abort/stop break).
25 //
26 // Wire format (protocol 84) — the reference server PlayerActionPacket::decode:
27 // [1 byte ID 0x20]
28 // [Int64 BE] eid (legacy entity id; client echoes its own)
29 // [Int32 BE] action (0=START_BREAK, 1=ABORT_BREAK, 2=STOP_BREAK, ...)
30 // [Int32 BE] x
31 // [Int32 BE] y
32 // [Int32 BE] z
33 // [Int32 BE] face

35 
36 #include "DataPacket.h"
37 #include <vector>
38 #include <cstdint>
39 
40 namespace cursestone::network::protocol {
41 
42 class PlayerActionPacket : public DataPacket {
43  public:
44  uint8_t pid() const override {
45  return PLAYER_ACTION_PACKET;
46  }
47 
48  // Action constants (subset of the reference server PlayerAction:: constants relevant
49  // for block breaking on protocol 84).
50  static constexpr int32_t ACTION_START_BREAK = 0;
51  static constexpr int32_t ACTION_ABORT_BREAK = 1;
52  static constexpr int32_t ACTION_STOP_BREAK = 2;
53  // ACTION_RELEASE_ITEM = 5 — releasing a drawn bow (the reference server
54  // PlayerActionPacket — protocol 84 value; the 22 that later
55  // protocol versions use is WRONG here and the client never sends it).
56  // The reference server Player maps this action to the bow release → Arrow
57  // spawn flow.
58  static constexpr int32_t ACTION_RELEASE_ITEM = 5;
59  // ACTION_STOP_SLEEPING = 6 (the reference server PlayerActionPacket) — the
60  // 0.15.10 client sends this when the player taps "Leave bed" (or the
61  // screen/jump button) while sleeping in a bed. Maps to
62  // Player::stopSleep() exactly like the reference server Player.
63  static constexpr int32_t ACTION_STOP_SLEEPING = 6;
64  // El cliente manda esto cuando el jugador toca el boton de
65  // "Respawn" en la pantalla de muerte (the reference server:
66  // ACTION_SPAWN_SAME_DIMENSION = 7).
67  static constexpr int32_t ACTION_SPAWN_SAME_DIMENSION = 7;
68  // Sprint toggle (protocol 84 — the reference server PlayerActionPacket:
69  // ACTION_START_SPRINT = 9, ACTION_STOP_SPRINT = 10. Values 5/6 are
70  // RELEASE_ITEM/STOP_SLEEPING in this protocol — using them here made
71  // the server misread those actions and never see real sprints).
72  static constexpr int32_t ACTION_START_SPRINT = 9;
73  static constexpr int32_t ACTION_STOP_SPRINT = 10;
74  // Crouch toggle (protocol 84 — Genisys 0.15.0 PlayerActionPacket:
75  // ACTION_START_SNEAK = 11, ACTION_STOP_SNEAK = 12). This is what the
76  // 0.15.10 client sends when the player taps the sneak button or holds
77  // shift on a touch device; the server has to mirror the pose to every
78  // other client or the crouch is invisible to everyone else.
79  static constexpr int32_t ACTION_START_SNEAK = 11;
80  static constexpr int32_t ACTION_STOP_SNEAK = 12;
81 
82  int64_t eid = 0;
83  int32_t action = 0;
84  int32_t x = 0, y = 0, z = 0;
85  int32_t face = 0;
86 
87  void encode(std::vector<uint8_t>& out) const override;
88  bool decode(const uint8_t* data, size_t len) override;
89 };
