39 // Linear interpolation.
40 static inline double mcp_math_lerp(double a, double b, double t) {
41  return a + (b - a) * t;
42 }
43 // Clamp.
44 static inline double mcp_math_clamp(double v, double lo, double hi) {
45  if (v < lo) return lo;
46  if (v > hi) return hi;
47  return v;
48 }
49 // Wrap to range [0, range).
50 static inline double mcp_math_wrap(double v, double range) {
51  double r = fmod(v, range);
52  if (r < 0) r += range;
53  return r;
54 }
55 // Map noise [-1, 1] → [0, 1].
56 static inline double mcp_math_norm(double v) {
57  return 0.5 + 0.5 * v;
58 }
59 // Smoothstep: Hermite interpolation.
60 static inline double mcp_math_smoothstep(double t) {
61  return t * t * (3.0 - 2.0 * t);
62 }
63 // Fade (Perlin): 6t^5 - 15t^4 + 10t^3
64 static inline double mcp_math_fade(double t) {
65  return t * t * t * (t * (t * 6.0 - 15.0) + 10.0);
66 }
67 // degrees <-> radians
68 static inline double mcp_math_deg2rad(double d) { return d * M_PI / 180.0; }
69 static inline double mcp_math_rad2deg(double r) { return r * 180.0 / M_PI; }
70 // Integer floor division for non-negative operands.
71 static inline int64_t mcp_math_floor_div(int64_t a, int64_t b) {
72  if (b == 0) return 0;
73  int64_t q = a / b;
74  int64_t r = a % b;
75  if ((r != 0) && ((r < 0) != (b < 0))) --q;
76  return q;
77 }
78 // Integer floor modulus for non-negative modulus.
79 static inline int64_t mcp_math_floor_mod(int64_t a, int64_t b) {
80  if (b == 0) return 0;
81  int64_t r = a % b;
82  if (r < 0) r += (b > 0) ? b : -b;
83  return r;
84 }
