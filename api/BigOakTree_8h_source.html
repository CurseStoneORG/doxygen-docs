21 // BigOakTree — the vanilla FANCY oak (feature "fancy_oak"), ported 1:1 from
22 // the reference implementation (FancyTrunkPlacer + FancyFoliagePlacer) and
23 // adapted to this generator's single-chunk clipped writer.
24 //
25 // The previous shape was hand-rolled (a small blob + 2-4 straight diagonal
26 // branches) and the player found it "muy feo". The vanilla tree instead:
27 // * trunk rising to floor(height * 0.618), height = 5..16,
28 // * one branch cluster PER trunk level above 0.3 * height: each cluster
29 // picks a polar offset (radius from the treeShape profile, random
30 // angle), verifies the branch path is clear, and records a foliage
31 // attachment whose branch base slides down the trunk by BRANCH_SLOPE,
32 // * every attachment gets the vanilla blob: 5 leaf layers (offset 4 down
33 // to 0), radius 2 at the top and bottom rows and 3 in between, cut by
34 // the circular (dx+0.5)^2 + (dz+0.5)^2 <= r^2 test — no random corner
35 // culling, which is why the vanilla crown reads as round.
36 #pragma once
37 #include "GenericTree.h"
38 #include <vector>
39 
40 namespace cursestone::world::generator::object::trees {
41 
42 class BigOakTree : public GenericTree {
43  public:
44  // The fancy oak interleaves trunk, branches and canopy, so it replaces
45  // the whole generate() flow instead of just the two place* hooks.
46  bool generate(mcp_chunk_t* c, int sourceX, int sourceY, int sourceZ, uint8_t soilId,
47  MersenneRandom& rng) override;
48 
49  protected:
50  void init(MersenneRandom& rng) override;
51 
52  private:
53  // One foliage attachment: where the leaf blob sits and where its branch
54  // meets the trunk (the Y the branch grows out of).
55  struct FoliageCoord {
56  int x, y, z;
57  int branchBase;
58  };
59 
60  int trunkHeight_ = 5;
61 
62  // FancyTrunkPlacer::treeShape — crown radius profile at relY; negative
63  // means "no branch below this level".
64  static double treeShape(int height, int relY);
65  // FancyTrunkPlacer::trimBranches — branches below 0.2 * height are cut.
66  static bool trimBranches(int height, int localY) {
67  return (double)localY >= (double)height * 0.2;
68  }
69 
70  // makeLimb: walk a straight line placing logs (place=true) or checking
71  // that every block is replaceable (place=false, the vanilla probe run).
72  bool limb(mcp_chunk_t* c, int sx, int sy, int sz, int ex, int ey, int ez, bool place) const;
73  // isFree in the reference: air/replaceable, or outside this chunk (which
74  // this writer cannot see and therefore treats as clear).
75  bool freeAt(mcp_chunk_t* c, int x, int y, int z) const;
76 
77  // FancyFoliagePlacer (radius 2, offset 4, height 4): five layers, r=2 at
78  // the ends and r=3 in the middle, circular cut, into replaceable blocks.
79  void foliageBlob(mcp_chunk_t* c, int fx, int fy, int fz) const;
80 
81  std::vector<FoliageCoord> coords_;
82 };
