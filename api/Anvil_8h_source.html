24 // Anvil.h — Nukkit Anvil WorldProvider, ported to CurseStone's C++ core.
25 //
26 // This is the on-disk world format used by Nukkit InfinityEdition:
27 // worlds/<name>/level.dat (GZIP-compressed big-endian NBT, root → "Data")
28 // worlds/<name>/region/r.X.Z.mca (region files, 32x32 chunks each)
29 //
30 // The format is auto-detected by `isValid(path)` (level.dat present +
31 // region/ directory + every region file ends in `.mca`).
32 //
33 // Chunk loading flow:
34 // loadChunk(x, z) → regionFor(x, z) → RegionLoader.readChunk()
35 // → ZLIB inflate → anvil_chunk_decode() → ChunkAdapter.
36 //
37 // World creation (generate):
38 // init(path) creates region/ if missing and writes a default level.dat
39 // if one doesn't exist, mirroring Nukkit's Anvil.generate() defaults:
40 // Data.{LevelName, RandomSeed, SpawnX/Y/Z, generatorName, version=19133, ...}

130  // ------------------------------------------------------------------
131  // Async chunk writer — the PocketMine-style autosave pipeline.
132  //
133  // saveChunk() runs on the MAIN thread but only does the part that
134  // must be there: anvil_chunk_encode() reads the live mcp_chunk_t
135  // while the world is quiesced on that thread. The expensive part —
136  // ZLIB compression and the region-file I/O — is handed to a single
137  // background worker so a big autosave (hundreds of dirty chunks
138  // after players explore new land) can never stall a tick.
139  //
140  // One worker, FIFO queue ⇒ two snapshots of the same chunk are
141  // always written in enqueue order (older first), so a re-dirty
142  // during a pending write can never end up overwritten by stale data.
143  //
144  // io_mtx_ serializes the region map (regions_/region_lru_ + the
145  // FILE* handles inside AnvilRegionLoader) between the worker's
146  // writes and the main thread's loadChunk()/eviction. Compression
147  // happens OUTSIDE io_mtx_ — it never touches shared state.
148  // ------------------------------------------------------------------
149  struct SaveJob {
150  int32_t x = 0;
151  int32_t z = 0;
152  std::vector<uint8_t> nbt; // big-endian NBT of the snapshot
153  };
154 
155  void startWorker();
156  void stopWorker(); // drains the queue, joins the thread
157  void workerLoop();
158  // Push one encoded chunk into the queue. Blocks (bounded) when the
159  // queue is over capacity so a slow disk cannot balloon memory.
160  void enqueueSave(SaveJob&& job);
161 
162  std::thread worker_;
163  std::mutex job_mtx_;
164  std::condition_variable job_cv_; // wakes the worker
165  std::condition_variable done_cv_; // wakes flushPendingSaves()
166  std::deque<SaveJob> job_queue_;
167  bool job_busy_ = false; // worker writing a job
168  bool worker_stop_ = false;
169  // Backpressure cap: raw NBT is ~53 KB/chunk, so 512 in flight is a
170  // ~27 MB ceiling — never hit in practice (the worker drains far
171  // faster than the main thread encodes), it only guards a wedged disk.
172  static constexpr size_t kMaxQueuedSaves = 512;
173 
174  std::string path_;
175  int64_t seed_ = 0;
176  int64_t time_ = 0;
177  math::Vector3 spawn_{0.0, 64.0, 0.0};
178  std::string generator_name_ = "normal";
179  generator::Generator* generator_ = nullptr;
180 
181  static constexpr size_t kMaxOpenRegions = 8;
182  // io_mtx_ guards regions_ / region_lru_ and every FILE* operation in
183  // the loaders — the worker thread writes while the main thread reads.
184  mutable std::mutex io_mtx_;
185  std::unordered_map<std::string, std::unique_ptr<AnvilRegionLoader>> regions_;
186  std::vector<std::string> region_lru_;
