48 // ---- CurseStone API (SINGLE version, unified) ----
49 // There is exactly ONE version constant now: CURSESTONE_API. It is the
50 // server version AND the plugin API version at the same time — plugins
51 // declare THIS number in their plugin.yml (`api:` field). Previously there
52 // were two constants (CURSESTONE_VERSION + CURSESTONE_API_VERSION) that
53 // could drift apart; they were unified per the CurseStoneTeam request.
54 // NOTE: bumped to 1.0.2 — the plugin API gained the resources/ loader
55 // (plugin.getResource/saveResource/listResources), player movement and
56 // camera control (player.getPosition/getRotation/setRotation/setMotion)
57 // and the PlayerMove/PlayerDropItem/PlayerItemHeld events plus the held
58 // item fields on PlayerInteract. Plugins declaring 1.0.x stay loadable.
59 #define CURSESTONE_API "1.0.1"
60 #define CURSESTONE_NAME "CurseStone"
61 #define CURSESTONE_CODENAME "Obsidian"

68 // ---- tick loop tuning ----
69 // NOTE: The macro is prefixed with CURSESTONE_ to avoid colliding with the
70 // `static constexpr int TICK_RATE = 30;` member declared in
71 // block/BlockFire.h (fire spreading tick rate). Without the prefix,
72 // the preprocessor would rewrite that member into `static constexpr int 20 = 30;`
73 // which is to hard compile error on both GCC and Clang.
74 #define CURSESTONE_TICK_RATE 20 // ticks per second (server tick loop)
75 #define CURSESTONE_TICK_MS 50 // milliseconds per tick

79 // doing so would re-introduce the macro collision with BlockFire::TICK_RATE.
