139 // Defines the vanilla dropped-item pickup volume for MCPE 0.15.10.
140 // Items are collected when their bounding boxes overlap the player's box grown by one block.
141 // The shared bounds keep attraction and pickup checks consistent.
142 constexpr double kItemHalfW = 0.125; // Half of the item width.
143 constexpr double kItemHeight = 0.25; // Item height.
144 constexpr double kPickupHalfExtent = 1.3; // Player half-width plus pickup range.
145 constexpr double kPickupBelowFeet = 1.0; // Pickup range below the feet.
146 constexpr double kPickupAboveFeet = 2.8; // Player height plus pickup range.
147 
148 // Checks strict AABB overlap between an item and the player's pickup volume.
149 inline bool itemWithinPickupVolume(double ix, double iy, double iz, double px, double py,
150  double pz) {
151  if (!(ix + kItemHalfW > px - kPickupHalfExtent && ix - kItemHalfW < px + kPickupHalfExtent))
152  return false;
153  if (!(iy + kItemHeight > py - kPickupBelowFeet && iy < py + kPickupAboveFeet))
154  return false;
155  if (!(iz + kItemHalfW > pz - kPickupHalfExtent && iz - kItemHalfW < pz + kPickupHalfExtent))
156  return false;
157  return true;
158 }
159 
160 inline bool itemWithinPickupVolume(const DroppedItem& di, double px, double py, double pz) {
161  return itemWithinPickupVolume(di.x, di.y, di.z, px, py, pz);
162 }

543  static uint64_t packChunkPos(int32_t cx, int32_t cz) {
544  uint64_t ux = static_cast<uint64_t>(static_cast<uint32_t>(cx));
545  uint64_t uz = static_cast<uint64_t>(static_cast<uint32_t>(cz));
546  return (ux << 32) | uz;
547  }
548 
549  // Packs 28-bit X/Z coordinates and an 8-bit Y coordinate into one key.
550  static uint64_t packTilePos(int32_t x, int32_t y, int32_t z) {
551  uint64_t ux = static_cast<uint64_t>(static_cast<uint32_t>(x)) & 0xFFFFFFFULL;
552  uint64_t uz = static_cast<uint64_t>(static_cast<uint32_t>(z)) & 0xFFFFFFFULL;
553  uint64_t uy = static_cast<uint64_t>(static_cast<uint8_t>(y));
554  return (ux << 36) | (uz << 8) | uy;
555  }
