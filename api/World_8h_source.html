144 // Server-side record of a mob. The struct carries only identity,
145 // position and the chunk index: all per-mob dynamic state (AI goals,
146 // intent, physics) lives in entity/ai/'s MobBrain, keyed by eid. The
147 // chunk index (cx/cz) mirrors mobs_by_chunk_ so chunk streams touch
148 // O(mobs of the chunk) instead of scanning the whole world map.
149 //
150 // Combat state lives here (not in the brain): damage can arrive from any
151 // thread through the plugin APIs, while brains are main-thread-only.
152 struct MobEntity {
153  int64_t eid = 0;
154  int32_t type = 0; // Protocol-84 network id (entity::EntityTypes).
155  double x = 0.0, y = 0.0, z = 0.0; // Feet position.
156  float yaw = 0.0f, pitch = 0.0f;
157  int32_t cx = 0, cz = 0; // Chunk the entity is indexed under.
158  // Vanilla-shaped health, seeded from entity::mobMaxHealth(type) on spawn.
159  float health = 20.0f;
160  float max_health = 20.0f;
161  // Pending knockback (blocks/tick) written by the damage pipeline and
162  // consumed by the next AI actuation pass; kb_tick is the server tick
163  // the shove was applied, so a stale field expires instead of firing
164  // after a long gate closure.
165  float kbx = 0.0f, kbz = 0.0f;
166  uint64_t kb_tick = 0;
167  // Death window: non-zero once the lethal blow landed, holding the
168  // server tick the record must be removed at (death animation plays
169  // for ~one second first). Zero means alive.
170  uint64_t death_tick = 0;
171 };

173 // Defines the vanilla dropped-item pickup volume for MCPE 0.15.10.
174 // Items are collected when their bounding boxes overlap the player's box grown by one block.
175 // The shared bounds keep attraction and pickup checks consistent.
176 constexpr double kItemHalfW = 0.125; // Half of the item width.
177 constexpr double kItemHeight = 0.25; // Item height.
178 constexpr double kPickupHalfExtent = 1.3; // Player half-width plus pickup range.
179 constexpr double kPickupBelowFeet = 1.0; // Pickup range below the feet.
180 constexpr double kPickupAboveFeet = 2.8; // Player height plus pickup range.
181 
182 // Checks strict AABB overlap between an item and the player's pickup volume.
183 inline bool itemWithinPickupVolume(double ix, double iy, double iz, double px, double py,
184  double pz) {
185  if (!(ix + kItemHalfW > px - kPickupHalfExtent && ix - kItemHalfW < px + kPickupHalfExtent))
186  return false;
187  if (!(iy + kItemHeight > py - kPickupBelowFeet && iy < py + kPickupAboveFeet))
188  return false;
189  if (!(iz + kItemHalfW > pz - kPickupHalfExtent && iz - kItemHalfW < pz + kPickupHalfExtent))
190  return false;
191  return true;
192 }
193 
194 inline bool itemWithinPickupVolume(const DroppedItem& di, double px, double py, double pz) {
195  return itemWithinPickupVolume(di.x, di.y, di.z, px, py, pz);
196 }

562  // ----- Floating text (particle/FloatingTextParticle) -----
563  // Protocol 84 has no text-rendering particle and its client draws
564  // nametags only on player entities, so floating text is the
565  // FloatingTextParticle packet composite: ONE invisible client-side
566  // fake player whose nametag is the whole text. The carrier spawns
567  // through AddPlayerPacket + SetEntityDataPacket (invisible flag,
568  // nametag, NoAI) and leaves through one RemoveEntityPacket; the
569  // Genisys-style title line and every '\n' body line ride on that
570  // single nametag and the client stacks the lines itself, so a text
571  // costs one entity and one fixed spawn pair no matter how many lines
572  // it carries.
573  //
574  // Five tag shapes share this one carrier:
575  // static fixed anchor (the default);
576  // follow anchors over one entity/player, repositioned by the
577  // budgeted follow pass past a movement threshold;
