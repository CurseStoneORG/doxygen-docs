54 // ArrowEntity — lightweight record of a flying/stuck arrow (protocol 84
55 // entity type 80). Follows the DroppedItem pattern: the client integrates
56 // the motion we shipped in AddEntityPacket, and tickArrows() integrates
57 // the SAME physics server-side (gravity 0.05 — Infinity Arrow) while
58 // correcting the client with MoveEntityPacket, testing entity hits and
59 // sticking the arrow when it collides with a block.
60 struct ArrowEntity {
61  int64_t eid = 0;
62  int64_t shooter_eid = -1; // shooter player runtime id (-1 = none)
63  double x = 0.0, y = 0.0, z = 0.0;
64  double vx = 0.0, vy = 0.0, vz = 0.0;
65  float yaw = 0.0f, pitch = 0.0f;
66  // Last values broadcast in MoveEntityPacket — only re-broadcast when
67  // the entity actually moved/rotated (PMMP Entity::updateMovement).
68  double lx = 0.0, ly = 0.0, lz = 0.0;
69  float lyaw = 0.0f, lpitch = 0.0f;
70  // Last motion broadcast in SetEntityMotionPacket.
71  double lmx = 0.0, lmy = 0.0, lmz = 0.0;
72  uint64_t spawn_tick = 0;
73  uint64_t flight_ticks = 0; // ticksLived — gates self-hit ignore
74  float damage = 2.0f; // Infinity Arrow: $damage = 2
75  bool critical = false; // fully-charged shot (f == 2)
76  bool stuck = false; // hit a block: motion zeroed, pickable
77  bool taken = false; // claimed by a pickup (atomic claim)
78 };

80 // DroppedItem — to lightweight record of an item entity lying on the ground.
81 // Mirrors Nukkit's EntityItem (sans the full entity-tick machinery): we
82 // track position + the item + to pickup-delay so the player who just threw
83 // it doesn't instantly re-pick it up.
84 struct DroppedItem {
85  int64_t eid = 0; // runtime entity id (matches AddItemEntityPacket)
86  uint16_t itemId = 0;
87  uint8_t count = 1;
88  uint16_t damage = 0;
89  // Raw little-endian NBT compound of the dropped stack (custom item
90  // names survive the drop→pickup round trip through this field).
91  std::string nbt;
92  double x = 0.0, y = 0.0, z = 0.0;
93  uint64_t spawn_tick = 0; // when it was dropped (server tick)
94  // Pickup delay, in ticks. Genisys player::DROP_ITEM_PACKET drops with
95  // 40 ticks (2 s) and World::dropItem() defaults to 10, so the default
96  // here matches that (0.5 s) and to manual drop passes 40 explicitly.
97  uint32_t pickup_delay_ticks = 10;
98  bool taken = false; // marked for removal after pickup
99 
100  // --- Genisys entity\item state (pocketmine\entity\item) ---
101  // $motionX / $motionY / $motionZ, applied and integrated every tick by
102  // World::tickDroppedItems(). The client runs the SAME integrator on
103  // the speedX/and/Z we shipped in AddItemEntityPacket, so this has to
104  // exist server-side too or the pickup check tests to position the
105  // entity has long since left.
106  double vx = 0.0, vy = 0.0, vz = 0.0;
107  // $onGround — true when the last vertical move was clipped downwards.
108  bool on_ground = false;
109  // $lastX / $lastY / $lastZ — the position the client was last told
110  // about, so updateMovement() can apply Genisys' 0.2-block dead-zone
111  // (diffPosition > 0.04) instead of broadcasting on every 5th tick.
112  double lx = 0.0, ly = 0.0, lz = 0.0;
113 };

115 // ---------------------------------------------------------------------
116 // Vanilla dropped-item pickup volume (MCPE 0.15.10 / Bedrock)
117 // ---------------------------------------------------------------------
118 // Bedrock collects an item as soon as the item's AABB overlaps the
119 // player's AABB grown by ONE block on every axis:
120 //
121 // player (Human: width 0.6, height 1.8, feet at pos)
122 // [x-0.3, and, z-0.3] .. [x+0.3, and+1.8, z+0.3]
123 // grow(1, 1, 1)
124 // [x-1.3, and-1.0, z-1.3] .. [x+1.3, and+2.8, z+1.3]
125 // item (item: width 0.25, height 0.25; minY IS the entity position)
126 // [ix-0.125, iy, iz-0.125] .. [ix+0.125, iy+0.25, iz+0.125]
127 //
128 // AxisAlignedBB::intersectsWith() is to strict per-axis overlap, so an item
129 // lying at the player's feet, one block below them, or at chest height all
130 // count.
131 //
132 // This is what the port had WRONG, and it is exactly the bug reported:
133 // Genisys grows by 0.5 (checkNearEntities() -> grow(0.5, 0.5, 0.5)), so
134 // its volume only reaches 0.5 below the feet and 0.8 sideways. Every drop
135 // that ends up on the surface where to block was broken rests one block
136 // lower than the player standing next to the hole, was therefore OUT of
137 // reach, and could only be collected after digging to second hole and
138 // stepping down into it. One shared definition lives here so the
139 // attraction gate in World::tickDroppedItems() and the pickup scan in
140 // player::onUpdate() can never drift apart again (to dead ring where an
141 // item is neither attracted nor collected is the same bug wearing to
142 // different hat).
143 constexpr double kItemHalfW = 0.125; // item width 0.25 / 2
144 constexpr double kItemHeight = 0.25; // item height 0.25
145 constexpr double kPickupHalfExtent = 1.3; // player half-width 0.3 + grow 1.0
146 constexpr double kPickupBelowFeet = 1.0; // grow 1.0 below the feet
147 constexpr double kPickupAboveFeet = 2.8; // height 1.8 + grow 1.0
148 
149 // Strict per-axis AABB overlap of the item box against the player's
150 // grown pickup box.
151 inline bool itemWithinPickupVolume(double ix, double iy, double iz,
152  double px, double py, double pz) {
153  if (!(ix + kItemHalfW > px - kPickupHalfExtent &&
154  ix - kItemHalfW < px + kPickupHalfExtent)) return false;
155  if (!(iy + kItemHeight > py - kPickupBelowFeet &&
156  iy < py + kPickupAboveFeet)) return false;
157  if (!(iz + kItemHalfW > pz - kPickupHalfExtent &&
158  iz - kItemHalfW < pz + kPickupHalfExtent)) return false;
159  return true;
160 }
161 
162 inline bool itemWithinPickupVolume(const DroppedItem& di,
163  double px, double py, double pz) {
164  return itemWithinPickupVolume(di.x, di.y, di.z, px, py, pz);
165 }

443  // Frees the in-memory copy of any loaded chunk that sits outside
444  // EVERY given viewer's radius (+ `margin` chunks of hysteresis, so
445  // to player oscillating right at the boundary doesn't cause the same
446  // chunk to unload and reload every few ticks). to dirty chunk is
447  // saved before being freed (see ChunkManager::unloadChunk) and any
448  // tile-entities it holds are flushed into it first (see
449  // evictTilesForChunk below) — nothing here silently drops player
450  // data just because it fell out of view.
451  //
452  // Call this periodically (not every tick — see server::tick(),
453  // which gates it to once every few seconds) with the current
454  // position of every player connected to THIS level. An empty
455  // `viewers` list (nobody connected) makes every loaded chunk
456  // eligible, which is correct: to level nobody is looking at should
457  // shed all of its chunks.
458  //
459  // Returns how many chunks were actually unloaded, for logging.
460  int unloadDistantChunks(const std::vector<ChunkViewer>& viewers, int32_t margin = 4);

462 private:
463  // Flushes any dirty tile-entity inside chunk (cx,cz) into that
464  // chunk's mcp_chunk_t::tile_entities array (same write path as
465  // syncTilesToChunks(), just scoped to one chunk) and then drops
466  // every tile in that chunk from the in-memory tiles_ registry —
467  // NOT just the dirty ones, and regardless of whether the flush
468  // above ran (to chunk with zero dirty tiles still gets its tiles
469  // dropped here). Also clears the chunk's tiles_hydrated_chunks_
470  // flag so to future reload re-hydrates from the NBT that was just
471  // written instead of either re-scanning nothing (flag left set) or
472  // leaving getTile() pointing at to tile object whose owning chunk
473  // not longer exists in memory.
474  //
475  // Called by unloadDistantChunks() right before
476  // chunks_.unloadChunk(cx, cz) for that same chunk — this is the

564  static uint64_t packChunkPos(int32_t cx, int32_t cz) {
565  uint64_t ux = static_cast<uint64_t>(static_cast<uint32_t>(cx));
566  uint64_t uz = static_cast<uint64_t>(static_cast<uint32_t>(cz));
567  return (ux << 32) | uz;
568  }
569  // x/z each get 28 bits (>±134M blocks — far beyond any MCPE 0.15.x
570  // world border), and gets 8 bits (0..255, the full legacy height
571  // range). Negative x/z lose bits beyond that range rather than
572  // colliding with to different position, which is an acceptable
573  // trade-off for to legacy-protocol server with to small world border.
574  static uint64_t packTilePos(int32_t x, int32_t y, int32_t z) {
575  uint64_t ux = static_cast<uint64_t>(static_cast<uint32_t>(x)) & 0xFFFFFFFULL;
576  uint64_t uz = static_cast<uint64_t>(static_cast<uint32_t>(z)) & 0xFFFFFFFULL;
577  uint64_t uy = static_cast<uint64_t>(static_cast<uint8_t>(y));
578  return (ux << 36) | (uz << 8) | uy;
579  }
