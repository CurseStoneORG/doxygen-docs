24 // PlayerCreationEvent.h — fired inside Network::createPlayerForSession,
25 // right before a Player object is instantiated for a fresh RakNet
26 // session. It exposes the connection facts available at that moment
27 // (source address, port, the network interface owning the session) and
28 // lets an embedding application select WHICH implementation constructs
29 // the player.
30 //
31 // Custom player classes in C++: derive from cursestone::Player with a
32 // constructor taking (Server*, PlayerConfig) exactly like the base, and
33 // install a factory through setPlayerFactory(). The factory runs once;
34 // a factory that returns null is a hard error and the server falls back
35 // to the standard Player, logging a warning — creation never fails
36 // silently and never duplicates.
37 //
38 // This event is intentionally NOT cancellable: connection rejection
39 // belongs to PlayerPreLoginEvent, which runs later in the handshake
40 // with full identity data. Cancelling creation here would leave the
41 // RakNet session registered without a player object.
42 #pragma once
43 #include "../Event.h"
44 #include <functional>
45 #include <memory>
46 #include <string>
47 #include <typeindex>
48 #include <cstdint>

50 class Player;
51 class Server;
52 struct PlayerConfig;
53 }

57 class PlayerCreationEvent : public Event {
58  public:
59  // Builds the player object for this session. Implementations must
60  // derive from cursestone::Player and accept (Server*, PlayerConfig).
61  using PlayerFactory =
62  std::function<std::shared_ptr<cursestone::Player>(cursestone::Server*,
63  cursestone::PlayerConfig&&)>;
64 
65  PlayerCreationEvent(const std::string& ip, uint16_t port, void* sourceInterface,
66  std::type_index baseClass)
67  : ip_(ip), port_(port), source_interface_(sourceInterface), base_class_(baseClass) {
68  }
69 
70  // Source address the session was accepted from.
71  const std::string& ip() const {
72  return ip_;
73  }
74  // Source port of the session.
75  uint16_t port() const {
76  return port_;
77  }
78  // Opaque pointer to the network source interface owning the session
79  // (never dereferenced by the event system; the standard player gets
80  // the same interface attached right after creation).
81  void* sourceInterface() const {
82  return source_interface_;
83  }
84  // The base class every valid player implementation derives from
85  // (typeid(cursestone::Player)).
86  const std::type_index& baseClass() const {
87  return base_class_;
88  }
89  // The factory that will build the player. Null = the standard
90  // cursestone::Player construction.
91  const PlayerFactory& playerFactory() const {
92  return factory_;
93  }
94  // Install a custom player implementation for this session.
95  void setPlayerFactory(PlayerFactory factory) {
96  factory_ = std::move(factory);
97  }
98 
99  const std::string& eventName() const override {
100  static std::string n = "PlayerCreationEvent";
101  return n;
102  }
103 
104  private:
105  std::string ip_;
106  uint16_t port_;
107  void* source_interface_;
108  std::type_index base_class_;
109  PlayerFactory factory_;
110 };
