52 // Connection timeout for the session sweep, in milliseconds (30s — the
53 // RakLib/the reference server default). The timeout clock means "no datagram received at
54 // all": keepalive PINGs and ACK/NACK traffic both count as activity, so
55 // a player who stays idle (AFK) with the game open is never kicked; only
56 // a genuinely dead connection (app closed, network lost) is reaped.

59 // FIX (audit priority 4/12): RakNet's 24-bit index spaces (datagram
60 // sequence number, reliable message index, ordering index) all wrap at
61 // this boundary. Shared here (not just in SessionManager.c, where the
62 // wrap-safe comparison helpers live) because Session.c's send path also
63 // needs it: send_seq is kept as a plain uint32_t counter that grows
64 // without ever wrapping in memory (simpler than wrapping the counter
65 // itself), while only the 3 low bytes are ever written to the wire —
66 // masking with this here keeps what's stored in the recovery queue
67 // (recovery[slot].seq) consistent with the 24-bit values that
68 // subsequently arrive back in ACK/NACK from the client, which are
69 // decoded straight off the wire and can never be anything but 24-bit.
70 // Without this mask, a session sending enough traffic to grow send_seq
71 // past 0xFFFFFF would find that recovery[slot].seq (still the full,
72 // un-masked 32-bit counter value) could never again match an incoming
73 // ACK's seq value (always 24-bit) for any datagram sent after that
74 // point — retransmission bookkeeping would silently stop being able to
75 // recognize its own acknowledgments.
76 #define MCP_SEQ24_MASK 0x00FFFFFFu

138 // FIX: raised from 512 → 4096. With chunk bursts of 60+ reliable
139 // fragments per tick (documented elsewhere in this codebase), 512
140 // in-flight slots could be exhausted before the corresponding ACKs
141 // came back, silently overwriting (and thus permanently losing) a
142 // reliable datagram that was never acknowledged. If that lost
143 // datagram carried a RELIABLE_ORDERED fragment (e.g. part of the
144 // login/spawn sequence on channel 0), the client's own ordering
145 // buffer would stall waiting for an index that would never arrive.
146 // 4096 slots (~32KB of pointers/metadata per session, negligible)
147 // gives enough headroom for several ticks worth of chunk bursts
148 // even under real-world latency before wraparound can occur.
149 #define MCP_RECOVERY_CAP 4096

151 // ---------------------------------------------------------------------------
152 // Flow control + liveness instrumentation (multiplayer hardening).
153 //
154 // The reliability layer had no ceiling on what it would keep in flight for
155 // ONE session: a client that stops acknowledging (crashed client, saturated
156 // phone, NAT/proxy that stopped forwarding) made the server keep queueing
157 // chunk fragments and then resend every single one of them on a fixed 5s
158 // cadence until the 30s timeout finally reaped the session. In the field
159 // dump that produced was: recovery_occupied=468, retransmits=5136 for a
160 // single stalled client — and because the RakLib thread is ONE thread for
161 // every player, that traffic also delays the ACK/PING processing of the
162 // OTHER sessions, which is how "one player joins and then everybody gets
163 // dropped" happens.
164 //
165 // The three knobs below bound that:
166 // * MCP_MAX_INFLIGHT_RELIABLE — per-session ceiling on un-ACKed reliable
167 // datagrams. Past it the session is considered stalled: new reliable
168 // data is refused (and counted) instead of queued forever. Chunk
169 // streaming is unaffected for a healthy client (its window stays in
170 // the tens), so this only ever fires on a session that is already lost.
171 // * MCP_RETRANSMIT_MAX_PER_TICK — GLOBAL per-tick retransmit budget
172 // shared by ALL sessions (was: 32 per session per tick, i.e. 6400
173 // datagrams/s with 20 players). One stalled client can no longer
174 // monopolise the network thread.
175 // * MCP_RETRANSMIT_BASE_MS/_MAX_MS — exponential backoff per recovery
176 // entry: a datagram that has already been resent is retried less and
177 // less often instead of every 5s forever.
178 // ---------------------------------------------------------------------------
179 #define MCP_MAX_INFLIGHT_RELIABLE 512
180 #define MCP_RETRANSMIT_BASE_MS 5000
181 #define MCP_RETRANSMIT_MAX_MS 40000
182 #define MCP_RETRANSMIT_MAX_PER_SESSION 8
183 #define MCP_RETRANSMIT_MAX_PER_TICK 24
184 // Server-side keepalive: if the server has had nothing to send to a
185 // session for this long it pings it and measures the round trip. That
186 // RTT is what tells "the client is gone" apart from "the client is alive
187 // but its uplink is drowning in our chunk data".
188 #define MCP_KEEPALIVE_PING_MS 5000
189 #define MCP_KEEPALIVE_PING_ID_BASE 0x4353000000000000LL

294  // Payload-aware duplicate detection for reliable messages.
295  //
296  // reliable_recv_window[] above can only answer "have I seen this
297  // message index?". That is enough to suppress a genuine RETRANSMIT
298  // (which reuses both the index and the bytes), but it cannot tell a
299  // retransmit apart from a client that REUSES an index for a different
300  // message. Some clients in the wild (e.g. the LegacyBot stress
301  // tester) never increment reliableMessageNumber and send 0 for every
302  // reliable-ordered packet, so the window accepted the first packet
303  // and silently discarded the rest as "duplicates" — the
304  // CLIENT_HANDSHAKE and the LoginPacket never reached the game layer,
305  // no session was ever opened, and the log stayed completely empty.
306  // This table remembers the (index, payload-hash) of recently accepted
307  // messages so a repeat is only dropped when the bytes match too.
308  //
309  // It is DIRECT-MAPPED by `message_index % MCP_RELIABLE_WINDOW_SIZE`
310  // and MUST cover exactly the same index space as reliable_recv_window
311  // above. It used to be a 256-slot ring while the window was 2048:
312  // a retransmit older than 256 messages but still inside the window
313  // found its window bit set yet its payload long since evicted, so
314  // reliable_already_delivered() returned false and the message was
315  // DELIVERED TO THE GAME LAYER A SECOND TIME. That needs real packet
316  // loss to trigger — i.e. it is invisible on a lossless LAN/loopback
317  // and only shows up on a hosted server over the Internet, exactly
318  // where duplicated actions/state hurt most. With one entry per window
319  // slot the lookup is also O(1) instead of a 256-entry scan per
320  // message.
321  struct {
322  uint32_t index;
323  uint32_t hash;
324  uint32_t len;
325  bool used;
326  } reliable_seen[MCP_RELIABLE_WINDOW_SIZE];

393  // ---- Congestion backlog (chunk-hole fix, "some chunks never render
394  // while exploring"): past the inflight ceiling the old code REFUSED
395  // the new reliable payload outright — but the game layer had already
396  // marked the chunk as sent, so the client simply never received it
397  // and the terrain stayed invisible until the player wandered far
398  // enough for the chunk to leave and re-enter the view square (or
399  // relogged). A slow-ACKing client (real phone over Wi-Fi/mobile
400  // data) sits at that ceiling during normal exploring, so chunks
401  // were being thrown away in normal play, not just for stalled
402  // sessions.

404  // Now a payload refused by the inflight ceiling is QUEUED here
