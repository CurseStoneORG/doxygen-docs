71  static BoundingBox orientBox(int footX, int footY, int footZ, int offX, int offY, int offZ, int width, int height, int depth, int orientation) {
72  switch (orientation) {
73  default:
74  return BoundingBox(footX + offX, footY + offY, footZ + offZ, footX + width - 1 + offX, footY + height - 1 + offY, footZ + depth - 1 + offZ);
75  case Direction::NORTH:
76  // foot is at x0, y0, z1
77  return BoundingBox(footX + offX, footY + offY, footZ - depth + 1 + offZ, footX + width - 1 + offX, footY + height - 1 + offY, footZ + offZ);
78  case Direction::SOUTH:
79  // foot is at x0, y0, z0
80  return BoundingBox(footX + offX, footY + offY, footZ + offZ, footX + width - 1 + offX, footY + height - 1 + offY, footZ + depth - 1 + offZ);
81  case Direction::WEST:
82  // foot is at x1, y0, z0, but width and depth are flipped
83  return BoundingBox(footX - depth + 1 + offZ, footY + offY, footZ + offX, footX + offZ, footY + height - 1 + offY, footZ + width - 1 + offX);
84  case Direction::EAST:
85  // foot is at x0, y0, z0, but width and depth are flipped
86  return BoundingBox(footX + offZ, footY + offY, footZ + offX, footX + depth - 1 + offZ, footY + height - 1 + offY, footZ + width - 1 + offX);
87  }
88  }
