83  float nextGaussian() {
84  if (haveNextNextGaussian) {
85  haveNextNextGaussian = false;
86  return nextNextGaussian;
87  } else {
88  float v1, v2, s;
89  do {
90  v1 = 2 * nextFloat() - 1; // between -1.0 and 1.0
91  v2 = 2 * nextFloat() - 1; // between -1.0 and 1.0
92  s = v1 * v1 + v2 * v2;
93  } while (s >= 1 || s == 0);
94  float multiplier = std::sqrt(-2 * std::log(s) / s);
95  nextNextGaussian = v2 * multiplier;
96  haveNextNextGaussian = true;
97  return v1 * multiplier;
98  }
99  }

117  /* initializes _mt[N] with a seed */
118  void init_genrand(unsigned long s) {
119  _mt[0] = s & 0xffffffffUL;
120  for (_mti = 1; _mti < N; _mti++) {
121  _mt[_mti] = (1812433253UL * (_mt[_mti - 1] ^ (_mt[_mti - 1] >> 30)) + _mti);
122  /* See Knuth TAOCP Vol2. 3rd Ed. P.106 for multiplier. */
123  /* In the previous versions, MSBs of the seed affect */
124  /* only MSBs of the array _mt[]. */
125  /* 2002/01/09 modified by Makoto Matsumoto */
126  _mt[_mti] &= 0xffffffffUL;
127  /* for >32 bit machines */
128  }
129  }

131  /* initialize by an array with array-length */
132  /* init_key is the array for initializing keys */
133  /* key_length is its length */
134  /* slight change for C++, 2004/2/26 */
135  void init_by_array(unsigned long init_key[], int key_length) {
136  int i, j, k;
137  init_genrand(19650218UL);
138  i = 1;
139  j = 0;
140  k = (N > key_length ? N : key_length);
141  for (; k; k--) {
142  _mt[i] = (_mt[i] ^ ((_mt[i - 1] ^ (_mt[i - 1] >> 30)) * 1664525UL)) + init_key[j] +
143  j; /* non linear */
144  _mt[i] &= 0xffffffffUL; /* for WORDSIZE > 32 machines */
145  i++;
146  j++;
147  if (i >= N) {
148  _mt[0] = _mt[N - 1];
149  i = 1;
150  }
151  if (j >= key_length)
152  j = 0;
153  }
154  for (k = N - 1; k; k--) {
155  _mt[i] =
156  (_mt[i] ^ ((_mt[i - 1] ^ (_mt[i - 1] >> 30)) * 1566083941UL)) - i; /* non linear */
157  _mt[i] &= 0xffffffffUL; /* for WORDSIZE > 32 machines */
158  i++;
159  if (i >= N) {
160  _mt[0] = _mt[N - 1];
161  i = 1;
162  }
163  }
164 
165  _mt[0] = 0x80000000UL; /* MSB is 1; assuring non-zero initial array */
166  }

168  /* generates a random number on [0,0xffffffff]-interval */
169  unsigned long genrand_int32(void) {
170  unsigned long y;
171  static unsigned long mag01[2] = {0x0UL, MATRIX_A};
172  /* mag01[x] = x * MATRIX_A for x=0,1 */
173 
174  if (_mti >= N) { /* generate N words at one time */
175  int kk;
176 
177  if (_mti == N + 1) /* if init_genrand() has not been called, */
178  init_genrand(5489UL); /* a default initial seed is used */
179 
180  for (kk = 0; kk < N - M; kk++) {
181  y = (_mt[kk] & UPPER_MASK) | (_mt[kk + 1] & LOWER_MASK);
182  _mt[kk] = _mt[kk + M] ^ (y >> 1) ^ mag01[y & 0x1UL];
183  }
184  for (; kk < N - 1; kk++) {
185  y = (_mt[kk] & UPPER_MASK) | (_mt[kk + 1] & LOWER_MASK);
186  _mt[kk] = _mt[kk + (M - N)] ^ (y >> 1) ^ mag01[y & 0x1UL];
187  }
188  y = (_mt[N - 1] & UPPER_MASK) | (_mt[0] & LOWER_MASK);
189  _mt[N - 1] = _mt[M - 1] ^ (y >> 1) ^ mag01[y & 0x1UL];
190 
191  _mti = 0;
192  }
193 
194  y = _mt[_mti++];
195 
196  /* Tempering */
197  y ^= (y >> 11);
198  y ^= (y << 7) & 0x9d2c5680UL;
199  y ^= (y << 15) & 0xefc60000UL;
200  y ^= (y >> 18);
201 
202  return y;
203  }

228  double genrand_res53(void) {
229  unsigned long a = genrand_int32() >> 5, b = genrand_int32() >> 6;

237  void rrDiff(float& x) {
238  unsigned long u = genrand_int32();
239  const float xx0 = (u & 0xffff) / 65536.0f; // 2 x 16 bits
240  const float xx1 = ((u >> 16) & 0xffff) / 65536.0f;
241  x = xx0 - xx1;
242  }
243  void rrDiff(float& x, float& y) {
244  unsigned long u = genrand_int32();
245  const float xx0 = ((u) & 0xff) / 256.0f; // 4 x 8 bits
246  const float xx1 = ((u >> 8) & 0xff) / 256.0f;
247  const float yy0 = ((u >> 16) & 0xff) / 256.0f;
248  const float yy1 = ((u >> 24) & 0xff) / 256.0f;
249  x = xx0 - xx1;
250  y = yy0 - yy1;
251  }
252  void rrDiff(float& x, float& y, float& z) {
253  unsigned long u = genrand_int32();
254  const float xx0 = ((u) & 0x1f) / 32.0f; // 6 x 5 bits
255  const float xx1 = ((u >> 5) & 0x1f) / 32.0f;
256  const float yy0 = ((u >> 10) & 0x1f) / 32.0f;
257  const float yy1 = ((u >> 15) & 0x1f) / 32.0f;
258  const float zz0 = ((u >> 20) & 0x1f) / 32.0f;
259  const float zz1 = ((u >> 25) & 0x1f) / 32.0f;
260  x = xx0 - xx1;
261  y = yy0 - yy1;
262  z = zz0 - zz1;
263  }
