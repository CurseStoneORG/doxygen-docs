23 // RedstoneManager.h — motor de propagación de señal de redstone.
24 //
25 // DISEÑO (aprendiendo del incidente de BlockFire v6: un freeze de 88s
26 // causado por propagación en cascada sin límite duro):
27 // - Toda propagación de wire está acotada a un máximo de 15 pasos de
28 // potencia (regla real de Minecraft: la señal decae 1 por bloque y
29 // nunca viaja más de 15 bloques desde la fuente). Esto es un límite
30 // matemático duro, no una heurística — nunca puede iterar sin fin.
31 // - No hay tareas programadas (scheduler) recurrentes que se
32 // re-encolen solas. Cada actualización de redstone es un cálculo
33 // síncrono y acotado que termina en el mismo tick en que se pide.
34 // - notifyNeighbors() es la única función que el resto del código
35 // debe llamar (desde Player.cpp, tras romper/colocar un bloque).
36 // Nunca se llama a sí misma de forma indirecta más allá de la
37 // propagación acotada del wire.
38 //
39 // FIX (reportado por el usuario tras probar la primera versión): el
40 // motor cambiaba el `data`/id del bloque en el World correctamente,
41 // pero nunca avisaba a los jugadores conectados — por eso el circuito
42 // solo se veía "prendido" al reconectar (momento en que el cliente
43 // vuelve a pedir los chunks desde cero), y nada que dependiera de la
44 // señal (lámparas) reaccionaba en vivo. Las funciones que cambian
45 // bloques ahora reciben también un Server*, y hacen el mismo broadcast
46 // de UpdateBlockPacket (+ sonido/partícula cuando aplica) que ya usa
47 // Player.cpp para romper/colocar — mismos jugadores, mismo patrón,
48 // para que la propagación se vea instantánea como en MCPE real.

57 // Se llama tras romper o colocar CUALQUIER bloque en (x,y,z). Revisa los
58 // 6 vecinos inmediatos y, si alguno es un componente de redstone
59 // (wire, antorcha, palanca, botón, placa de presión, bloque de
60 // redstone), dispara su actualización correspondiente — y transmite
61 // cada cambio visible a los jugadores que estén viendo ese chunk.
62 //
63 // Esta es la función principal que el resto del proyecto necesita
64 // llamar — todo lo demás en este módulo es implementación interna.
65 void notifyNeighbors(cursestone::world::World* world, cursestone::Server* server,
66  int32_t x, int32_t y, int32_t z);

68 // Recalcula la potencia (0-15) que debería tener un bloque de redstone
69 // wire en (x,y,z) dado su entorno actual, y actualiza su `data` en el
70 // mundo si cambió — propagando en cascada ACOTADA (máx. 15 pasos) a los
71 // wires vecinos si el cambio de potencia los afecta a ellos también.
72 // Transmite cada wire que cambie a los jugadores cercanos.
73 void updateWireAt(cursestone::world::World* world, cursestone::Server* server,

76 // Recalcula si una lámpara de redstone en (x,y,z) debe estar encendida
77 // (id LIT_REDSTONE_LAMP) o apagada (id REDSTONE_LAMP) según si recibe
78 // señal actualmente, y transmite el cambio si corresponde. A diferencia
79 // del wire, la lámpara solo consume señal, nunca la propaga — no hay
80 // riesgo de cascada al llamar esto.
81 void updateLampAt(cursestone::world::World* world, cursestone::Server* server,

84 // Devuelve la potencia de señal (0-15) que el bloque en (x,y,z) aporta
85 // hacia afuera — considera antorchas encendidas, bloques de redstone,
86 // wire ya calculado, y palancas/botones activados. Devuelve 0 si el
87 // bloque no es una fuente de señal o está apagado. No cambia nada en
88 // el mundo, así que no necesita Server*.
89 int getEmittedPower(cursestone::world::World* world, int32_t x, int32_t y, int32_t z);

91 // True si el bloque en (x,y,z) está actualmente recibiendo señal de

93 // mismo (para wire). Es lo que consultan puertas/pistones/lámparas para

95 bool isPowered(cursestone::world::World* world, int32_t x, int32_t y, int32_t z);

97 // Intenta interactuar con un componente de redstone accionable por el
98 // jugador (palanca, botón, puerta de madera — la de hierro NO
99 // reacciona a clic, sólo a redstone, igual que en vanilla) en
100 // (x,y,z). Si el bloque en esa posición es uno de esos tipos,
101 // alterna/activa su estado, actualiza el mundo, transmite el cambio
102 // y devuelve true. Palanca/botón además notifican a los vecinos
103 // (pueden alimentar un circuito); la puerta NO lo hace al recibir
104 // clic manual — ver nota en el .cpp sobre por qué. Si no es un
105 // componente accionable, no hace nada y devuelve false — así el
106 // llamador (Player) sabe si debe seguir con su propia lógica de
107 // colocación de item.
108 bool tryInteract(cursestone::world::World* world, cursestone::Server* server,
109  int32_t x, int32_t y, int32_t z);
110 
111 // Recalcula si una antorcha de redstone en (x,y,z) debe estar
112 // encendida (id REDSTONE_TORCH_ON) o apagada (id REDSTONE_TORCH_OFF).
113 // Comportamiento vanilla real (la antorcha es un INVERSOR lógico, no un
114 // simple foco): se apaga si el bloque que la sostiene (el vecino hacia
115 // donde apunta, normalmente abajo) está recibiendo señal, y se enciende
116 // si ese bloque de soporte NO tiene señal. Esto es lo que hace posible
117 // que la antorcha actúe como NOT gate en circuitos reales. Solo maneja
118 // antorchas orientadas hacia abajo (colocadas sobre un bloque) — es el
119 // caso que cubre notifyNeighbors hoy; antorchas en pared quedan como
120 // extensión futura documentada (ver comentario en el .cpp).
121 void updateRedstoneTorchAt(cursestone::world::World* world, cursestone::Server* server,
122  int32_t x, int32_t y, int32_t z);
123 
124 // Recalcula si una puerta (madera o hierro) debe estar abierta o
125 // cerrada según si su mitad INFERIOR está recibiendo señal de
126 // redstone. La puerta ocupa 2 bloques verticales; sólo la mitad
127 // inferior guarda el bit de orientación+abierta (0x04) y sólo ella se
128 // consulta para señal — igual que en vanilla. Si (x,y,z) es la mitad
129 // superior, la función localiza la inferior automáticamente (y-1) para
130 // que notifyNeighbors pueda llamarla con cualquiera de las dos mitades
131 // sin distinguir cuál es cuál. Sincroniza el bit 0x04 en ambas mitades
132 // y transmite el cambio de las que hayan cambiado.
133 void updateDoorAt(cursestone::world::World* world, cursestone::Server* server,
134  int32_t x, int32_t y, int32_t z);
135 
136 // Recalcula si un pistón (normal o pegajoso) en (x,y,z) debe estar
137 // extendido o retraído según si está recibiendo señal de redstone.
138 // Al extender: si el bloque inmediatamente enfrente (según la
139 // dirección guardada en los bits 0-2 de `data`) es aire o un bloque
140 // reemplazable, coloca ahí un PISTON_HEAD y empuja ese contenido fuera
141 // (lo reemplaza por aire, sin generar drop — igual que vanilla, que no
142 // dropea el bloque empujado, lo reubica). Empuje limitado a UN bloque
143 // simple no adyacente a otro sólido: cadenas largas de bloques
144 // empujados y el límite de 12 bloques de vanilla quedan como extensión
145 // futura documentada (ver comentario en el .cpp) — cubre el circuito
146 // típico (pistón empujando 1 bloque) sin arriesgar recorrer cadenas
147 // sin límite. Al retraer: quita el PISTON_HEAD (vuelve a aire); un
148 // pistón pegajoso además intentaría traer de vuelta el bloque
149 // empujado, lo cual también queda documentado como extensión futura.
150 void updatePistonAt(cursestone::world::World* world, cursestone::Server* server,
151  int32_t x, int32_t y, int32_t z);
