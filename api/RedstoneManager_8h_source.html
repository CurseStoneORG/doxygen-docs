23 // RedstoneManager.h — redstone signal propagation engine.
24 //
25 // Design (hardened after the BlockFire v6 88s freeze from unbounded
26 // cascades): wire propagation is capped at 15 power steps (a hard
27 // mathematical bound, not a heuristic); every update is a synchronous,
28 // bounded computation finishing within the tick that requested it;
29 // notifyNeighbors() is the only entry point external code calls.
30 //
31 // FIX: state changes are broadcast with UpdateBlockPacket (+ sound /
32 // particle when applicable) the same way Player.cpp does for
33 // break/place, so propagation is visible live instead of only after
34 // reconnecting.

85 // Intenta interactuar con un componente de redstone accionable por el
86 // player (lever, button, wooden door — iron doors never react to
87 // clicks, only to redstone, like vanilla) at (x,y,z). If the block is
88 // one of those types,
89 // alterna/activa su estado, actualiza el mundo, transmite el cambio
90 // and returns true. Levers and buttons additionally notify neighbors
91 // (pueden alimentar un circuito); la puerta NO lo hace al recibir
92 // manual click — see the .cpp note for why. Non-interactive blocks do
93 // nothing and return false, so the caller (Player) knows to continue
94 // with its own item placement logic.
95 //
96 // `player` identifies who toggled the component (for DoorToggleEvent;
97 // nullable). Doors also emit the vanilla open/close sound
98 // (LevelEventPacket evid=1003 — fix de "las puertas no emiten sonido")
99 // y dispara DoorToggleEvent, cancelable.
100 bool tryInteract(cursestone::world::World* world, cursestone::Server* server, int32_t x, int32_t y,
101  int32_t z, cursestone::Player* player = nullptr);

103 // Recalcula si una antorcha de redstone en (x,y,z) debe estar
104 // encendida (id REDSTONE_TORCH_ON) o apagada (id REDSTONE_TORCH_OFF).
105 // Vanilla behavior (the torch is a logical INVERTER, not a source like
106 // simple foco): se apaga si el bloque que la sostiene (el vecino hacia
107 // it points at, usually below) receives signal, and lights up when that
108 // support has none — the NOT gate of real circuits. Only the standing
109 // antorchas orientadas hacia abajo (colocadas sobre un bloque) — es el
110 // caso que cubre notifyNeighbors hoy; antorchas en pared quedan como
111 // documented future work (see the .cpp comment).
112 void updateRedstoneTorchAt(cursestone::world::World* world, cursestone::Server* server, int32_t x,
113  int32_t y, int32_t z);

115 // Recalcula si una puerta (madera o hierro) debe estar abierta o
116 // on whether its LOWER half receives redstone signal. Doors span two
117 // vertical blocks; only the lower half stores the hinge+open bit (0x04)
118 // and only it is queried — like vanilla. Given the upper half, the
119 // function resolves the lower one (y-1) automatically
120 // que notifyNeighbors pueda llamarla con cualquiera de las dos mitades
121 // without needing to know which was passed. Syncs bit 0x04 across both
122 // halves
123 // y transmite el cambio de las que hayan cambiado.
124 void updateDoorAt(cursestone::world::World* world, cursestone::Server* server, int32_t x, int32_t y,
125  int32_t z);

127 // Recomputes whether a piston (regular or sticky) at (x,y,z) should be
128 // extended. Extending: if the block in front (facing from bits 0-2 of
129 // `data`) is air or replaceable, places a PISTON_HEAD there and pushes
130 // that content out
131 // (lo reemplaza por aire, sin generar drop — igual que vanilla, que no
132 // dropea el bloque empujado, lo reubica). Empuje limitado a UN bloque
133 // and only when it is not adjacent to another solid: long push chains
134 // and the vanilla 12-block limit are documented future work
135 // futura documentada (ver comentario en el .cpp) — cubre el circuito
136 // for the typical case (pushing a single block) without risking
137 // unbounded chains. Retracting removes the PISTON_HEAD; a sticky piston
138 // pulling the block back is also documented future work.
139 void updatePistonAt(cursestone::world::World* world, cursestone::Server* server, int32_t x,
140  int32_t y, int32_t z);
