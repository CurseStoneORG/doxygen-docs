24 // DBTable — the .ldb SSTable codec: builds and reads the real LevelDB table
25 // format (data blocks with restart points, block trailer + compression byte,
26 // index block, metaindex, 4096-byte footer with the leveldb magic).
27 //
28 // Reads tolerate every block compression the target ecosystem writes:
29 // raw (0), snappy (1, real MCPE device worlds) and zlib (2, PocketMine).
30 // Writes emit zlib level-1 blocks — valid for every reader of the format
31 // and the best speed/size trade for a live server (see Anvil's notes on
32 // compression level).

43 // One (internal key, value) row. value.empty() + type deletion encodes a
44 // tombstone via the trailer type instead of a payload type flag.
45 struct TableEntry {
46  std::string key; // internal key (user key + 8-byte trailer)
47  std::string value; // payload bytes ("" for tombstones)
48 };
49 
50 // Encode one table from SORTED entries (internal_key_less order) into a
51 // complete .ldb image. Returns false on internal error (allocation/deflate).
52 bool build_table(const std::vector<TableEntry>& entries, std::vector<uint8_t>* out);
53 
54 // Parsed view of one data block: entries in key order.
55 struct ParsedBlock {
56  std::vector<TableEntry> entries;
57 };
58 
59 // Reader over a finished table image. The image stays owned by the caller
60 // and must outlive the reader. The index is parsed once; data blocks are
61 // decompressed on demand and never cached (the OS page cache handles it).
62 class TableReader {
63  public:
64  // Parses the footer + index. Returns nullptr-class (ok()==false) on
65  // corruption.
66  bool open(const uint8_t* image, size_t len);
67 
68  // Number of entries claimed by the index (upper bound, informational).
69  size_t block_count() const { return index_.size(); }
70 
71  // Point lookup by internal key. Returns the newest entry for the user
72  // key of `internal` (the caller passes the target internal key; the
73  // search matches user keys per leveldb semantics). Sets *found.
74  bool get(const std::string& internal, TableEntry* out) const;
75 
76  // Full scan in key order (used by compaction and listChunks).
77  void scan(std::vector<TableEntry>* out) const;
78 
79  // Boundaries as recorded in the index.
80  const std::string& first_key_hint() const { return index_.front().separator; }
81  const std::string& last_key_hint() const { return last_key_; }
82 
83  private:
84  // BlockHandle: varint64 offset + varint64 size.
85  struct IndexEntry {
86  std::string separator; // ≥ all keys in the block
87  uint64_t offset = 0;
88  uint64_t size = 0;
89  };
90 
91  static bool read_handle(const uint8_t* data, size_t len, size_t* pos,
92  uint64_t* offset, uint64_t* size);
93  // Parses one (possibly compressed) block body into entries.
94  static bool parse_block(const uint8_t* data, size_t len, ParsedBlock* out);
95  // Fetches + decompresses one block by handle; also verifies its CRC
96  // trailer (last 5 bytes: crc32c(4) + compression type(1), CRC over the
97  // uncompressed bytes).
98  bool read_block(uint64_t offset, uint64_t size, ParsedBlock* out) const;
99 
100  const uint8_t* image_ = nullptr;
101  size_t image_len_ = 0;
102  std::vector<IndexEntry> index_;
103  std::string last_key_; // largest internal key in the table
104 };
