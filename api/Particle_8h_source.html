23 // Particle — abstract level particle. Particles are sent to players
24 // in range via LevelEventPacket (evid = EVENT_ADD_PARTICLE_MASK | type).
25 // The server-side particle object only computes position and forwards it
26 // to the network layer.
27 #pragma once
28 #include "world/Position.h"
29 #include <cstdint>
30 namespace cursestone::world::particle {
31 
32 // Particle types of protocol 84. The id travels in the low 12 bits of the
33 // LevelEventPacket event field (0x4000 | type) and the client maps it to
34 // its own particle renderer.
35 enum ParticleId : int32_t {
36  PARTICLE_BUBBLE = 1,
37  PARTICLE_CRITICAL = 2,
38  PARTICLE_SMOKE = 3,
39  PARTICLE_EXPLODE = 4,
40  PARTICLE_WHITE_SMOKE = 5,
41  PARTICLE_FLAME = 6,
42  PARTICLE_LAVA = 7,
43  PARTICLE_LARGE_SMOKE = 8,
44  PARTICLE_REDSTONE = 9,
45  PARTICLE_ITEM_BREAK = 10,
46  PARTICLE_SNOWBALL_POOF = 11,
47  PARTICLE_LARGE_EXPLODE = 12,
48  PARTICLE_HUGE_EXPLODE = 13,
49  PARTICLE_MOB_FLAME = 14,
50  PARTICLE_HEART = 15,
51  PARTICLE_TERRAIN = 16,
52  PARTICLE_TOWN_AURA = 17,
53  PARTICLE_PORTAL = 18,
54  PARTICLE_WATER_SPLASH = 19,
55  PARTICLE_WATER_WAKE = 20,
56  PARTICLE_DRIP_WATER = 21,
57  PARTICLE_DRIP_LAVA = 22,
58  PARTICLE_DUST = 23,
59  PARTICLE_MOB_SPELL = 24,
60  PARTICLE_MOB_SPELL_AMBIENT = 25,
61  PARTICLE_MOB_SPELL_INSTANT = 26,
62  PARTICLE_INK = 27,
63  PARTICLE_SLIME = 28,
64  PARTICLE_RAIN_SPLASH = 29,
65  PARTICLE_VILLAGER_ANGRY = 30,
66  PARTICLE_VILLAGER_HAPPY = 31,
67  PARTICLE_ENCHANTMENT_TABLE = 32,
68 };
69 class Particle {
70 public:
71  // data = event payload (particle scale / color), protocol 84.
72  Particle(ParticleId id, Position pos, int32_t data = 0)
73  : id_(id), pos_(pos), data_(data) {}
74  ParticleId id() const { return id_; }
75  const Position& position() const { return pos_; }
76  // Wire type id (low 12 bits of the LevelEventPacket event field).
77  virtual int32_t networkId() const { return (int32_t)id_; }
78  // Event payload carried in the LevelEventPacket data field.
79  virtual int32_t networkData() const { return data_; }
80  virtual ~Particle() = default;
81 private:
82  ParticleId id_;
83  Position pos_;
84  int32_t data_;
85 };
86 } // namespace
