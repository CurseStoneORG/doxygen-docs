24 // requests to the active WorldProvider. Mirrors pocketmine's
25 // ChunkManager role. The World owns one of these. Generators run

48  // Non-blocking chunk fetch used by the per-player streaming path
49  // (Player::onUpdate). Never generates on the calling thread:
50  // - if the chunk is already in memory, returns it immediately.
51  // - otherwise, if a generation request for (x,z) isn't already
52  // in flight, submits one to `pool` (runs generate()+populate()
53  // on a worker thread, per Generator's "stateless w.r.t. main
54  // thread" contract) and returns nullptr right away.
55  // - if a request IS already in flight, just returns nullptr
56  // again (no duplicate work submitted).
57  // The caller should treat nullptr as "not ready yet, try again
58  // next tick" rather than as a permanent miss.
59  mcp_chunk_t* getChunkOrRequestAsync(int32_t x, int32_t z, scheduler::AsyncPool* pool);
60 
61  void setChunk(int32_t x, int32_t z, mcp_chunk_t* chunk);
62  // Called every time a chunk enters the manager (generated, loaded
63  // from disk, or replaced). World installs a callback here so a chunk
64  // that appears later than its neighbours still gets its lighting
65  // swept together with them — without it a chunk keeps the lighting it
66  // was generated with in isolation and shows a dark border where the
67  // light should carry over. Must not block or touch ChunkManager:
68  // setChunk() runs on the generation worker thread.
69  void setOnChunkAdded(std::function<void(int32_t, int32_t)> cb) {
70  on_chunk_added_ = std::move(cb);
71  }
72  void markDirty(int32_t x, int32_t z);
73  // Escribe TODO lo sucio de golpe, con mtx_ cogido de principio a fin.
74  // Bloquea al que llama (y a los workers de generacion que esperan ese
75  // lock) hasta que termina. Solo para las rutas que necesitan el mundo
76  // entero en disco: apagado / saveAll() / comando /save.
77  int flushDirty();
78  // Igual que flushDirty() pero con PRESUPUESTO: escribe como maximo
79  // `max_chunks` chunks sucios y se corta antes si ya consumio
80  // `max_ms` milisegundos. Devuelve cuantos escribio.
81  //
82  // Es la version que usa el autosave: llamandola cada tick, el coste
83  // de guardar el mundo queda repartido entre varios ticks en vez de
84  // caer como un pico unico en el tick loop. Medido antes de este
85  // cambio, con chunks de terreno real (generador Normal): 400 chunks
86  // sucios = ~300 ms de una sola vez (y 759 ms en el caso reportado,
87  // con el mundo mas cargado, ~1000 chunks) — eso es el congelamiento
88  // que el cliente acaba leyendo como desconexion.
89  int flushDirtyBudget(size_t max_chunks, uint64_t max_ms);
90  // Frees a single chunk's memory. If `save` is true and a provider
91  // is set, the chunk is saved first (only if it's actually dirty —
92  // an unmodified chunk doesn't need a disk write just because it's
93  // being evicted). Returns false if the chunk wasn't loaded.
94  //
95  // NOTE: this does NOT touch tile-entities living in that chunk —
96  // World::unloadDistantChunks() calls World::evictTilesForChunk()
97  // first to flush/drop them, because ChunkManager has no visibility
98  // into World::tiles_ (tile-entity registry lives one layer up).
99  // Calling this directly without doing that first would still work
100  // (the chunk's mcp_chunk_t::tile_entities blob only gets rewritten
101  // by World::syncTilesToChunks()/evictTilesForChunk() in the first
102  // place), but any tile whose dirty NBT was never flushed into the
103  // chunk stays unsaved, and World::tiles_ would go on holding a Tile
104  // object for a chunk that's no longer in memory.
105  bool unloadChunk(int32_t x, int32_t z, bool save = true);
106  // Coordinates of every chunk currently held in memory. The caller
107  // (World::unloadDistantChunks()) uses this to decide what's
108  // eligible for eviction. Returns a snapshot copy rather than a
109  // reference so the caller can iterate it without holding mtx_ for
110  // the whole sweep (each unloadChunk() call above takes the lock
111  // itself, per-chunk).
112  std::vector<std::pair<int32_t,int32_t>> loadedChunkCoords() const;
113  // Numero de chunks en memoria con cambios sin escribir. El autosave
114  // lo usa para saber si tiene que seguir drenando el flush.
115  size_t dirtyCount() const;
116  void setProvider(std::shared_ptr<format::WorldProvider> p) { provider_ = std::move(p); }
117  std::shared_ptr<format::WorldProvider> provider() const { return provider_; }
118  // Set the world generator used to create new chunks when they're
119  // not on disk. Mirrors Nukkit where the generator is queried
120  // for terrain + population.
121  void setGenerator(std::shared_ptr<generator::Generator> g) { generator_ = std::move(g); }
122  size_t count() const;
123 private:
124  struct ChunkKey {
125  int32_t x, z;
126  bool operator==(const ChunkKey& o) const { return x==o.x && z==o.z; }
127  };
128  struct ChunkKeyHash {
129  size_t operator()(const ChunkKey& k) const noexcept {
130  return std::hash<int64_t>()((int64_t)k.x * 0x9e3779b97f4a7c15LL ^ (int64_t)k.z);
131  }
132  };
133  mutable std::mutex mtx_;
134  std::function<void(int32_t, int32_t)> on_chunk_added_;
135  std::unordered_map<ChunkKey, mcp_chunk_t*, ChunkKeyHash> chunks_;
136  std::shared_ptr<format::WorldProvider> provider_;
137  std::shared_ptr<generator::Generator> generator_;
138 
139  // Coordinates currently being generated on an AsyncPool worker,
140  // so getChunkOrRequestAsync() doesn't submit the same chunk twice
141  // while it's still in flight. Guarded by pending_mtx_ (separate
142  // from mtx_ since the worker thread touches chunks_ via setChunk()
143  // — which takes mtx_ itself — right before it clears this set).
144  mutable std::mutex pending_mtx_;
145  std::unordered_set<ChunkKey, ChunkKeyHash> pending_;
146 };
