24 // AnvilRegionLoader.h — single .mca region file (32x32 chunks).
25 //
26 // Ported from Nukkit InfinityEdition's
27 // `cn.nukkit.level.format.anvil.RegionLoader` and
28 // `cn.nukkit.level.format.generic.BaseRegionLoader` to C++.
29 //
30 // File layout (vanilla Minecraft Java / Nukkit Anvil):
31 // [ 0 .. 4095] Location table — 1024 entries × 4 bytes BE
32 // Each entry: (sectorOffset << 8) | sectorCount
33 // offset == 0 → chunk not present
34 // [4096 .. 8191] Timestamp table — 1024 entries × 4 bytes BE
35 // Unix-seconds of last write per chunk slot.
36 // [8192 ..] Chunk sectors (each 4096 bytes).
37 // Each chunk: [4-byte length BE][1-byte compression][payload]
38 // compression: 1 = GZIP, 2 = ZLIB (Nukkit always writes ZLIB).
39 //
40 // The chunk-slot index inside a region for a local (lx, lz) is
41 // index = lx + (lz << 5) (= z * 32 + x).

53 constexpr int32_t kRegionSize = 32; // 32x32 chunks per region

57 constexpr size_t kTimestampTableSize = 4096;

65  uint8_t count = 0; // in sectors
66  uint32_t timestamp = 0; // unix seconds

74  // Open (or create) the region file at `path` for region (rx, rz).
75  // On a brand-new file, writes a blank 8 KiB header (location table
76  // all-zero, timestamp table filled with current unix time).
77  bool open(const std::string& path, int32_t rx, int32_t rz);
78  void close();
79 
80  // Read the chunk payload bytes at local (lx, lz).
81  // Returns std::nullopt if the slot is empty / corrupt.
82  // The returned bytes are the *compressed* chunk payload (without
83  // the 4-byte length / 1-byte compression prefix); callers must
84  // inflate (ZLIB) before NBT decoding.
85  std::optional<std::vector<uint8_t>> readChunk(int32_t lx, int32_t lz);
86 
87  // Write compressed payload at slot (lx, lz). The compression byte
88  // is written by this function; `payload` is the already-compressed
89  // bytes (ZLIB). Returns false on I/O error or chunk-too-big.
90  bool writeChunk(int32_t lx, int32_t lz, uint8_t compression,
91  const std::vector<uint8_t>& payload);
92 
93  bool hasChunk(int32_t lx, int32_t lz) const;
94  int32_t regionX() const {
95  return region_x_;
96  }
97  int32_t regionZ() const {
98  return region_z_;
99  }
100  long lastUsed() const {
101  return last_used_;
102  }
103 
104  // Compute the chunk-slot index from local (lx, lz).
105  static inline size_t slotIndex(int32_t lx, int32_t lz) {
106  return (size_t)((lx & 31) + ((lz & 31) << 5));
107  }
108 
109  private:
110  bool loadLocationTable();
111  bool writeLocationTable();
112  bool writeLocationIndex(size_t index);
113  bool createBlank();

115  std::string path_;

117  int32_t region_x_ = 0;

119  uint32_t last_sector_ = 1; // next free sector (1-based; 0 = header)
120  std::array<LocationEntry, kChunkSlotCount> table_{};
121  long last_used_ = 0;
122 };
