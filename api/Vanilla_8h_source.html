300 struct Color {
301  uint8_t r = 0, g = 0, b = 0, a = 255;
302  Color() = default;
303  Color(uint8_t rr, uint8_t gg, uint8_t bb, uint8_t aa = 255) : r(rr), g(gg), b(bb), a(aa) {}
304  static const Color NIL;
305  static Color fromHSB(float hue, float saturation, float brightness) {
306  float h = hue - ::floorf(hue);
307  h = h * 6.0f;
308  float f = h - ::floorf(h);
309  float p = brightness * (1.0f - saturation);
310  float q = brightness * (1.0f - f * saturation);
311  float t = brightness * (1.0f - (1.0f - f) * saturation);
312  float r, g, b;
313  switch ((int)h % 6) {
314  case 0: r = brightness; g = t; b = p; break;
315  case 1: r = q; g = brightness; b = p; break;
316  case 2: r = p; g = brightness; b = t; break;
317  case 3: r = p; g = q; b = brightness; break;
318  case 4: r = t; g = p; b = brightness; break;
319  default: r = brightness; g = p; b = q; break;
320  }
321  return Color((uint8_t)(r * 255.0f), (uint8_t)(g * 255.0f), (uint8_t)(b * 255.0f));
322  }
323  static Color fromARGB(uint32_t argb) {
324  return Color((uint8_t)((argb >> 16) & 0xff), (uint8_t)((argb >> 8) & 0xff),
325  (uint8_t)(argb & 0xff), (uint8_t)((argb >> 24) & 0xff));
326  }
327  bool operator==(const Color& o) const { return r == o.r && g == o.g && b == o.b && a == o.a; }
328  bool operator!=(const Color& o) const { return !(*this == o); }
329  uint32_t toARGB() const { return ((uint32_t)a << 24) | ((uint32_t)r << 16) | ((uint32_t)g << 8) | b; }
330 };
