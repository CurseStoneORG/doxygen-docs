131 // Returns the creative items as ItemStack values (count=1, meta=damage).
132 inline std::vector<cursestone::item::ItemStack> getCreativeItems() {
133  const auto& table = creativeItemTable();
134  std::vector<cursestone::item::ItemStack> out;
135  out.reserve(table.size());
136  for (const auto& e : table) {
137  cursestone::item::ItemStack s;
138  s.id = e.id;
139  s.meta = e.damage;
140  s.count = 1;
141  out.push_back(s);
142  }
143  return out;
144 }
