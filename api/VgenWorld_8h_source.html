118  DataID getData(const ChunkBlockPos& p) const {
119  uint32_t idx = (uint32_t)p.y + ((uint32_t)p.z + (uint32_t)p.x * 16) * 128;
120  const NibblePair& pair = mData[idx / 2];
121  return idx % 2 == 0 ? (DataID)pair.first : (DataID)pair.second;
122  }
123  void setBlock(const ChunkBlockPos& p, BlockID id) {
124  uint32_t col = ((uint32_t)p.z + (uint32_t)p.x * 16) * 128;
125  uint32_t idx = (uint32_t)p.y + col;
126  uint8_t oldId = (uint8_t)mBlocks[idx];
127  mBlocks[idx] = id;
128  const int slot = p.x | (p.z << 4);
129  mRainHeights[slot] = -999;
130  // Incremental heightmap update — same result as rescanning the
131  // whole column, O(1) for the common cases:
132  // * invariant: blocks[mHeightmap-1] is the topmost block with
133  // mLightBlock != 0 and every block above it is transparent;
134  // * placing an opaque block at/above the top raises the map;
135  // * clearing exactly the top block rescans downward from it.
136  // Decorators and structures write thousands of blocks per
137  // chunk, so the per-write column rescan dominated generation.
138  uint8_t nl = Block::mLightBlock[(uint8_t)id.value];
139  if (nl != 0) {
140  if ((int)p.y >= mHeightmap[slot])
141  mHeightmap[slot] = (int)p.y + 1;
142  } else if (Block::mLightBlock[oldId] != 0 && (int)p.y == mHeightmap[slot] - 1) {
143  int y = p.y;
144  while (y > 0 && Block::mLightBlock[(uint8_t)mBlocks[(uint32_t)(y - 1) + col]] == 0)
145  --y;
146  mHeightmap[slot] = y;
147  }
148  }
149  void setData(const ChunkBlockPos& p, DataID v) {
150  uint32_t idx = (uint32_t)p.y + ((uint32_t)p.z + (uint32_t)p.x * 16) * 128;
151  NibblePair& pair = mData[idx / 2];
152  if (idx % 2 == 0)
153  pair.first = (uint8_t)v;
154  else
155  pair.second = (uint8_t)v;
156  }
