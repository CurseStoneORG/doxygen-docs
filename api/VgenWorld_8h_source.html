1 // CurseStone | vgen — world-surface shim (LevelChunk / Level / Dimension /
2 // BlockSource) over a 3x3 virtual neighborhood.
3 //
4 // The vanilla population step (RandomLevelSource::postProcess) runs against
5 // a 3x3 chunk neighborhood reached through a BlockSource. CurseStone
6 // generates and populates one chunk at a time on AsyncPool workers, so
7 // this shim materializes the neighborhood virtually:
8 //
9 // * the CENTER chunk is a working copy of the real sink chunk;
10 // * the 8 neighbors are deterministic terrain-only chunks pulled from
11 // VgenSource's terrain cache (generated on demand — the vanilla
12 // terrain pipeline is seed-deterministic per chunk);
13 // * every write aimed at a NON-center chunk is also recorded as a
14 // "spill" on the source; when that neighbor populates later it
15 // replays the spill before running its own features, which gives the
16 // exact vanilla cross-chunk feature behavior (trees spanning chunk
17 // borders, lake edges, structure pieces) without the engine ever
18 // having to hold 9 real chunks.
19 //
20 // Heightmap semantics stay vanilla: mHeightmap = topmost light-blocking
21 // block, getTopRainBlockPos = first y whose material blocks motion or is
22 // liquid, plus one.

54 class ITerrainFeed {
55 public:
56  virtual ~ITerrainFeed() = default;
57  // Terrain-only chunk for (cx,cz); may come from the cache or be
58  // generated on demand. The pointer stays owned by the feed.
59  virtual LevelChunk* acquireTerrain(int cx, int cz) = 0;
60  // Record a write aimed outside the center chunk so the owning chunk
61  // can replay it when it populates.
62  virtual void recordSpill(int cx, int cz, uint32_t index, uint8_t id, uint8_t data) = 0;
63  // Replay every spill recorded for (cx,cz) into `center`.
64  virtual void applySpill(LevelChunk& center, int cx, int cz) = 0;
65  virtual Level& getLevel() = 0;
66  virtual Dimension& getDimension() = 0;
