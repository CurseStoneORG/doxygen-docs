23 // Command — base class for any registered command. Mirrors the reference server
24 // /command/Command. Subclasses define execute() which is invoked by
25 // SimpleCommandMap when a player or the console types /<name>.
26 #pragma once
27 #include "CommandSender.h"
28 #include <string>
29 #include <vector>
30 namespace cursestone {
31 class Command {
32  public:
33  // `permission` es el nodo de permiso que el emisor debe tener para
34  // poder ejecutar este comando. VACIO = cualquiera puede ejecutarlo
35  // (help, list, version...). Un nodo como "cursestone.command.stop"
36  // loMarco SimpleCommandMap::dispatch() antes de invocar execute(),
37  // de modo que un comando sin permiso NUNCA llega al execute() — ni
38  // siquiera si un comando forgets de revisar isOp() en su cuerpo.
39  Command(std::string name, std::string description = "", std::string usage = "",
40  std::string permission = "")
41  : name_(std::move(name)), description_(std::move(description)), usage_(std::move(usage)),
42  permission_(std::move(permission)) {
43  }
44  virtual ~Command() = default;
45  virtual bool execute(CommandSender* sender, const std::string& label,
46  const std::vector<std::string>& args) = 0;
47  const std::string& name() const {
48  return name_;
49  }
50  const std::string& description() const {
51  return description_;
52  }
53  const std::string& usage() const {
54  return usage_;
55  }
56  // Nodo requerido, o "" si es público.
57  const std::string& permission() const {
58  return permission_;
59  }
60  bool isPublic() const {
61  return permission_.empty();
62  }
63  // ¿Puede este emisor usar el comando? SimpleCommandMap::dispatch() lo
64  // consulta ANTES de execute() y /help lo usa para filtrar la lista.
65  // Por defecto: público, o el emisor tiene el nodo de permiso pedido.
66  // (LuaCommand lo sobreescribe: op / nodo custom con fallback a op.)
67  virtual bool canUse(CommandSender* sender) const {
68  return isPublic() || sender->hasPermission(permission_);
69  }
70  // Owner tag for commands registered at RUNTIME by plugins: LuaCommand
71  // returns its LuaPlugin*, core commands return nullptr. Used by
72  // /plugins reload to unregister a plugin's commands BEFORE its
73  // lua_State is destroyed (the command holds Lua refs).
74  virtual void* owner() const {
75  return nullptr;
76  }
77 
78  private:
79  std::string name_, description_, usage_, permission_;
80 };
81 } // namespace
