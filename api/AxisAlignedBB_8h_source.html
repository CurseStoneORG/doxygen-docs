45 static inline mcp_aabb_t mcp_aabb_from_corners(mcp_vector3_t a, mcp_vector3_t b) {
46  mcp_aabb_t out;
47  out.minX = (a.x < b.x) ? a.x : b.x;
48  out.minY = (a.y < b.y) ? a.y : b.y;
49  out.minZ = (a.z < b.z) ? a.z : b.z;
50  out.maxX = (a.x > b.x) ? a.x : b.x;
51  out.maxY = (a.y > b.y) ? a.y : b.y;
52  out.maxZ = (a.z > b.z) ? a.z : b.z;
53  return out;
54 }

93 // Swept-move primitives — net.minecraft.world.phys.AABB (2013 MCPE
94 // client). Entity::move() is built entirely from these four functions
95 // plus the three clip?Collide() below, so they are the ground truth for
96 // "where does this box end up after being pushed by (dx,dy,dz)".

99 // `expand` (Genisys AxisAlignedBB::addCoord): grow the box by the
100 // offset ON THE SIDE the offset pushes towards, so the result covers
101 // BOTH the original box and the box translated by (dx,dy,dz). This is
102 // the region a swept move has to look at.
103 static inline mcp_aabb_t mcp_aabb_expand(mcp_aabb_t b, double dx, double dy, double dz) {
104  double x0 = b.minX, y0 = b.minY, z0 = b.minZ;
105  double x1 = b.maxX, y1 = b.maxY, z1 = b.maxZ;
106  if (dx < 0)
107  x0 += dx;
108  if (dx > 0)
109  x1 += dx;
110  if (dy < 0)
111  y0 += dy;
112  if (dy > 0)
113  y1 += dy;
114  if (dz < 0)
115  z0 += dz;
116  if (dz > 0)
117  z1 += dz;
118  return mcp_aabb_make(x0, y0, z0, x1, y1, z1);
119 }

121 // `cloneMove` (the client's AABB::cloneMove / the original's offset()): plain
122 // translation of the whole box. This — not expand() — is what the
123 // collision-cube lookup is fed today, matching the existing item
124 // physics exactly; expand() is kept for callers that want the full
125 // swept volume.
126 static inline mcp_aabb_t mcp_aabb_clone_move(mcp_aabb_t b, double dx, double dy, double dz) {
127  return mcp_aabb_make(b.minX + dx, b.minY + dy, b.minZ + dz, b.maxX + dx, b.maxY + dy,
128  b.maxZ + dz);
129 }

157 // The client's AABB::contains(Vec3): STRICT containment, every bound
158 // exclusive. Deliberately not mcp_aabb_contains_point() above, which is
159 // inclusive on every face — a ray that grazes a block corner has to
160 // read differently in the two, and Entity::move() relies on the strict
161 // one.
162 static inline bool mcp_aabb_contains_strict(mcp_aabb_t b, mcp_vector3_t p) {
163  return p.x > b.minX && p.x < b.maxX && p.y > b.minY && p.y < b.maxY && p.z > b.minZ &&
164  p.z < b.maxZ;
165 }

167 // ---- per-axis swept clipping ---------------------------------------
168 // calculateXOffset / calculateYOffset / calculateZOffset (Genisys
169 // AxisAlignedBB, what Entity::move() clips with). Given the moving box
170 // `b`, one obstacle `c` and the requested displacement along that axis,
171 // return the displacement that is actually free: `d` unchanged when the
172 // move does not push into `c`, or the exact remaining gap up to the
173 // obstacle's facing side when it does.
174 //
175 // Read the branches as "the obstacle is AHEAD of me in the direction I
176 // am moving": moving +x, `c` must start at or beyond my maxX, and the
177 // free distance is c.minX - b.maxX. (Note the 2013 leak's AABB.h spells
178 // these with c's min/max roles swapped — `c.x1 <= x0` — which would clip
179 // a rightward move against a block behind you. The form below is the one
180 // the working integrator used and the only one that produces correct
181 // collisions, so it is the one that is pinned by tools/phys_test.cpp.)
182 //
183 // `c` must already overlap `b` on the other two axes; that is what the
184 // two early-outs test.
185 static inline double mcp_aabb_clip_x_collide(mcp_aabb_t b, mcp_aabb_t c, double d) {
186  if (c.maxY <= b.minY || c.minY >= b.maxY)
187  return d;
188  if (c.maxZ <= b.minZ || c.minZ >= b.maxZ)
189  return d;
190  if (d > 0.0 && c.minX >= b.maxX) {
191  double gap = c.minX - b.maxX;
192  if (gap < d)
193  d = gap;
194  }
195  if (d < 0.0 && c.maxX <= b.minX) {
196  double gap = c.maxX - b.minX;
197  if (gap > d)
198  d = gap;
199  }
200  return d;
201 }
202 static inline double mcp_aabb_clip_y_collide(mcp_aabb_t b, mcp_aabb_t c, double d) {
203  if (c.maxX <= b.minX || c.minX >= b.maxX)
204  return d;
205  if (c.maxZ <= b.minZ || c.minZ >= b.maxZ)
206  return d;
207  if (d > 0.0 && c.minY >= b.maxY) {
208  double gap = c.minY - b.maxY;
209  if (gap < d)
210  d = gap;
211  }
212  if (d < 0.0 && c.maxY <= b.minY) {
213  double gap = c.maxY - b.minY;
214  if (gap > d)
215  d = gap;
216  }
217  return d;
218 }
219 static inline double mcp_aabb_clip_z_collide(mcp_aabb_t b, mcp_aabb_t c, double d) {
220  if (c.maxX <= b.minX || c.minX >= b.maxX)
221  return d;
222  if (c.maxY <= b.minY || c.minY >= b.maxY)
223  return d;
224  if (d > 0.0 && c.minZ >= b.maxZ) {
225  double gap = c.minZ - b.maxZ;
226  if (gap < d)
227  d = gap;
228  }
229  if (d < 0.0 && c.maxZ <= b.minZ) {

231  if (gap > d)

233  }
