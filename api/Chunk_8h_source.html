24 // Chunk — raw in-memory chunk, POD struct.
25 //
26 // A Chunk holds 16 ChunkSections stacked on Y (0..15), giving 16*16*256
27 // = 65536 blocks total. Each section is 16*16*16 = 4096 blocks. This
28 // matches the MCPE 0.15.x chunk model that the anvil / leveldb formats
29 // serialize (only the lower 8 sections are persisted, since MCPE 0.15.x
30 // is 128-tall).
31 //
32 // POD (not RAII): the storage layer stays memcpy/memset-able so it can be used from
33 // the RakLib thread and from generators without dragging C++ runtime
34 // overhead. The FullChunk C++ interface wraps this struct.
35 //
36 // Memory model: a Chunk owns all 16 section pointers (each
37 // malloc'd/free'd via mcp_chunk_new/mcp_chunk_destroy). A section
38 // may be NULL if the chunk hasn't been populated at that Y-range yet
39 // (lazy allocation, like Minecraft Java).

48 // 256-byte per-column biome array + 256-int heightmap live on the
49 // Chunk (not on the Section). A "column" is (x, z); the per-section
50 // blocks live in the section_t above.
51 typedef struct mcp_chunk {
52  int32_t x; // chunk X coordinate
53  int32_t z; // chunk Z coordinate
54  int64_t last_update; // unix ms of last disk write
55  bool terrain_populated; // generator ran terrain pass
56  bool light_populated; // generator ran light pass
57  bool dirty; // has unsaved block writes
58 
59  mcp_chunk_section_t* sections[16]; // NULL = not yet allocated
60  uint8_t* biomes; // 256 bytes (one byte per column)
61  int32_t* height_map; // 256 ints (one int per column)
62 
63  // Tile-entity and entity payload lists. Each entry is an opaque
64  // NBT blob (CompoundTag) — we store raw bytes for the C storage
65  // layer; the C++ World takes ownership of decoded entities.
66  uint8_t** tile_entities; // array of malloc'd NBT blobs
67  size_t* tile_entity_lens; // matching lengths
68  size_t tile_entity_count;
69  size_t tile_entity_cap;
70 
71  uint8_t** entities; // same shape as tile_entities above
72  size_t* entity_lens;
73  size_t entity_count;
74  size_t entity_cap;
75 } mcp_chunk_t;

114 // Real sky-light propagation pass (replaces the old "memset 0xFF and
115 // call it done" approach). Must be called AFTER
116 // mcp_chunk_recalculate_height_map, since it uses the heightmap to
117 // know where open sky starts per column. Fills every column with 15
118 // down to the heightmap surface, then decrements by each opaque
119 // block's light filter going down, and finally flood-fills sideways
120 // so light also wraps under overhangs / around leaves — same
121 // algorithm the reference server's SimpleChunkManager::populateSkyLight +
122 // updateBlockSkyLight uses (per-column bright-to-heightmap plus a
123 // horizontal BFS spread for occluded neighbours). Single-chunk only:
124 // does not read neighbouring chunks, so light right at a chunk edge
125 // under an overhang from the next chunk over may need a follow-up
126 // pass once neighbours are loaded (same limitation the reference server has before
127 // its cross-chunk light-populator queue runs).
128 void mcp_chunk_propagate_sky_light(mcp_chunk_t* c);
