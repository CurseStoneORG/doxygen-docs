24 // TerrainShape — pure, world-coordinate terrain functions.
25 //
26 // The whole point of this module is that terrain SHAPE and biome are a pure
27 // function of (seed, worldX, worldZ). Normal::generate() builds the height
28 // grid of a chunk through these functions, and the feature populators
29 // (trees) can evaluate the *neighbouring* chunks' columns without ever
30 // loading those chunks. That is what lets trees span chunk borders: every
31 // chunk re-renders the overlapping part of the trees of the 3x3 chunks
32 // around it, and because the shape is deterministic the pieces line up
33 // exactly.

45 // Owns the noise layers: terrain shape, climate and rivers.
46 //
47 // El ruido ya NO es propio: son las capas del PerlinNoise vanilla
48 // (noise/vanilla, copiado 1:1 de SSMGAlt/PocketEdition). Cada campo del
49 // generador se apoya en una de las ocho capas del juego completo vanilla, y
50 // el paquete entero se construye con la semilla del mundo, una sola vez.
51 // Un campo del generador: una capa del ruido vanilla + su calibracion.
52 //
53 // La calibracion no es capricho. El PerlinNoise vanilla suma las octavas al
54 // reves que el fbm que usaba el generador (la frecuencia se divide y la
55 // amplitud se multiplica) y ademas tiene offsets xo/yo/zo de hasta 256, asi
56 // que un campo de baja frecuencia muestreado sobre un continente sale
57 // practicamente constante: un desplazamiento (offset) de valor arbitrario
58 // mas un gradiente suave. Sin corregirlo, el clima queda sesgado (toda la
59 // humedad corrida hacia un lado) y el relieve se aplana.
60 //
61 // `offset` y `gain` se miden UNA vez por mundo y por campo, en el constructor
62 // (ver TerrainNoise): el offset es la media del campo en una zona de
63 // referencia y el gain lo devuelve a la desviacion que tenia el ruido
64 // anterior. Asi cada campo conserva la estadistica con la que estan pensados
65 // los umbrales y las mascaras del generador.
66 struct FieldNoise {
67  PerlinNoise& layer;
68  double gain = 1.0;
69  double offset = 0.0;
70 
71  int levelCount() const { return layer.levelCount(); }
72 };

74 struct TerrainNoise {
75  VanillaNoise vanilla; // las 8 capas, semilla y orden vanilla
76 
77  // Id unico de INSTANCIA. Los caches por hilo de TerrainShape.cpp lo
78  // usan para invalidarse solos: si una TerrainNoise se destruye y
79  // otra nace en la MISMA direccion (paso normal con los optionales de
80  // los populators), el uid es distinto y ningun hit viejo se sirve.
81  uint64_t uid = 0;
82 
83  // Cada campo se apoya en una de esas capas (el paquete completo sigue
84  // disponible en `vanilla`, que es quien las construye y las posee).
85  FieldNoise height; // continentes + colinas + crestas + detalle
86  FieldNoise temp; // clima: temperatura
87  FieldNoise humid; // clima: humedad
88  FieldNoise river; // red de rios (el valor cruza el 0)
89  FieldNoise surface; // grava + suelo sin arboles

91  explicit TerrainNoise(int64_t seed);
92  ~TerrainNoise();
93  TerrainNoise(const TerrainNoise&) = delete;
94  TerrainNoise& operator=(const TerrainNoise&) = delete;
95 };

108 // River proximity mask in [0,1]: 1 at the channel centre, 0 outside it.
109 // This is the NARROW mask — it decides where the water actually is, so it
110 // is deliberately thin: a vanilla river is a few blocks wide, not a lake.
111 // Rivers follow the zero contour of the river noise, so they form a
112 // connected network that cuts through hills and mountains.
113 double riverMask(const TerrainNoise& n, double wx, double wz);
114 
115 // Distance in BLOCKS from the river's centre line, estimated as
116 // |field| / |gradient|. This is the coordinate the whole river shape is
117 // built on: the terrain carve, the biome choice, the shoreline test and
118 // the bank all read it, so they can never disagree about where the river
119 // is or how wide it is.
120 double riverDistance(const TerrainNoise& n, double wx, double wz);
121 
122 // Wide, gentle mask in [0,1] shaping the VALLEY the river runs through.
123 // This is deliberately much broader than riverMask(): the water sits in a
124 // thin channel at the bottom of a wide, softly sloping valley, which is
125 // what vanilla does. Keeping the two separate is what stops the river from
126 // becoming a deep slot canyon with vertical walls.
127 double riverValley(const TerrainNoise& n, double wx, double wz);
128 
129 // Even wider and much shallower skirt around riverValley(). Together the two
130 // carve a valley with a long, gentle approach instead of a step down to the
131 // water — a single wide valley measurably made the banks steeper, because
132 // widening the mask only pushes the same drop further from the centre.
133 double riverShoulder(const TerrainNoise& n, double wx, double wz);
134 
135 // True when this column is a shoreline: at (or within a couple of blocks
136 // of) the water level AND within `radius` columns of open water.
137 bool isShore(const TerrainNoise& n, double wx, double wz, int h, int radius = 2);
138 
139 // Same test, but the water it touches must be OCEAN and not a river
140 // channel. This is what separates the two very different cases the
141 // reference generator makes: sandy beaches are the NORM along the sea,
142 // while a river bank is normally grass and only occasionally sand.
143 bool isOceanShore(const TerrainNoise& n, double wx, double wz, int h, int radius = 2);
144 
145 // Rare but LONG sand banks along rivers, in [0,1]. Deliberately built from
146 // a very low-frequency field so that when it does fire it runs alongside
147 // the river for hundreds of blocks instead of making round patches, and it
148 // is gated on the river valley so it can only ever appear next to water.
149 double riverSandBank(const TerrainNoise& n, double wx, double wz);
150 
151 // Continent field (the long-wavelength ocean/land shape), exposed so the
152 // biome choice and the height field can agree on where the coast is.
153 double continentField(const TerrainNoise& n, double wx, double wz);
154 
155 // How "plain" this column is, in [0,1]. Broad, flat regions that are
156 // strongly biased toward coastal lowland, so the big open grassland sits
157 // against the sea the way the reference does. The terrain amplitude and
158 // the biome choice both read this, which is what keeps "plains" meaning
159 // the same flat thing in both.
160 double plainsFactor(const TerrainNoise& n, double wx, double wz);
161 
162 // Climate at a world column, in [-1, 1]. Exposed so the terrain and the
163 // biome selection read the exact same temperature/humidity (the savanna
164 // flattening below has to agree with BiomeRegistry::select).
165 double tempAt(const TerrainNoise& n, double wx, double wz);

168 // ---- Plains variant -----------------------------------------------------
169 // Raised plains variant gate in [0,1]: broad patches that stand a few
170 // blocks ABOVE the surrounding plains and stay flat on top. Already gated
171 // on plainsFactor, so it can only ever fire on actual grassland.
172 double plainsPlateau(const TerrainNoise& n, double wx, double wz);
173 
174 // Forest patches inside the plains, in [0,1]. Broad woodland-scale
175 // patches that turn open grassland into a patch of forest (the biome
176 // selection uses the same mask, so the trees and the biome agree).
177 double plainsForest(const TerrainNoise& n, double wx, double wz);
178 
179 // ---- Savanna -----------------------------------------------------------
180 // How strongly this column is a savanna PLAIN, in [0,1]: warm and
181 // moderately dry (never as dry as the desert, never as wet as the jungle)
182 // and inside a broad patch mask, so savanna comes in long stretches of
183 // low-relief ground instead of a thin climate stripe.
184 double savannaFactor(const TerrainNoise& n, double wx, double wz);
185 
186 // Savanna canyon gate in [0,1]: a broad raised table ~25 blocks above the
187 // savanna plain, with steep canyon walls. Gated hard on the CORE of a
188 // savanna (savannaFactor), so it can never leak into plains, desert or
189 // jungle, and its own mask is thresholded sharply so the transition to the
190 // surrounding savanna reads as a canyon wall.
191 double savannaCanyon(const TerrainNoise& n, double wx, double wz);
192 
193 // Gravel patch mask in [0,1]: rare patches of gravel on the surface,
194 // clustered around rivers and coastlines the way vanilla scatters them.
195 double gravelPatch(const TerrainNoise& n, double wx, double wz, int h);

197 // Tree-free mask in [0,1]: rare open patches where no tree grows, biased
