23 // TileChest — chest block entity. Stores 27 item slots (small chest).
24 // Pairs with a paired ChestTile for double-chest pairing (handled
25 // by the inventory layer).
26 //
27 // The backing storage is a ChestInventory (src/inventory/ChestInventory.h),
28 // NOT a private std::vector<ItemStack> — ChestInventory is the single
29 // source of truth for slot contents. Container::setItem/getItem/removeItem
30 // below just delegate to it. This keeps TileChest as the one object
31 // stored in World's tile registry (see World::getTile/setTile) instead
32 // of having two parallel places (a raw vector here + an Inventory
33 // elsewhere) that could drift out of sync.
34 #pragma once
35 #include "Tile.h"
36 #include "Container.h"
37 #include "Nameable.h"
38 #include "Spawnable.h"
39 #include "item/ItemStack.h" // required for ItemStack return/param types below
40 #include "inventory/ChestInventory.h"
41 #include <cstdint>
42 #include <memory>

50 class TileChest : public Tile, public Container, public Nameable, public Spawnable {
51  public:
52  TileChest(cursestone::world::Position pos)
53  : Tile("Chest", pos),
54  inventory_(std::make_shared<cursestone::inventory::ChestInventory>()) {
55  }
56  int size() const override {
57  return SIZE;
58  }
59  void setItem(int slot, const cursestone::item::ItemStack& it) override;
60  cursestone::item::ItemStack getItem(int slot) const override;
61  void removeItem(int slot) override;
62  void spawnTo(void* /*player*/) override {
63  }
64  void writeNBT(mcp_tag_t* tag) const override;
65  void readNBT(const mcp_tag_t* tag) override;
66 
67  // The backing inventory. Player::openChestAt() hands this shared_ptr
68  // straight to InventoryOpenEvent / ContainerSetContentPacket — no
69  // copy, so edits made through either the Container interface above
70  // or directly via this Inventory land in the same 27 slots. A paired
71  // chest is opened through TileChest::getOrCreateDouble() instead,
72  // which wraps both halves without copying any slot.
73  std::shared_ptr<cursestone::inventory::ChestInventory> inventory() const {
74  return inventory_;
75  }
76 
77  // How many players currently have this chest open. Player::openChestAt/
78  // closeOpenChest keep this in sync (drives the lid open/close events).
79  // Multiple players CAN hold this chest open at once — slot changes are
80  // resynced to every co-viewer via Player::broadcastChestSlotToViewers,
81  // so all of them see the same live content.
82  int viewerCount() const {
83  return viewer_count_;
84  }
85  void addViewer() {
86  ++viewer_count_;
87  }
88  void removeViewer() {
89  if (viewer_count_ > 0)
90  --viewer_count_;
91  }
92 
93  // ---- Vanilla double-chest pairing ----
94  // The pairing is the vanilla one: each half stores the integer
95  // position of its partner ("pairx"/"pairz", same-Y implied) — the
96  // fields persist through the tile NBT, so a double chest saved and
97  // reloaded stays a double chest.
98  bool isPaired() const {
99  return has_pair_;
100  }
101  int32_t pairX() const {
102  return pair_x_;
103  }
104  int32_t pairZ() const {
105  return pair_z_;
106  }
107 
108  // Resolves the paired half through the world tile registry. Returns
109  // null when this chest is single, the chunk is not loaded, or the
110  // registered tile at the pair position is not a chest anymore.
111  std::shared_ptr<TileChest> pairTile(cursestone::world::World* lvl) const;
112 
113  // Vanilla Chest::createPair — writes the pairing onto BOTH halves and
114  // invalidates the cached double inventories.
115  void setPair(const std::shared_ptr<TileChest>& other);
116 
117  // Drops the pairing state of THIS half only (pair position + cached
118  // double inventory) and marks the tile dirty for the next save.
119  void clearPair();
120 
121  // Vanilla InventoryLargeChest for the pair a/b — built once and SHARED
122  // by both tiles so every viewer of either half works on the same 54
123  // slot space. Slot order follows the vanilla orientation rule: the
124  // east half of an X-axis pair (and the north half of a Z-axis pair)
125  // addresses slots 0-26. Returns null when a/b are not a valid pair.
126  static std::shared_ptr<cursestone::inventory::DoubleChestInventory> getOrCreateDouble(
127  const std::shared_ptr<TileChest>& a, const std::shared_ptr<TileChest>& b);
128 
129  private:
130  static constexpr int SIZE = 27;
131  std::shared_ptr<cursestone::inventory::ChestInventory> inventory_;
132  int viewer_count_ = 0;
133  // Vanilla "pairx"/"pairz" — the paired half's block position (same Y).
134  bool has_pair_ = false;
135  int32_t pair_x_ = 0;
136  int32_t pair_z_ = 0;
137  // Cached double window, always shared by both halves when set.
138  std::shared_ptr<cursestone::inventory::DoubleChestInventory> double_;
139 };
140 } // namespace
