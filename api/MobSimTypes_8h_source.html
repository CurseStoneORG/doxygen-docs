24 // MobSimTypes — the value types of the stage-3 parallel mob simulation
25 // (docs/MOB_AI.md section 9). One frame of the pass is a fixed-shape
26 // batch: the main thread collects MobSimInput records (immutable world
27 // snapshots + steering cell windows), the worker team fills MobSimOutput
28 // records per mob, and the main thread applies the results in batch
29 // order. Workers never touch World, chunks, mob records or shared
30 // containers: every value they read lives in the frame, every value
31 // they write lands in their own output slot.

70 struct MobSimCellGrid {
71  static constexpr int kRad = 1;
72  static constexpr int kRowsBelow = 1;
73  static constexpr int kRowsAbove = 2;
74  static constexpr int kW = 2 * kRad + 1;
75  static constexpr int kH = kRowsBelow + 1 + kRowsAbove;
76 
77  int32_t ax = 0, ay = 0, az = 0;
79  uint8_t ids[kW * kH * kW] = {};
80 
82  uint8_t at(int32_t bx, int32_t by, int32_t bz) const {
83  const int32_t dx = bx - ax + kRad;
84  const int32_t dy = by - (ay - kRowsBelow);
85  const int32_t dz = bz - az + kRad;
86  if (dx < 0 || dx >= kW || dy < 0 || dy >= kH || dz < 0 || dz >= kW)
87  return 0;
88  return ids[(size_t)(dy * kW + dx) * kW + dz];
89  }
92  bool isWater(int32_t bx, int32_t by, int32_t bz) const;
93 };
