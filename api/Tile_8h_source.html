23 // Tile — block entity base. Tiles live in chunks as CompoundTag
24 // entries on disk and are synced to clients via BlockEntityDataPacket.
25 // Each Tile subclass (TileChest, TileFurnace, TileSign) holds its
26 // runtime state and writes back to NBT on save.
27 //
28 // NOTE on the tag type: this project's NBT layer (src/nbt/) is pure C
29 // (`mcp_tag_t`, extern "C" — see nbt/tag/CompoundTag.h). writeNBT/readNBT
30 // use `mcp_tag_t*` directly instead of the `class CompoundTag*` this
31 // header used to forward-declare. That old forward-declare pointed at
32 // a C++ class that was never defined anywhere in the project (a leftover
33 // from an earlier design pass) — every real NBT compound in this
34 // codebase, including the one Anvil already reads/writes per-chunk
35 // (see level/format/anvil/AnvilChunkCodec.c), is an `mcp_tag_t` of type
36 // MCP_TAG_COMPOUND. Using the real type here is what lets writeNBT/
37 // readNBT actually be called with the tags AnvilChunkCodec hands them,
38 // instead of only type-checking against a phantom class that could
39 // never receive a real object without an unsafe cast.
40 #pragma once
41 #include <cstdint>
42 #include <memory>
43 #include <string> // required for std::string id_ and Tile(std::string, ...)
44 #include "world/Position.h"
45 #include "nbt/tag/Tag.h" // mcp_tag_t
46 namespace cursestone::tile {
47 class Tile {
48  public:
49  Tile(std::string id, cursestone::world::Position pos) : id_(std::move(id)), pos_(pos) {
50  }
51  virtual ~Tile() = default;
52  const std::string& getId() const {
53  return id_;
54  }
55  const cursestone::world::Position& position() const {
56  return pos_;
57  }
58  void markDirty() {
59  dirty_ = true;
60  }
61  bool isDirty() const {
62  return dirty_;
63  }
64  // Clears the dirty flag after a successful write to the chunk's
65  // tile_entities array (World::syncTilesToChunks calls this once a
66  // tile's NBT has actually been handed to mcp_chunk_add_tile_entity —
67  // NOT before, so a tile whose owning chunk isn't loaded stays dirty
68  // and gets retried on the next save()).
69  void clearDirty() {
70  dirty_ = false;
71  }
72  virtual void tick(uint64_t /*current_tick*/) {
73  }
74  // Serialise tile state into `tag` (a fresh MCP_TAG_COMPOUND owned by
75  // the caller — World::syncTilesToChunks() creates it, hands it here,
76  // then hands it to mcp_chunk_add_tile_entity()). Base implementation
77  // writes the fields every tile needs regardless of subclass: block
78  // position (x/y/z) and the "id" string the reference server/Nukkit-style loaders key
79  // on ("Chest", "Furnace", "Sign", ...). Subclasses override, call
80  // Tile::writeNBT() first, then add their own fields.
81  virtual void writeNBT(mcp_tag_t* tag) const;
82  // Client-facing spawn compound (the reference Spawnable::getSpawnCompound
83  // split): what BlockEntityDataPacket carries on the wire. Defaults to
84  // the disk layout; a tile whose client reads different keys than the
85  // save format overrides it. The 0.15 furnace screen binds BurnTime/
86  // CookTime at open time, so TileFurnace must send THOSE keys even
87  // though the save format keeps the legacy Nukkit ones.
88  virtual void writeSpawnNBT(mcp_tag_t* tag) const;
89  // Read tile state back from `tag`. Base implementation is a no-op
90  // (x/y/z/id are already known from how the tile was constructed —
91  // World::ensureTilesLoaded() reads x/y/z itself to decide which
92  // position to construct the Tile at, before readNBT ever runs).
93  virtual void readNBT(const mcp_tag_t* tag);
94 
95  protected:
96  std::string id_; // "Chest", "Furnace", "Sign", ...
97  cursestone::world::Position pos_;
98  bool dirty_ = true;
99 };
100 } // namespace
