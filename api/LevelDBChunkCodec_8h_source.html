24 // LevelDBChunkCodec — the MCPE 0.15.x records that ride inside the db/.
25 //
26 // The device writes one record family per chunk (little-endian everywhere):
27 //
28 // key = int32 LE chunkX + int32 LE chunkZ + 1 record byte
29 // 0x30 "0" LegacyTerrain blocks(32768) + data(16384) + sky(16384) +
30 // block light(16384) + heightmap(256) +
31 // biome colors(1024) = 83200 bytes
32 // (column layout, index = x<<11 | z<<7 | y)
33 // 0x66 "f" flags 1 byte: bit0 generated | bit1 populated |
34 // bit2 light populated (default 0x03)
35 // 0x76 "v" version 1 byte chunk format version (0x02)
36 // 0x31 "1" tiles stream of little-endian NBT compounds
37 // 0x32 "2" entities same stream layout
38 //
39 // This codec turns those records into/from the in-memory mcp_chunk_t, and
40 // handles level.dat: [int32 LE storage version = 3][int32 LE size]
41 // [uncompressed little-endian NBT] — the same header the real client's
42 // ExternalFileLevelStorage writes.

61  // Record byte for each key family (the real MCPE LevelChunk tags).
62  static constexpr uint8_t kRecordTerrain = 0x30; // "0"

64  static constexpr uint8_t kRecordEntities = 0x32; // "2"
65  static constexpr uint8_t kRecordFlags = 0x66; // "f"
66  static constexpr uint8_t kRecordVersion = 0x76; // "v"
67  static constexpr uint8_t kLegacyChunkVersion = 0x02; // device value
68  static constexpr size_t kLegacyTerrainSize = 83200;
69 
70  // Builds the 9-byte db key: LE int32 x, LE int32 z, record byte.
71  static std::vector<uint8_t> record_key(int32_t cx, int32_t cz, uint8_t record);
72 
73  // Inverse of mcp_chunk_snapshot_legacy_terrain_serialize: builds a live
74  // mcp_chunk_t (sections in XZY, heightmap, biomes, light) from the
75  // 83200-byte legacy record. Returns NULL on malformed input. Caller
76  // owns the chunk (mcp_chunk_destroy).
77  static mcp_chunk_t* decode_legacy_terrain(const uint8_t* data, size_t len,
78  int32_t cx, int32_t cz);
79 
80  // Splits a "1"/"2" record into individual little-endian compound blobs
81  // (one per tile entity / entity) without materializing a tag tree —
82  // each blob is a byte range ready for mcp_chunk_add_tile_entity.
83  static bool split_compound_stream(const uint8_t* data, size_t len,
84  std::vector<std::vector<uint8_t>>* out);
85 
86  // --- level.dat ---------------------------------------------------------
87 
88  // Reads + parses level.dat (or level.dat_old as fallback). On success
89  // *root owns the parsed little-endian NBT root compound and
90  // *storage_version is the header version (3 on every modern file).
91  static bool read_level_dat(const std::string& world_dir,
92  cursestone::nbt::Tag** root,
93  int32_t* storage_version);
94 
95  // Serializes `root` (named little-endian compound) with the
96  // [version 3][size][NBT] header and writes it as level.dat; the
97  // previous level.dat, when present, is renamed to level.dat_old first
98  // (the device's rotate-on-save behavior and the crash-safety seam).
99  static bool write_level_dat(const std::string& world_dir,
100  const cursestone::nbt::Tag* root);
101 };
