22 // MobChunkCodec — the serialization seam between the world's mob records
23 // and the chunk entity lists inside the region files (.mca).
24 //
25 // Mobs are chunk state, exactly like blocks and tile entities: every
26 // chunk's Anvil NBT carries an "Entities" list and each mob rides it as
27 // one little-endian NBT compound blob (the same blob channel
28 // mcp_chunk_t already round-trips through anvil_chunk_encode/decode).
29 // The World rebuilds a chunk's list from its live records whenever the
30 // mobs of that chunk changed, and re-creates records from the list when
31 // the chunk (re)enters the level — so a mob lives and dies with its
32 // chunk and there is no side file left to drift out of sync.
33 //
34 // Blob layout (one per mob, little-endian like every chunk-embedded
35 // blob; the region writer re-serializes big-endian):
36 //
37 // CompoundTag {
38 // "id" : String — registry name ("sheep", "zombie_pigman")
39 // "Pos" : List<FLOAT>[3] — feet position
40 // "Rotation" : List<FLOAT>[2] — yaw, pitch
41 // "Health" : FLOAT
42 // "MaxHealth": FLOAT
43 // }
44 //
45 // The runtime eid is deliberately NOT stored (ids are allocated fresh
46 // whenever the chunk loads), and knockback / death-window state is
47 // transient by design. Structural sanity (finite numbers, world bounds,
48 // sane health ceilings) is checked here; registry membership is
49 // re-validated by the World on restore, so the format layer never
50 // drifts from the entity registry.

52 #include <cstdint>
53 #include <string>
54 #include <vector>

56 #include "world/format/Chunk.h" // mcp_chunk_t (chunk entity blob channel)

60 // One persisted mob: the subset of world::MobEntity that survives a
61 // chunk unload or a restart. Plain struct on purpose — it is copied out
62 // of the world's mob map under the mob lock and dropped right after the
63 // write.
64 struct PersistedMob {
65  int32_t type = 0; // Protocol-84 network id (entity::EntityTypes).
66  double x = 0.0, y = 0.0, z = 0.0; // Feet position.
67  float yaw = 0.0f, pitch = 0.0f;
68  // Head facing, persisted as the optional "HeadYaw" float tag. Rows
69  // saved before the head-sync work (and the legacy mobs.dat lines)
70  // decode with head yaw equal to body yaw — the rotation those rows
71  // carried.
72  float head_yaw = 0.0f;
73  float health = 0.0f;
74  float max_health = 0.0f;
75 };

77 class MobChunkCodec {
78  public:
79  // Serializes one mob as a little-endian NBT compound blob.
80  static std::vector<uint8_t> encodeBlob(const PersistedMob& mob);
81 
82  // Parses one blob. Tolerant by contract: a corrupted row must never
83  // cost the chunk its other mobs. Returns false for malformed or
84  // out-of-range rows (missing id/Pos, non-finite numbers, world
85  // bounds, dead health values); rotations are wrapped into the wire
86  // ranges so hand-edited data cannot push a weird yaw through the
87  // spawn packet.
88  static bool decodeBlob(const uint8_t* data, size_t len, PersistedMob* out);
89 
90  // Replaces the chunk's whole entity list with fresh blobs for
91  // `mobs`. Returns true when the on-chunk bytes actually changed —
92  // the caller marks the chunk dirty only then, so an unchanged mob
93  // roster never triggers a pointless chunk rewrite.
94  static bool writeChunkEntities(mcp_chunk_t* chunk,
95  const std::vector<PersistedMob>& mobs);
96 
97  // Reads every parsable mob row back out of a chunk's entity list.
98  static std::vector<PersistedMob> readChunkEntities(const mcp_chunk_t* chunk);
99 
100  // One-time importer for the legacy side file ("<world
101  // dir>/mobs.dat" of builds that persisted mobs outside the region
102  // files): same tolerant row-by-row contract the old reader had, so
103  // a hand-corrupted line costs one mob instead of the migration.
104  static std::vector<PersistedMob> loadLegacyMobsDat(const std::string& path);
105 };
