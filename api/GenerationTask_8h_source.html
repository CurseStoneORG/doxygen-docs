23 // GenerationTask — wraps a single chunk generation request for the
24 // AsyncPool. Worker thread picks up the GenerationTask, runs
25 // generator->generate + populate, and posts the chunk back to the
26 // World thread via a thread-safe queue.
27 #pragma once
28 #include "world/generator/Generator.h"
29 #include "world/format/Chunk.h"
30 #include "world/format/ChunkSection.h"
31 #include <functional>
32 #include <memory>
33 #include <string.h>
34 namespace cursestone::world::generator {
35 class GenerationTask {
36  public:
37  using DoneCallback = std::function<void(mcp_chunk_t*)>;
38  GenerationTask(std::shared_ptr<Generator> gen, int32_t x, int32_t z, int64_t seed,
39  DoneCallback done)
40  : gen_(std::move(gen)), x_(x), z_(z), seed_(seed), done_(std::move(done)) {
41  }
42  void run() {
43  // FIX (chunks that never render): everything below runs on an
44  // AsyncPool worker. If generate()/populate() throws for some
45  // chunk, runOne() swallows the exception but run() never reaches
46  // done_(c) — the only place that clears pending_. The chunk would
47  // stay "in flight" forever and the world would keep a permanent
48  // hole. Fix: catch here, build an emergency flat chunk (same
49  // pattern as the synchronous fallback) and guarantee done_() runs
50  // either way.
51  mcp_chunk_t* partial = nullptr;
52  try {
53  runInner(partial);
54  } catch (const std::exception& e) {
55  // 'partial' is the chunk runInner had already allocated
56  // before generate()/populate() threw — free it.
57  if (partial)
58  mcp_chunk_destroy(partial);
59  done_(makeEmergencyChunk(e.what()));
60  } catch (...) {
61  if (partial)
62  mcp_chunk_destroy(partial);
63  done_(makeEmergencyChunk("unknown exception"));
64  }
65  }
66 
67  private:
68  mcp_chunk_t* makeEmergencyChunk(const char* reason) {
69  mcp_chunk_t* c = mcp_chunk_new(x_, z_);
70  if (!c)
71  return nullptr;
72  for (int bx = 0; bx < 16; ++bx) {
73  for (int bz = 0; bz < 16; ++bz) {
74  mcp_chunk_set_block_id(c, bx, 0, bz, 7); // BEDROCK
75  mcp_chunk_set_block_id(c, bx, 1, bz, 3); // DIRT
76  mcp_chunk_set_block_id(c, bx, 2, bz, 3); // DIRT
77  mcp_chunk_set_block_id(c, bx, 3, bz, 2); // GRASS
78  mcp_chunk_set_biome(c, bx, bz, 1);
79  }
80  }
81  c->terrain_populated = true;
82  c->dirty = true;
83  mcp_chunk_recalculate_height_map(c);
84  mcp_chunk_propagate_sky_light(c);
85  (void)reason; // kept for future logging if a logger ever lands here
86  return c;
87  }
88  void runInner(mcp_chunk_t*& partial_out) {
89  mcp_chunk_t* c = mcp_chunk_new(x_, z_);
90  if (!c) {
91  done_(nullptr);
92  return;
93  }
94  partial_out = c; // visible al catch de run() si algo de abajo explota
95 
96  // Defensive default biome (plains) in case the generator
97  // doesn't set its own (Normal/Flat both do, but this keeps
98  // behaviour identical for any future one that doesn't). Runs
99  // on the worker thread — Generator is documented stateless
100  // w.r.t. the main thread, so this is safe.
101  //
102  // REAL BUG (black world/grass): this used to fill sky_light with
103  // 0xFF before a single block existed and never computed real
104  // light — full brightness everywhere plus a broken heightmap that
105  // counted leaves/flowers as ceiling, rendering solid black under
106  // any tree canopy.
107  // Fix: generate terrain first, THEN recalc the heightmap, then
108  // run the real light pass (which replaces the old memset).
109  for (int bx = 0; bx < 16; ++bx)
110  for (int bz = 0; bz < 16; ++bz)
111  mcp_chunk_set_biome(c, bx, bz, 1);
112 
113  GenerationContext ctx{seed_, x_, z_, nullptr, c};
114  gen_->generate(ctx);
115  gen_->populate(ctx);
116  c->terrain_populated = true;
117  c->dirty = true;
118  mcp_chunk_recalculate_height_map(c);
119  mcp_chunk_propagate_sky_light(c); // sets c->light_populated = true internally
120  done_(c);
121  }
122 
123  private:
124  std::shared_ptr<Generator> gen_;
125  int32_t x_, z_;
126  int64_t seed_;
127  DoneCallback done_;
128 };
129 } // namespace
