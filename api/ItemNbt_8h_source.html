25 // ItemNbt — an item <-> NBT in ONE place.
26 //
27 // The entry layout is the client's own: net.minecraft.world.item.
28 // ItemInstance::save/load (handheld/src/world/item/ItemInstance.cpp),
29 // which is what every container entry in the world uses:
30 //
31 // id SHORT item id
32 // Count BYTE stack size
33 // Damage SHORT aux value (durability for tools/armor)
34 //
35 // on top of it, the enchantment list, which the client's 2013 loader
36 // never had but Bedrock/Genisys always stored:
37 //
38 // ench LIST<COMPOUND> { id SHORT, lvl SHORT }
39 //
40 // That second tag is the whole point of this module: the damage pipeline
41 // already queries the armour's EPF and the weapon's Sharpness / Knockback /
42 // Fire Aspect, but nothing ever filled ItemStack::ench from disk, so every
43 // one of those lookups read 0. Items now load their enchantments on the
44 // same path that loads their id.
45 //
46 // The same three fields used to be hand-copied in four loaders/savers
47 // (Player::saveData, TileChest, TileFurnace, Player::loadNBT) — which is
48 // also why enchantments could never have worked in any of them. They all
49 // go through here now.
50 // ============================================================

58 // NBT keys of an item entry (see the header comment).
59 constexpr const char* kNbtId = "id";
60 constexpr const char* kNbtCount = "Count";
61 constexpr const char* kNbtDamage = "Damage";
62 constexpr const char* kNbtEnch = "ench";
63 // Bedrock/Genisys write the enchantment level as "lvl"; a few Nukkit-era
64 // writers spell it out as "level". The reader accepts both, the writer
65 // always emits "lvl".
66 constexpr const char* kNbtEnchLvl = "lvl";
67 constexpr const char* kNbtEnchLvlAlt = "level";
68 // Vanilla item-extra-data key. Java saves the client's {display:{Name}},
69 // {ench}, etc. compound under "tag"; CurseStone persists the stack's raw
70 // little-endian network NBT blob (ItemStack::nbt) the same way so a
71 // custom-named item survives player/chest/furnace save + load round
72 // trips. See itemStackToNbt()/itemStackFromNbt().
73 constexpr const char* kNbtTag = "tag";
74 
75 // Serialize `stack` into the compound `entry`, replacing any previous
76 // id/Count/Damage/ench it held. A stack with no enchantments has its
77 // "ench" key removed rather than written as an empty list, so ordinary
78 // items keep the exact byte layout they had before this module existed.
79 // No-op if `entry` is not a CompoundTag.
80 void itemStackToNbt(const ItemStack& stack, mcp_tag_t* entry);
81 
82 // Read a container entry back. Returns false when `entry` is not a
83 // compound or holds no item (id == 0 or Count <= 0) — in that case `out` is
84 // left empty. Otherwise id / meta / count / enchantments all come back.
85 bool itemStackFromNbt(const mcp_tag_t* entry, ItemStack& out);
86 
87 // Just the "ench" list of `entry`, decoded into `out` (which is NOT
88 // cleared first: entries accumulate, which is what EnchantmentList::add
89 // is for). Exposed for callers that already have the item fields, and for
90 // the regression test.
91 void enchantmentsFromNbt(const mcp_tag_t* entry, enchantment::EnchantmentList& out);
92 
93 // Builds the little-endian network NBT compound that gives an item a
94 // custom visible name on 0.15.10 clients: {display:{Name:"..."}}. The
95 // returned blob goes verbatim into ItemStack::nbt; every slot writer
96 // prefixes it with its u16 LE length (Genisys putSlot). An empty name
97 // yields an empty blob (an unnamed item has no NBT at all).
98 std::string buildDisplayNbt(const std::string& name);
