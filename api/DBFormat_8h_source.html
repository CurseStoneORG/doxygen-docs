24 // DBFormat — the byte-level primitives of the real LevelDB on-disk format.
25 //
26 // The LevelDB world provider must read and write the SAME files the real
27 // MCPE 0.15.x client (and PocketMine) produce, so this is not a private
28 // container: every constant, record layout and CRC discipline here is the
29 // one Google's LevelDB (and the MCPE fork of it) defines:
30 //
31 // Internal key user key + 8-byte trailer, LE: sequence(56) << 8 | type(8)
32 // Log/manifest 32 KiB blocks; record = CRC32C(4) + len(2) + type(1)
33 // + payload, types FULL/FIRST/MIDDLE/LAST (7/2/3/4, +1
34 // for masked-CRC-recycled records)
35 // VersionEdit varint32 tag stream (comparator, log number, next file,
36 // last sequence, deleted file, new file ...)
37 // Table block entries: shared(Var32) + non_shared(Var32) + value_len
38 // (Var32) + key delta + value; restart points every 16
39 // entries; 5-byte trailer (CRC32C + compression byte)
40 // Block types 0 = raw, 1 = snappy, 2 = zlib (the MCPE fork addition
41 // PocketMine also writes)
42 // Footer metaindex handle + index handle, padded to 4096 total,
43 // 8-byte magic 0xdb4775248b80fb57 LE
44 //
45 // Reads accept raw/snappy/zlib blocks so worlds written by the real
46 // device (snappy) and by PocketMine (zlib) both open; writes emit zlib
47 // level 1 (measured on Anvil's chunk payloads: ~3x faster than the
48 // default level for +14% size — the same trade a live server wants).

170 // Build the 8-byte trailer (sequence<<8|type, LE) and the full internal key.
171 inline std::string make_internal_key(const std::string& user_key, uint64_t seq,
172  uint8_t type) {
173  std::string out = user_key;
174  uint64_t trailer = (seq << 8) | type;
175  for (int i = 0; i < 8; ++i)
176  out.push_back((uint8_t)(trailer >> (8 * i)));
177  return out;
178 }

180 inline std::string user_key_of(const std::string& internal) {
181  return internal.size() >= 8 ? internal.substr(0, internal.size() - 8) : internal;
182 }
183 
184 // The 8-byte trailer is little-endian: byte 0 = type, bytes 1..7 = sequence.
185 inline uint64_t sequence_of(const std::string& internal) {
186  if (internal.size() < 8)
187  return 0;
188  uint64_t t = 0;
189  for (int i = 0; i < 7; ++i)
190  t |= (uint64_t)(uint8_t)internal[internal.size() - 7 + i] << (8 * i);
191  return t;
192 }
193 
194 inline uint8_t type_of(const std::string& internal) {
195  return internal.size() >= 8 ? (uint8_t)internal[internal.size() - 8] : 0;
196 }
