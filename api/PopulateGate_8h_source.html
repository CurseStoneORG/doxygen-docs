23 // PopulateGate — reproduce the vanilla populate order under concurrent
24 // workers.
25 //
26 // Vanilla populates chunks strictly in chunk order, and chunk output is
27 // NOT order-independent: a populate replays the cross-chunk feature
28 // writes (spills) that earlier chunks parked, and structure pieces
29 // mutate during postProcess (lazy ground height, consumed pieces), so
30 // the windows of one structure depend on which windows ran before them.
31 // With N pool workers the completion order races, and measured digests
32 // change run to run (tools/gen_equiv_matrix caught it: 47 of 49 chunks
33 // differed from the sequential pass, threads=4, every run a different
34 // digest).
35 //
36 // The fix coordinates instead of serializing: every chunk gets a rank —
37 // its position in the reference order (ascending chunk coords, the same
38 // order the sequential tools use). A populate may only run once every
39 // lower-ranked chunk that can still affect it has finished:
40 // * the 3x3 neighbourhood — their spills must be parked before this
41 // populate replays them, or the replay set depends on timing;
42 // * the chunks whose window intersects a structure start this chunk
43 // also claims — piece state must move in reference order.
44 // Chunks outside those sets populate fully in parallel, so terrain
45 // throughput keeps scaling; only genuinely interacting chunks wait.
46 //
47 // tryEnter() is NON-blocking on purpose: it runs inside an AsyncPool
48 // task, and a worker that blocks on an unfinished dependency can
49 // deadlock the whole pool (every worker waiting on tasks still sitting
50 // in the queue). The caller re-queues its task instead and retries —
51 // dependencies always point down-rank, so the lowest unfinished populate
52 // is always runnable and progress is guaranteed.

55 #include <cstdint>

62 // Rank in the reference populate order: ascending (cx, cz). The +2^31
63 // bias keeps negative coords ahead of positives, matching signed order.
64 inline uint64_t populateRank(int32_t cx, int32_t cz) {
65  const uint64_t bx = (uint64_t)((int64_t)cx + 0x80000000LL);
66  const uint64_t bz = (uint64_t)((int64_t)cz + 0x80000000LL);
67  return (bx << 32) | bz;
68 }

71 // Plains decorator reference state (the vanilla 1-in-10 forest-grass branch).
72 //
73 // Vanilla BiomeDecorator::decorate reads Biome::plains->decorator->grassCount
74 // for its 1-in-10 "forest" branch, and PlainsBiome::decorate retunes that
75 // same shared decorator on every plains populate. The value the branch sees
76 // is therefore plains' decorator state carried over from earlier populations
77 // — a strictly-sequential-populate concept that concurrent workers can
78 // neither share safely (the pre-fix singleton raced, digest e21c87 -> race)
79 // nor simply freeze (the constructor default diverges from the sequential
80 // reference: seed 1337 radius 5 shifted one brown-mushroom cell because the
81 // branch read 10 instead of the 5 a plains chunk two ranks earlier had
82 // retuned).
83 //
84 // The reference order makes the carry-over reproducible WITHOUT shared
85 // mutable state: under the gate's rank discipline, plains' decorator state
86 // at the moment rank K populates is exactly the retune of the highest plains
87 // rank <= K in the registered chunk set — a plains chunk reads its own just
88 // retuned value, a non-plains chunk reads the nearest lower-ranked plains
89 // chunk's. Each retune is a pure function of the chunk coordinates
90 // (seed-independent biome-info noise), so the table is fully known at
91 // register time, before any populate runs, and every read returns the same
92 // value on every worker and every run. The table lives per VgenSource; a
93 // thread-local pointer carries it into BiomeDecorator::decorate for the
94 // duration of one populate.
95 class PlainsRefTable {
96 public:
97  void note(uint64_t rank, int grass) {
98  std::lock_guard<std::mutex> lk(mtx_);
99  plainsByRank_[rank] = grass;
100  }
101 
102  // The decorator state rank K's 1-in-10 branch would read under the
103  // reference order; the PlainsBiome constructor default when no plains
104  // chunk precedes it (vanilla: nothing populated plains' decorator yet).
105  int grassAtOrBefore(uint64_t rank) const {
106  std::lock_guard<std::mutex> lk(mtx_);
107  auto it = plainsByRank_.upper_bound(rank);
108  if (it == plainsByRank_.begin())
109  return kDefaultGrassCount;
110  --it;
111  return it->second;
112  }
113 
114  static constexpr int kDefaultGrassCount = 10; // PlainsBiome ctor value
115 
116 private:
117  mutable std::mutex mtx_;
118  std::map<uint64_t, int> plainsByRank_;
119 };

121 // Installed by VgenSource::populate for the duration of one populate. Reads
122 // outside a populate (tools, tests) fall back to the constructor default —
123 // the vanilla singleton's never-populated state.
124 inline PlainsRefTable*& currentPlainsRef() {
125  static thread_local PlainsRefTable* t = nullptr;
126  return t;
127 }

129 // RAII installer — populate has early returns, so the restore must not

131 class PlainsRefGuard {
132 public:

134  : saved_(currentPlainsRef()) {

136  }

145 class PopulateGate {
146 public:
147  // Claim a slot for a chunk that was requested and will populate.
148  // Idempotent; called when the request is accepted, before any task
149  // runs, so a dependency that is queued but not started is still
150  // visible as "not done" to everyone.
151  void registerSlot(uint64_t key) {
152  std::lock_guard<std::mutex> lk(mtx_);
153  slots_.emplace(key, Pending);
154  }
155 
156  // Non-blocking: returns true only when the slot is pending AND every
157  // dependency in `deps` is done (absent from the map = never
158  // requested = can never write = done). On success the slot moves to
159  // Running; the caller MUST eventually markDone() it, or waiters
160  // would spin forever.
161  template <class DepRange>
162  bool tryEnter(uint64_t key, const DepRange& deps) {
163  std::lock_guard<std::mutex> lk(mtx_);
164  auto it = slots_.find(key);
165  if (it == slots_.end() || it->second != Pending)
166  return false;
167  for (const uint64_t d : deps) {
168  auto dit = slots_.find(d);
169  if (dit != slots_.end() && dit->second != Done)
170  return false;
171  }
172  it->second = Running;
173  return true;
174  }
175 
176  // Release the slot (erase + wake re-queuers). Absent keys are fine.
177  void markDone(uint64_t key) {
178  {
179  std::lock_guard<std::mutex> lk(mtx_);
180  slots_.erase(key);
181  }
182  cv_.notify_all();
183  }
184 
185  // Debug/telemetry helper: -1 = absent (done or never registered),
186  // 0 = pending, 1 = running.
187  int stateOf(uint64_t key) {
188  std::lock_guard<std::mutex> lk(mtx_);
189  auto it = slots_.find(key);
190  if (it == slots_.end())
191  return -1;
192  return (int)it->second;
193  }
194 
195  size_t size() {
196  std::lock_guard<std::mutex> lk(mtx_);
197  return slots_.size();
198  }
199 
200 private:
201  enum State : uint8_t { Pending, Running, Done };
202  std::mutex mtx_;
203  std::condition_variable cv_; // kept for future blocking waiters
204  std::unordered_map<uint64_t, State> slots_;
205 };
