24 // Structure — a fixed block template stamped into the world by
25 // StructurePopulator / VillagePopulator.
26 //
27 // A structure is stored the way the CurseStone-Structures exports store
28 // it: a palette plus one DENSE volume — every cell of the bounding box
29 // is present, air included. Keeping the air matters: placement is then
30 // a plain overwrite, so the hollow interior of a desert temple really
31 // is hollow instead of leaving whatever the terrain pass happened to
32 // leave inside it.
33 //
34 // The volume is a string, row-major in (z, y): row `ly * sizeZ + lz`
35 // holds `sizeX` characters and character 'a' + n is palette entry n.
36 // Structures are small (the desert temple is 11x7x10 = 770 cells) so the
37 // whole template stays readable in the source and needs no data file at
38 // runtime. Invariant: the string has exactly sizeX*sizeY*sizeZ entries.
39 //
40 // Two palette flavours:
41 //
42 // * Direct: the palette is finished (id, data) pairs — the desert
43 // temple, whose Java states were already hand-mapped to legacy.
44 // * Spec: the palette is biome-independent BlockSpecs — the village
45 // pieces, whose wood/cobble/farm blocks must resolve DIFFERENTLY per
46 // village biome (a desert village is sandstone where a plains one is
47 // wood, vanilla's getBiomeSpecificBlockState). The caller passes a
48 // resolver when placing.
49 //
50 // Rotation: villages are stamped at one of four 90-degree rotations so
51 // the same handful of pieces does not always read the same way. Rotation
52 // is clockwise viewed from above (+X east, +Z south); facings inside the
53 // blocks rotate with the shape, so doors still face their opening after
54 // the piece is turned.

62 // One palette entry: the MCPE 0.15.x legacy block id and its metadata
63 // byte, which is what the chunk actually stores.
64 struct BlockState {
65  uint8_t id;
66  uint8_t data;
67 };

69 // One biome-independent palette entry for village templates. `kind` is a
70 // SpecKind (defined in village/VillagePalette.h — it is the vocabulary
71 // the per-biome resolver speaks); the other fields carry the export's
72 // Java state, normalised:
73 //
74 // facing: 0 none, 1 east, 2 west, 3 south, 4 north, 5 up — the JAVA
75 // compass the exports use. Legacy data bits are derived from
76 // it at resolve time, per block kind.
77 // aux: stairs/slab: 1 = top half. door: 1 = upper half. farmland:
78 // moisture. everything else: 0.
79 // misc: door: 1 = open (the legacy format does not store the hinge
80 // side, so the export's hinge property is dropped). wool:
81 // colour. fluids: level.
82 struct BlockSpec {
83  uint8_t kind;
84  uint8_t facing;
85  uint8_t aux;
86  uint8_t misc;
87 };

89 // Clockwise quarter turns applied to a template at placement time.
90 enum Rotation : uint8_t { ROT_0 = 0, ROT_90 = 1, ROT_180 = 2, ROT_270 = 3 };
91 
92 // Turns a spec into the legacy (id, data) pair to write, honouring the
93 // rotation already applied to the template (facing values rotate with
94 // it) and whatever user context the caller chained through (the village
95 // resolver reads the village biome from it).
96 using SpecResolver = BlockState (*)(const BlockSpec& spec, int rot, void* user);
97 
98 // Rotate a facing (JAVA compass: 1 east, 2 west, 3 south, 4 north; 0
99 // none and 5 up are rotation-invariant) by `rot` clockwise quarter
100 // turns. One turn maps east->south, south->west, west->north,
101 // north->east. Lives here because both the placement core and the
102 // per-biome resolvers need the exact same turn.
103 inline uint8_t rotateFacing(uint8_t f, int rot) {
104  switch (rot & 3) {
105  case ROT_90: {
106  constexpr uint8_t cw[5] = {0, 3, 4, 2, 1};
107  return f <= 4 ? cw[f] : f;
108  }
109  case ROT_180: {
110  constexpr uint8_t cc[5] = {0, 2, 1, 4, 3};
111  return f <= 4 ? cc[f] : f;
112  }
113  case ROT_270: {
114  constexpr uint8_t cw3[5] = {0, 4, 3, 1, 2};
115  return f <= 4 ? cw3[f] : f;
116  }
117  default:
118  return f;
119  }
120 }

154  // Footprint of the template after `rot` quarter turns.
155  int rotatedSizeX(int rot) const {
156  return (rot == ROT_90 || rot == ROT_270) ? sizeZ_ : sizeX_;
157  }

159  return (rot == ROT_90 || rot == ROT_270) ? sizeX_ : sizeZ_;

162  // Block at local coordinates (lx, ly, lz). Coordinates outside the
163  // volume read as AIR, so callers never have to bounds-check.
164  //
165  // Spec palettes have no meaning outside a placement (their blocks
166  // depend on the village biome), so this only answers for direct
167  // palettes and reads AIR for spec ones.
168  BlockState blockAt(int lx, int ly, int lz) const {
169  if (lx < 0 || ly < 0 || lz < 0 || lx >= sizeX_ || ly >= sizeY_ || lz >= sizeZ_)
170  return {0, 0};
171  const size_t i = (size_t)((ly * sizeZ_ + lz) * sizeX_ + lx);
172  const size_t entry = (size_t)(volume_[i] - 'a');
173  if (palette_ == nullptr || entry >= paletteCount_)
174  return {0, 0};
175  return palette_[entry];
176  }
177 
178  // Does the bounding box, placed with its (0,0,0) cell at world
179  // (wx, wy, wz), touch the chunk (chunkX, chunkZ)?
180  //
181  // Y is NOT tested: the caller only asks this to decide whether a
182  // candidate structure is worth scanning, and the vertical extent is
183  // checked when the base height is chosen.
184  bool overlapsChunk(int chunkX, int chunkZ, int wx, int wz) const {
185  const int bx = chunkX * 16, bz = chunkZ * 16;
186  return wx < bx + 16 && wx + sizeX_ > bx && wz < bz + 16 && wz + sizeZ_ > bz;
187  }

189  // Write every cell of the volume into `chunk`, clipped to that chunk.
190  // Blocks that fall outside the chunk, or above the 128-block world,
191  // are skipped — so a structure is never left half written just
192  // because its neighbour has not been generated yet: every chunk
193  // re-derives the same template from the seed and writes only its own
194  // share. Returns how many cells were actually written.
195  size_t placeInto(mcp_chunk_t* chunk, int chunkX, int chunkZ, int wx, int wy, int wz) const {
196  return placeRotated(chunk, chunkX, chunkZ, wx, wy, wz, ROT_0, nullptr, nullptr);
197  }

199  // Same, rotated by `rot` quarter turns clockwise, with the palette
200  // resolved per entry through `resolve` (village pieces). A nullptr
201  // resolver on a spec palette resolves entries as-is-unrotated and is
202  // only useful for tests.
203  size_t placeInto(mcp_chunk_t* chunk, int chunkX, int chunkZ, int wx, int wy, int wz,
204  Rotation rot, SpecResolver resolve, void* user) const {
205  return placeRotated(chunk, chunkX, chunkZ, wx, wy, wz, rot, resolve, user);
206  }

211  (void)biome;

224  // Template cell -> source cell, for `rot` clockwise turns. Kept in
225  // lockstep with rotateRect() in VillagePopulator.cpp: a point (x, z)
226  // of the source lands at (sizeZ-1-z, x) after one CW turn, so both
227  // the pixels and the caller's footprint rectangle rotate identically.
228  void sourceCell(int lx, int lz, int rot, int& sx, int& sz) const {
229  switch (rot & 3) {
230  case ROT_90:
231  sx = lz;
232  sz = sizeZ_ - 1 - lx;
233  break;
234  case ROT_180:
235  sx = sizeX_ - 1 - lx;
236  sz = sizeZ_ - 1 - lz;
237  break;
238  case ROT_270:
239  sx = sizeX_ - 1 - lz;
240  sz = lx;
241  break;
242  default:
243  sx = lx;
244  sz = lz;
245  break;
246  }
247  }

249  // Spec entry at source coordinates, or {0,0,0,0} out of range.
250  BlockSpec specAt(int lx, int ly, int lz) const {
251  if (lx < 0 || ly < 0 || lz < 0 || lx >= sizeX_ || ly >= sizeY_ || lz >= sizeZ_)
252  return {0, 0, 0, 0};
253  const size_t i = (size_t)((ly * sizeZ_ + lz) * sizeX_ + lx);
254  const size_t entry = (size_t)(volume_[i] - 'a');
255  if (entry >= specCount_)
256  return {0, 0, 0, 0};
257  return specs_[entry];
258  }
