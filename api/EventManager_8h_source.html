23 // EventManager — bus de eventos de todo el servidor. Es dueño de UN
24 // HandlerList por cada subclase concreta de Event (indexado por
25 // std::type_index vía RTTI sobre el Event base, que es polimórfico) y
26 // es el único punto de entrada que el resto del código usa para
27 // disparar eventos y escucharlos.
28 //
29 // Esta es la pieza que hace que las ~120 subclases de Event bajo
30 // src/event/ hagan algo de verdad. Antes de esto:
31 // - Ninguna de esas ~120 clases se instanciaba en ningún lugar del
32 // código fuera de su propia carpeta.
33 // - HandlerList::dispatch() recibía el evento por const-ref, así que
34 // ningún listener podía cancelar nada aunque quisiera.
35 // - EventPriority.h ni siquiera existía como archivo.
36 //
37 // USO — escuchar un evento:
38 // EventManager::get().listen<event::block::BlockBreakEvent>(this,
39 // [](event::block::BlockBreakEvent& e) {
40 // if (algunaCondicion) e.setCancelled();
41 // }, EventPriority::HIGH);
42 //
43 // USO — disparar un evento ya construido y leer si fue cancelado:
44 // event::block::BlockBreakEvent ev(player, pos, blockId);
45 // EventManager::get().call(ev);
46 // if (ev.isCancelled()) return;
47 //
48 // USO — construir + disparar + devolver por valor en un solo paso:
49 // auto ev = EventManager::get().fire<event::player::PlayerChatEvent>(
50 // this, message);
51 // if (ev.isCancelled()) return;
52 // broadcast(ev.message); // el listener pudo haber reescrito el mensaje
53 //
54 // HILOS: listen()/unlisten()/call() pueden invocarse desde cualquier
55 // hilo — cada HandlerList protege sus propios registros con un mutex.
56 // Dicho esto, por convención (igual que el resto del tick loop) los
57 // eventos de juego deben dispararse desde el hilo principal: nada acá
58 // lo impide, pero World/Player no son thread-safe por sí mismos, así
59 // que un callback que los toque debe seguir corriendo en el hilo
60 // principal.

73 namespace cursestone::event {

75 class EventManager {
76 public:
77  static EventManager& get();
78 
79  // Registra `handler` para EventT en la prioridad dada. `owner` es
80  // una etiqueta opaca (normalmente `this`) que solo sirve para que
81  // unlistenAll(owner) pueda encontrar y quitar esta suscripción más
82  // adelante — nunca se desreferencia. Devuelve un id de registro
83  // utilizable con unlisten<EventT>().
84  template <typename EventT>
85  uint64_t listen(void* owner,
86  std::function<void(EventT&)> handler,
87  EventPriority priority = EventPriority::NORMAL,
88  bool ignoreCancelled = false) {
89  static_assert(std::is_base_of<Event, EventT>::value,
90  "EventManager::listen<T>() requiere que T herede de event::Event");
91  HandlerList& list = listFor(std::type_index(typeid(EventT)));
92  return list.registerListener(owner, priority, ignoreCancelled,
93  [handler](Event& e) { handler(static_cast<EventT&>(e)); });
94  }
95 
96  // Quita un único registro de EventT por id.
97  template <typename EventT>
98  bool unlisten(uint64_t id) {
99  static_assert(std::is_base_of<Event, EventT>::value,
100  "EventManager::unlisten<T>() requiere que T herede de event::Event");
101  HandlerList* list = findList(std::type_index(typeid(EventT)));
102  return list ? list->unregister(id) : false;
103  }
104 
105  // Quita TODOS los registros (de cualquier tipo de evento) que
106  // pertenezcan a `owner`. Llamar esto cuando un objeto Listener se
107  // destruye, para que jamás quede un callback colgante apuntando a
108  // memoria liberada.
109  size_t unlistenAll(void* owner);
110 
111  // Despacha `event` a cada listener registrado para su tipo
112  // DINÁMICO exacto, en orden de prioridad. No hace nada (aparte del
113  // lookup) si nadie se registró jamás para ese tipo concreto.
114  void call(Event& event);
115 
116  // Construye un EventT con `args`, lo despacha y lo devuelve por
117  // valor para que el llamador pueda inspeccionar el resultado
118  // (isCancelled(), campos reescritos por listeners, etc).
119  template <typename EventT, typename... Args>
120  EventT fire(Args&&... args) {
121  static_assert(std::is_base_of<Event, EventT>::value,
122  "EventManager::fire<T>() requiere que T herede de event::Event");
123  EventT ev(std::forward<Args>(args)...);
124  call(ev);
125  return ev;
126  }
127 
128  // Cantidad de listeners registrados actualmente para EventT.
129  // Principalmente para diagnóstico/tests.
130  template <typename EventT>
131  size_t listenerCount() const {
132  HandlerList* list = findList(std::type_index(typeid(EventT)));
133  return list ? list->size() : 0;
134  }
135 
136 private:
137  HandlerList& listFor(const std::type_index& idx);
138  HandlerList* findList(const std::type_index& idx) const;
139 
140  mutable std::mutex map_mtx_;
141  std::unordered_map<std::type_index, std::unique_ptr<HandlerList>> lists_;
142 };
