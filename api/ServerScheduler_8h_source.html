23 // ServerScheduler — main-thread tick scheduler + async task dispatcher.
24 //
25 // scheduleSync() runs tasks on the main thread (ticked every 50ms).
26 // scheduleAsync() posts tasks to the Server's AsyncPool (worker threads)
27 // and collects their completion callbacks for execution on the main
28 // thread during the next tick.
29 //
30 // STABILITY NOTES (post fire-cascade incident — see BlockFire.cpp):
31 // - tick() no longer runs any Task::onRun() while holding mtx_. It
32 // snapshots the due handlers under the lock, releases the lock,
33 // then runs them. This means a task that calls scheduleSync()
34 // from inside its own onRun() (e.g. a repeating world-update task
35 // that reschedules itself, or schedules a follow-up task) can no
36 // longer deadlock against the scheduler's own tick — and newly
37 // scheduled tasks are simply picked up on a later tick, not
38 // executed immediately in an unbounded chain.
39 // - A hard cap (MAX_TASKS_PER_TICK) limits how many due tasks run
40 // in a single tick. Any remainder is deferred to the next tick.
41 // This is what actually prevents a cascade (e.g. fire spread)
42 // from freezing a single tick no matter how many tasks a chain
43 // of onRun() calls schedules.
44 // - The async completion queue is capped (MAX_PENDING_COMPLETIONS).
45 // If a burst of AsyncTasks finishes faster than the main thread
46 // can drain them, further completions are dropped with a logged
47 // warning instead of growing the queue (and memory) without
48 // bound.
49 #pragma once
50 #include "TaskHandler.h"
51 #include <vector>
52 #include <memory>
53 #include <atomic>
54 #include <mutex>
55 #include <deque>
56 #include <utility>
57 #include <cstdint>
58 namespace cursestone::scheduler {
59 class AsyncTask;
60 class ServerScheduler {
61  public:
62  static ServerScheduler& get();
63  uint64_t scheduleSync(std::shared_ptr<Task> task, int delay_ticks = 0, int period_ticks = 0);
64  uint64_t scheduleAsync(std::shared_ptr<AsyncTask> task, int delay_ticks = 0);
65  void tick(uint64_t current_tick);
66  size_t size() const;
67 
68  // Cancel a previously scheduled task (sync or async) by the id
69  // returned from scheduleSync()/scheduleAsync(). Safe to call from
70  // the main thread at any time, including from inside a task's own
71  // onRun(). For sync tasks not yet run, this prevents them from
72  // ever running. For async tasks already handed to the AsyncPool,
73  // this prevents onCompletion() from firing on the main thread and
74  // (best-effort) skips onRun() if it hasn't started yet — but a
75  // worker thread that already began onRun() will still finish it;
76  // AsyncTask::onRun() implementations that do meaningful work
77  // should keep checking isCancelled() themselves if they can bail
78  // out early.
79  bool cancel(uint64_t task_id);
80 
81  // Number of async completions dropped so far because the
82  // completion queue was full. Diagnostic only.
83  size_t droppedCompletions() const {
84  return dropped_completions_.load(std::memory_order_relaxed);
85  }
86 
87  private:
88  static constexpr size_t MAX_TASKS_PER_TICK = 64;
89  static constexpr size_t MAX_PENDING_COMPLETIONS = 512;
90 
91  std::vector<std::unique_ptr<TaskHandler>> handlers_;
92  // Task ids whose TaskHandler::tick() returned true (finished/
93  // cancelled) during the current tick()'s run phase, collected
94  // while handlers_ is unlocked so we know exactly what to erase
95  // once we re-lock — instead of re-deriving "is this done?" from
96  // task state after the fact, which can disagree with what
97  // TaskHandler::tick() itself decided (e.g. for periodic tasks
98  // whose nextRun() has already been advanced past current_tick).
99  std::vector<uint64_t> finished_task_ids_;
100  mutable std::mutex mtx_;
101  std::atomic<uint64_t> next_id_{1};
102 
103  // Async completion queue — tasks that finished on worker threads
104  // and need their onCompletion() called on the main thread.
105  std::deque<std::shared_ptr<AsyncTask>> completion_queue_;
106  mutable std::mutex completion_mtx_;
107  std::atomic<size_t> dropped_completions_{0};
108 
109  // task_id -> AsyncTask, kept only long enough to support cancel()
110  // before the task is done and removed from the completion path.
111  // Guarded by completion_mtx_ (same lock domain as the completion
112  // queue, since both are touched from the worker-thread lambda).
113  std::vector<std::pair<uint64_t, std::shared_ptr<AsyncTask>>> in_flight_async_;
114 };
115 } // namespace
