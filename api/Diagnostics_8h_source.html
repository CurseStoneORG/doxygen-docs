24 // Diagnostics.h — always-on diagnostic recorder for crashdumps and errors.
25 //
26 // Why this exists (feedback from the user):
27 // "nuestro crashdump system es muy cerrado y no muestra cosas reveladoras
28 // ... mejoralo y que los crashdumps o errores muestren como el de Nukkit".
29 //
30 // The old CrashDump only wrote {version, reason, backtrace}. That is useless
31 // to debug a crash that happens while a player is opening a chest or while a
32 // malformed packet is being handled: you cannot tell WHICH player, WHAT
33 // packet, or what the server was doing one tick before it died.
34 //
35 // Diagnostics is a tiny, lock-protected singleton that ALWAYS records (no
36 // /workbench start needed):
37 // * the last N packets IN/OUT (who, pid, size)
38 // * the last N reported errors (unknown packet, decode failure, exception)
39 // * a periodic snapshot of every online player (name, eid, ip, position,
40 // chunk, gamemode, open container, time since last activity)
41 // * the current "action" string (e.g. "handling pid=0x2c from Steve")
42 // * server uptime + current tick
43 //
44 // CrashDump and the non-fatal error reports both render this context, so a
45 // crashdump now reads like a Nukkit crash report instead of a lone backtrace.
46 //
47 // Signal-safety: the render helpers take the mutex with try_lock and never
48 // block, so they are safe to call from CurseStone.cpp's SIGSEGV handler. If the
49 // lock is held (rare — a crash landing mid-record), they emit a marker
50 // instead of deadlocking.

67 // One recorded packet (IN = client->server, OUT = server->client).
68 struct PacketRecord {
69  uint64_t wall_ms = 0; // epoch milliseconds
70  bool out = false;
71  std::string who; // player name (OUT) or "ip:port" (IN)
72  uint8_t pid = 0;
73  size_t size = 0;
74 };
75 
76 // One recorded error/diagnostic event.
77 struct ErrorRecord {
78  uint64_t wall_ms = 0;
79  std::string category; // "packet", "player", "exception", "network", ...
80  std::string message; // short one-line summary (shown on kick)
81  std::string detail; // optional long detail
82 };
83 
84 // A cheap snapshot of one online player, refreshed from the main thread.
85 struct PlayerSnapshot {
86  std::string name;
87  std::string ip;
88  uint16_t port = 0;
89  int64_t eid = 0;
90  double x = 0.0, y = 0.0, z = 0.0;
91  int32_t cx = 0, cz = 0;
92  int gamemode = 0;
93  bool connected = false;
94  bool logged_in = false;
95  bool spawned = false;
96  bool open_container = false;
97  uint64_t ms_since_activity = 0;
98 };
