22 // Nether — vanilla-style Nether generator for the 0.15.x world height
23 // (128 blocks, no biomes beyond HELL):
24 //
25 // * bedrock floor (solid at y=0, jagged up to y=4) and bedrock roof
26 // (solid at y=127, jagged down to y=123), the natural vanilla
27 // distribution,
28 // * netherrack terrain shaped by 3D density noise: a mostly solid
29 // shell at both ends and a large cavern belt around y=64,
30 // * a still-lava sea filling the open space up to y=32,
31 // * features: quartz ore, soul sand and gravel patches, hidden lava
32 // pockets, glowstone clusters hanging under ceilings, ground fire.
33 //
34 // The noise is the shared vanilla module (noise/): one MersenneRandom
35 // feeding a single 4-octave PerlinNoise evaluated in 3D, sampled
36 // sparsely and trilinear-interpolated like the classic fast 3D field.
37 #pragma once
38 #include "Generator.h"
39 #include "world/generator/noise/VanillaNoise.h"
40 namespace cursestone::world::generator {
41 class Nether : public Generator {
42  public:
43  explicit Nether(int64_t seed);
44  ~Nether() override;
45  const char* name() const override {
46  return "nether";
47  }
48  void generate(GenerationContext& ctx) override;
49  void populate(GenerationContext& ctx) override;
50  // Mirrors Nukkit Nether.getSpawn(): (128, 64, 128)
51  mcp_vector3_t getSpawn() const override {
52  return mcp_v3(128.0, 64.0, 128.0);
53  }
54 
55  private:
56  // Must be declared first: the noise layer holds a pointer to it.
57  MersenneRandom random_;
58  PerlinNoise noise_;
59  int64_t seed_;
60 };
61 } // namespace
