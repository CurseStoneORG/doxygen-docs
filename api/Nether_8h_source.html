37 // Nether — the vanilla hell generator wired onto the Generator interface.
38 // The terrain half (netherrack density field, lava sea to sea level 32,
39 // bedrock shell, hell caves) builds terrain-only chunks into a cache; the
40 // feature half (fortresses, lava springs, fire, glowstone, mushrooms,
41 // quartz ore) populates over a virtual 3x3 neighborhood exactly like the
42 // overworld VgenSource does.
43 class Nether : public Generator, public vgen::ITerrainFeed, public ISpillQueue {
44 public:
45  explicit Nether(int64_t seed);
46  ~Nether() override;

49  return "nether";
50  }

52  void populate(GenerationContext& ctx) override;

54  // Mirrors the era reference Nether.getSpawn(): (128, 64, 128)

56  return mcp_v3(128.0, 64.0, 128.0);
57  }
