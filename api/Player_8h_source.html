23 // player — full player session. Owns the network session, inventory,
24 // view distance, current chunk, gamemode, position, and to per-player
25 // tick. The player class is the most complex class in CurseStone;
26 // it's the bridge between the protocol layer (LoginPacket → MovePlayerPacket)
27 // and the world layer (World, Chunk, Entity).
28 //
29 // Mirrors pocketmine/player.php but in C++. Lifecycle:
30 // 1. RakLibInterface receives CLIENT_HANDSHAKE → server::openSession()
31 // 2. LoginPacket validated → player::handleLogin() sends PlayStatusPacket
32 // 3. player::doFirstSpawn() sends StartGamePacket + PlayerListPacket
33 // 4. Each tick: player::onUpdate() sends chunk updates, processes input
34 // 5. Disconnect: player::close() saves inventory, sends DisconnectPacket

392  // --- tick ---
393  void onUpdate(uint64_t current_tick);
394  // Client spawn ACK (PlayStatus PLAYER_SPAWN). Marks this client as
395  // actually rendering the world; clears used_chunks_ and re-centers the
396  // streaming queue so any chunk lost during the client's loading screen
397  // gets delivered. See handleDataPacket(PLAY_STATUS_PACKET).
398  //
399  // ROOT CAUSE this closes ("to veces the chunks not is renderizan"):
400  // every FullChunkDataPacket sent BEFORE the client's spawn ACK can be
401  // silently discarded by the client while it is still on its loading
402  // screen, yet used_chunks_ had already marked those chunks as sent —
403  // so they were never re-sent and the world showed to permanent hole
404  // exactly where the pre-ACK burst went. Clearing the set and letting
405  // onUpdate() re-stream in view is safe: chunks the client DID keep are
406  // ignored by it as duplicates; the ones it lost now arrive. The current
407  // sending stays inside the usual MAX_CHUNKS_PER_TICK throttle.
408  //
409  // only the PRIMER ACK CUENTA (bug of "not puedo ver the terrain to the
410  // enter"). the re-stream arranca again by the centro of the
411  // espiral, this way that vaciar used_chunks_ tira tambien the anillo
412  // outside already streamed. if the client scans again to send
413  // PlayStatus(PLAYER_SPAWN) before that the espiral termine, the
414  // progress is reinicia a and again and the horizon never crece.
415  // Medido with tools/terrain_check.py mandando a ACK each 2 s:
416  // distinct is clavaba in 138 of 289 chunks with 750 reenvios, while
417  // each ACK next volvia to tirar everything. the primer ACK is that which
418  // aporta informacion ("already arme the world"); the siguientes not dicen
419  // nothing new, this way that is ignoran.
420  void handleClientSpawnAck() {
421  if (!srv_ || !srv_->getDefaultWorld())
422  return;
423  // compare_exchange: only entra the PRIMER ACK; the rest leaves aca.
424  bool expected = false;
425  if (!spawn_ack_done_.compare_exchange_strong(expected, true))
426  return;
427  {
428  std::lock_guard<std::mutex> lk(chunks_mtx_);
429  used_chunks_.clear();
430  }
431  // Recenter the streaming queue on the player's current chunk;
432  // onUpdate() rebuilds the spiral order every tick (center-outward,
433  // throttled) and, with used_chunks_ empty, re-sends the whole view.
434  int cx = (int)(pos_.x) >> 4;
435  int cz = (int)(pos_.z) >> 4;
436  {
437  std::lock_guard<std::mutex> lk(queue_mtx_);
438  chunk_queue_.clear();
439  chunk_queue_.emplace_back(cx, cz);
440  }
441  if (srv_ && srv_->advancedLog()) {
442  utils::MainLogger::get().info("[CHUNKS] Client spawn ACK for " + cfg_.username +
443  " - re-streaming view radius " +
444  std::to_string(cfg_.view_distance));
445  }
446  }
