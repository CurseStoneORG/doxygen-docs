23 // player — full player session. Owns the network session, inventory,
24 // view distance, current chunk, gamemode, position, and to per-player
25 // tick. The player class is the most complex class in CurseStone;
26 // it's the bridge between the protocol layer (LoginPacket → MovePlayerPacket)
27 // and the world layer (World, Chunk, Entity).
28 //
29 // Mirrors pocketmine/player.php but in C++. Lifecycle:
30 // 1. RakLibInterface receives CLIENT_HANDSHAKE → server::openSession()
31 // 2. LoginPacket validated → player::handleLogin() sends PlayStatusPacket
32 // 3. player::doFirstSpawn() sends StartGamePacket + PlayerListPacket
33 // 4. Each tick: player::onUpdate() sends chunk updates, processes input
34 // 5. Disconnect: player::close() saves inventory, sends DisconnectPacket

324  // --- tick ---
325  void onUpdate(uint64_t current_tick);
326  // Client spawn ACK (PlayStatus PLAYER_SPAWN). Marks this client as
327  // actually rendering the world; clears used_chunks_ and re-centers the
328  // streaming queue so any chunk lost during the client's loading screen
329  // gets delivered. See handleDataPacket(PLAY_STATUS_PACKET).
330  //
331  // ROOT CAUSE this closes ("to veces the chunks not is renderizan"):
332  // every FullChunkDataPacket sent BEFORE the client's spawn ACK can be
333  // silently discarded by the client while it is still on its loading
334  // screen, yet used_chunks_ had already marked those chunks as sent —
335  // so they were never re-sent and the world showed to permanent hole
336  // exactly where the pre-ACK burst went. Clearing the set and letting
337  // onUpdate() re-stream in view is safe: chunks the client DID keep are
338  // ignored by it as duplicates; the ones it lost now arrive. The current
339  // sending stays inside the usual MAX_CHUNKS_PER_TICK throttle.
340  //
341  // only the PRIMER ACK CUENTA (bug of "not puedo ver the terrain to the
342  // enter"). the re-stream arranca again by the centro of the
343  // espiral, this way that vaciar used_chunks_ tira tambien the anillo
344  // outside already streamed. if the client scans again to send
345  // PlayStatus(PLAYER_SPAWN) before that the espiral termine, the
346  // progress is reinicia a and again and the horizon never crece.
347  // Medido with tools/terrain_check.py mandando a ACK each 2 s:
348  // distinct is clavaba in 138 of 289 chunks with 750 reenvios, while
349  // each ACK next volvia to tirar everything. the primer ACK is that which
350  // aporta informacion ("already arme the world"); the siguientes not dicen
351  // nothing new, this way that is ignoran.
352  void handleClientSpawnAck() {
353  if (!srv_ || !srv_->getDefaultWorld()) return;
354  // compare_exchange: only entra the PRIMER ACK; the rest leaves aca.
355  bool expected = false;
356  if (!spawn_ack_done_.compare_exchange_strong(expected, true))
357  return;
358  {
359  std::lock_guard<std::mutex> lk(chunks_mtx_);
360  used_chunks_.clear();
361  }
362  // Recenter the streaming queue on the player's current chunk;
363  // onUpdate() rebuilds the spiral order every tick (center-outward,
364  // throttled) and, with used_chunks_ empty, re-sends the whole view.
365  int cx = (int)(pos_.x) >> 4;
366  int cz = (int)(pos_.z) >> 4;
367  {
368  std::lock_guard<std::mutex> lk(queue_mtx_);
369  chunk_queue_.clear();
370  chunk_queue_.emplace_back(cx, cz);
371  }
372  if (srv_ && srv_->advancedLog()) {
373  utils::MainLogger::get().info("[CHUNKS] Client spawn ACK for " +
374  cfg_.username + " - re-streaming view radius " +
375  std::to_string(cfg_.view_distance));
376  }
377  }
