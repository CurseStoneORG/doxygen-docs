287  // Re-streams chunks after the client's first spawn acknowledgment.
288  void handleClientSpawnAck() {
289  if (!srv_ || !srv_->getDefaultWorld())
290  return;
291  // Process only the first spawn acknowledgment.
292  bool expected = false;
293  if (!spawn_ack_done_.compare_exchange_strong(expected, true))
294  return;
295  {
296  std::lock_guard<std::mutex> lk(chunks_mtx_);
297  used_chunks_.clear();
298  }
299  // Restart chunk streaming around the player's current position.
300  int cx = (int)(pos_.x) >> 4;
301  int cz = (int)(pos_.z) >> 4;
302  {
303  std::lock_guard<std::mutex> lk(queue_mtx_);
304  chunk_queue_.clear();
305  chunk_queue_.emplace_back(cx, cz);
306  }
307  if (srv_ && srv_->advancedLog()) {
308  utils::MainLogger::get().info("[CHUNKS] Client spawn ACK for " + cfg_.username +
309  " - re-streaming view radius " +
310  std::to_string(cfg_.view_distance));
311  }
312  }
