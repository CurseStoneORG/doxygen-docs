295  // The WorldEdit-route block write for one tile cell: an
296  // UpdateBlockPacket carrying the LIVE grid's id + meta (a chest cell
297  // reads 54 + facing 2..5, a furnace cell 61 + facing 2..5) with
298  // FLAG_ALL_PRIORITY — byte-for-byte the packet World::
299  // broadcastBlockUpdate builds for world.setBlock(). The 0.15 client
300  // auto-creates the block actor when a block update lands (the exact
301  // path a player-placed chest takes — the /we set 54:0 chests and
302  // /we set 61:0 furnaces that render and open), while a standalone
303  // BlockEntityData can only UPDATE an actor the client already holds,
304  // never create one. Sent on the chunk stream (ordered channel 2) so
305  // it can never overtake its own chunk: chunk -> block update (actor
306  // born) -> BlockEntityData (actor refreshed). Read-only over the
307  // tile state — no tile is created, moved or emptied by this packet.
308  // Call it for the tiles tileNeedsActorAnnounce() accepts (chest,
309  // furnace); signs and spawners bind from the chunk suffix alone.

437  // Re-streams chunks after the client's first spawn acknowledgment.
438  void handleClientSpawnAck() {
439  if (!srv_ || !srv_->getDefaultWorld())
440  return;
441  // Process only the first spawn acknowledgment.
442  bool expected = false;
443  if (!spawn_ack_done_.compare_exchange_strong(expected, true))
444  return;
445  {
446  std::lock_guard<std::mutex> lk(chunks_mtx_);
447  used_chunks_.clear();
448  }
449  // Fresh encode pipeline: the re-stream sends everything again, and
450  // encodes already in flight for the old view must not be flushed
451  // as duplicates (in-flight workers finish into the orphaned
452  // pipeline and their payloads are dropped).
453  chunk_encodes_.reset();
454  // Restart chunk streaming around the player's current position.
455  int cx = (int)(pos_.x) >> 4;
456  int cz = (int)(pos_.z) >> 4;
457  {
458  std::lock_guard<std::mutex> lk(queue_mtx_);
459  chunk_queue_.clear();
460  chunk_queue_.emplace_back(cx, cz);
461  }
462  if (srv_ && srv_->debugMode()) {
463  utils::MainLogger::get().info("[CHUNKS] Client spawn ACK for " + cfg_.username +
464  " - re-streaming view radius " +
465  std::to_string(cfg_.view_distance));
466  }
467  }
