67 // ============================================================
68 // Puente del ruido vanilla a un valor [-1, 1]
69 // ============================================================
70 // El PerlinNoise vanilla suma las octavas AL REVES que un fbm normal:
71 //
72 // vanilla(x, y) = sum(noise(x/2^i, y/2^i) * 2^i) (pow /= 2)
73 // fbm(x, y) = sum(noise(x*2^i, y*2^i) * 0.5^i)
74 //
75 // asi que su octava dominante (la ultima) cae 2^(levels-1) veces mas lejos
76 // que la frecuencia que le pasan, y a la escala de bloque del generador el
77 // campo sale casi constante. Estas funciones devuelven la octava dominante a
78 // la frecuencia base (multiplicando la entrada por 2^(levels-1)) y dividen por
79 // la amplitud total sum(2^i) = 2^levels - 1, dejando un valor acotado en
80 // [-1, 1] comparable al del fbm que usaba el generador.
81 inline double layerAmplitudeSum(int levels) {
82  double s = 0.0, amp = 1.0;
83  for (int i = 0; i < levels; ++i) {
84  s += amp;
85  amp *= 2.0;
86  }
87  return s;
88 }
89 
90 inline double layerDominantScale(int levels) {
91  return (levels <= 0) ? 1.0 : (double)(1u << (levels - 1));
92 }

94 inline double noiseValue2d(PerlinNoise& p, double x, double y) {
95  // Las dos constantes ya vienen precalculadas en la capa (init());
96  // antes se recalculaban con un bucle en CADA evaluacion de campo, y
97  // cada columna de terreno pide decenas de estas. Mismos valores,
98  // mismos bits, sin el bucle.
99  const double sum = p.amplitudeSum();
100  if (sum <= 0.0)
101  return 0.0;
102  const double k = p.dominantScale();
103  return (double)p.getValue((float)(x * k), (float)(y * k)) / sum;
104 }

106 inline double noiseValue3d(PerlinNoise& p, double x, double y, double z) {
107  const double sum = p.amplitudeSum();
108  if (sum <= 0.0)
109  return 0.0;
110  const double k = p.dominantScale();
111  return (double)p.getValue((float)(x * k), (float)(y * k), (float)(z * k)) / sum;
112 }
