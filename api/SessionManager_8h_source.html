49 // =========================================================================
50 // Traffic pacing tables
51 // =========================================================================
52 // Every datagram that enters the manager crosses two fixed-size, O(1)
53 // open-addressing tables BEFORE any parsing happens:
54 //
55 // * endpoint table (key = ip:port) — a per-sweep datagram budget. A
56 // source that exceeds it has its excess datagrams dropped right at the
57 // recv loop; three consecutive over-budget sweeps start pacing the
58 // endpoint. This is what keeps one noisy source from consuming the
59 // whole per-sweep drain cap.
60 //
61 // * source-IP table (key = ip) — per-SECOND budgets for the unconnected
62 // packet classes (0x01/0x02 pings, 0x05/0x07 handshake requests, fresh
63 // session creations) and for concurrent sessions per IP. These are the
64 // vectors that cost server work without ever authenticating: pings
65 // cost a reply, handshake storms cost session allocations, and both
66 // arrive from ports that rotate faster than any per-endpoint state can
67 // track. Over budget -> the packet is dropped and the source strikes
68 // once; when strikes reach the limit the source is PACED: every
69 // unconnected datagram from it is silently ignored for a hold that
70 // doubles per repeat offense up to the ceiling. Established sessions
71 // NEVER cross these per-IP budgets (their datagrams are dispatched by
72 // endpoint before the budget check), so one abusive source on a shared
73 // NAT can never starve the players behind the same IP, and an honest
74 // in-game client cannot pace itself out by talking to its own server.
75 //
76 // Both tables live in fixed arrays inside the manager (memset()-able, zero
77 // allocation on the hot path). Entries expire after expire_ms idle and the
78 // whole set is swept once per second by the manager tick: idle entries are
79 // evicted (backward-shift, like the session index) and strikes decay after
80 // ten quiet seconds. When pacing is disabled the recv loop skips the whole
81 // gate: disabled costs literally nothing.
82 //
83 // The budgets are operator-tunable from cursestone.yml (traffic.* keys are
84 // read by Server::loadConfig and applied through
85 // mcp_rak_session_manager_apply_limits before the socket binds).
86 typedef struct mcp_traffic_limits {
87  bool enabled; // master switch (traffic.pacing)
88  uint32_t max_packets_per_tick; // datagrams per endpoint per sweep (0 = off)
89  uint32_t max_pings_per_sec; // 0x01/0x02 per source IP per second
90  uint32_t max_handshakes_per_sec; // 0x05/0x07 per source IP per second
91  uint32_t max_new_sessions_per_sec; // fresh sessions per source IP per second
92  uint32_t max_sessions_per_ip; // concurrent sessions from one source IP
93  uint32_t strikes_before_pacing; // budget overflows before pacing starts
94  uint32_t hold_ms; // first pacing hold
95  uint32_t hold_max_ms; // escalation ceiling (doubles per offense)
96  uint32_t expire_ms; // idle table-entry lifetime
97 } mcp_traffic_limits_t;

124 // Table sizes (power of two). The endpoint table must absorb every distinct
125 // source a scan can throw at us between sweeps; 4096 covers the 512-per-sweep
126 // drain cap with room for the honest population. The IP table is smaller
127 // because a second-window budget saturates long before thousands of
128 // distinct IPs matter.
129 #define MCP_TRAFFIC_EP_BITS 12
130 #define MCP_TRAFFIC_EP_SIZE (1u << MCP_TRAFFIC_EP_BITS)
131 #define MCP_TRAFFIC_IP_BITS 10
132 #define MCP_TRAFFIC_IP_SIZE (1u << MCP_TRAFFIC_IP_BITS)

221 // Tear down the session for (ip, port), if any, keeping the owner array and
222 // the O(1) endpoint index coherent: the index entry is erased with the exact
223 // (key, pointer) pair, the session is destroyed and freed, and the owner
224 // array is compacted with a swap-remove. The session is located by scanning
225 // the owner array — never the endpoint index — so a stale index entry can
226 // never cause the wrong (or a reused) session to be destroyed.
227 // Returns true if a session was found and torn down; false if there was no
228 // session for that endpoint (safe no-op).
