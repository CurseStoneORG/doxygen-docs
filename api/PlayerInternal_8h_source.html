37 // Vanilla MCPE 0.15.10 player eye height (Human::getEyeHeight() == 1.62).
38 //
39 // RakNet's MovePlayerPacket and MoveEntityPacket carry the EYE altitude on
40 // the wire; EVERY other position-carrying packet (StartGame, AddPlayerPacket,
41 // SetSpawnPosition, Respawn) and the whole entity/physics model use the FEET.
42 // pos_ holds the FEET, exactly like Genisys' Entity::$y — so handleMovePlayer()
43 // subtracts this on receive and the two movement packets add it on send.
44 //
45 // Before this constant existed pos_ stored the raw wire value, i.e. the EYE,
46 // which left every feet-based computation 1.62 too high. The item-pickup
47 // volume (anchored on the feet) was the visible casualty: its 1.0-below-the-
48 // feet reach actually pointed 0.62 ABOVE the real feet, so items resting on
49 // the ground could never be collected while items floating at chest height
50 // could — the "only my upper half can pick things up" report.
51 constexpr double kEyeHeight = 1.62;

53 // Builds the metadata entry describing a player's pose:
54 // Entity::DATA_FLAGS = 0 (byte) — bit 0 = ON_FIRE, bit 1 = SNEAKING,
55 // bit 3 = SPRINTING (protocol 84 entity metadata). Protocol 84 clients
56 // render the burning effect (flames + fire overlay) from the ON_FIRE bit
57 // alone. All bits share the same property, so they always travel together
58 // as the full byte.
59 inline network::protocol::SetEntityDataPacket::Entry makePoseFlagEntry(bool sneaking,
60  bool sprinting,
61  bool action = false,
62  bool onFire = false) {
63  network::protocol::SetEntityDataPacket::Entry e;
64  e.index = network::protocol::SetEntityDataPacket::PROP_FLAGS;
65  e.type = network::protocol::SetEntityDataPacket::TYPE_BYTE;
66  uint8_t flags = 0;
67  if (onFire) flags |= (1 << network::protocol::SetEntityDataPacket::FLAG_BIT_ONFIRE);
68  if (sneaking) flags |= (1 << network::protocol::SetEntityDataPacket::FLAG_BIT_SNEAKING);
69  if (sprinting) flags |= (1 << network::protocol::SetEntityDataPacket::FLAG_BIT_SPRINTING);
70  if (action) flags |= (1 << network::protocol::SetEntityDataPacket::FLAG_BIT_ACTION);
71  e.ival = flags;
72  return e;
73 }
