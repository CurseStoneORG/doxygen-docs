24 // CommandReader — runs a background thread that reads stdin line by
25 // line. When a line is received, it forwards it to the Server's
26 // CommandMap. Mirrors command/CommandReader.
27 //
28 // CRITICAL: stdin is kept off C stdio on every platform. fgets() holds
29 // the stdin FILE lock for the whole call, including the blocking read —
30 // a reader blocked on an idle terminal keeps that lock forever, and
31 // glibc's exit-time _IO_cleanup then deadlocks trying to lock every
32 // open FILE, so the process never actually leaves exit() after
33 // "Shutdown complete". Mixing C stdio (fputs) with C++ streams on the
34 // same FILE also causes data corruption. POSIX reads fd 0 with
35 // poll()+read(); Windows mirrors the same 200 ms cadence on the raw
36 // stdin handle (console handles are waitable, pipes need PeekNamedPipe),
37 // so stop() stays responsive and no FILE lock is held while idle.
38 #pragma once
39 #ifdef _WIN32
40 #ifndef WIN32_LEAN_AND_MEAN
41 #define WIN32_LEAN_AND_MEAN
42 #endif
43 #ifndef NOMINMAX
44 #define NOMINMAX
45 #endif
46 #include <windows.h>
47 #else
48 #include <poll.h>
49 #include <unistd.h>
50 #endif
51 #include <atomic>
52 #include <functional>
53 #include <string>
54 #include <thread>
55 namespace cursestone {
56 class CommandReader {
57  public:
58  using LineCallback = std::function<void(const std::string&)>;
59  explicit CommandReader(LineCallback cb) : cb_(std::move(cb)) {
60  }
61  ~CommandReader() {
62  stop();
63  }
64  void start() {
65  running_ = true;
66  thread_ = std::thread([this]() {
67  this->loop();
68  });
69  }
70  void stop() {
71  if (!running_.exchange(false))
72  return;
73  // Both loops wake every 200 ms and re-check the flag, so the
74  // thread is gone shortly after the flag drops.
75  if (thread_.joinable())
76  thread_.join();
77  }
78 
79  private:
80 #ifdef _WIN32
81  void loop() {
82  std::string pending;
83  char buf[1024];
84  HANDLE in = ::GetStdHandle(STD_INPUT_HANDLE);
85  if (!in || in == INVALID_HANDLE_VALUE)
86  return;
87  // FILE_TYPE_REMOTE is ORed in for network shares; strip it so
88  // remote disk handles fall into the plain-file branch.
89  const DWORD type = ::GetFileType(in) & ~FILE_TYPE_REMOTE;
90  while (running_.load()) {
91  bool ready;
92  if (type == FILE_TYPE_PIPE) {
93  // Anonymous pipes are not waitable, so peek instead.
94  DWORD avail = 0;
95  if (!::PeekNamedPipe(in, nullptr, 0, nullptr, &avail, nullptr))
96  break; // stdin closed or broken.
97  if (avail == 0) {
98  ::Sleep(200); // Same cadence as the wait branches.
99  continue;
100  }
101  ready = true;
102  } else if (type == FILE_TYPE_CHAR) {
103  // Console input handles are waitable; a timed wait keeps
104  // the loop free of any held lock.
105  ready = ::WaitForSingleObject(in, 200) == WAIT_OBJECT_0;
106  } else {
107  ready = true; // Regular files: ReadFile returns promptly.
108  }
109  if (!ready)
110  continue; // Nothing yet — re-check the flag.
111  DWORD n = 0;
112  if (!::ReadFile(in, buf, (DWORD) sizeof(buf), &n, nullptr) || n == 0)
113  break; // EOF or error — stdin closed.
114  pending.append(buf, (size_t) n);
115  dispatchLines(pending);
116  }
117  }
118 #else
119  void loop() {
120  std::string pending;
121  char buf[1024];
122  while (running_.load()) {
123  struct pollfd pfd {0, POLLIN, 0};
124  int pr = ::poll(&pfd, 1, 200);
125  if (pr <= 0)
126  continue; // Timeout or interrupt — re-check the flag.
127  ssize_t n = ::read(0, buf, sizeof(buf));
128  if (n <= 0)
129  break; // EOF or error — stdin closed.
130  pending.append(buf, (size_t) n);
131  dispatchLines(pending);
132  }
133  }
134 #endif
135  // Forwards complete '\n'-terminated lines and keeps the partial
136  // tail for the next read.
137  void dispatchLines(std::string& pending) {
138  size_t pos;
139  while ((pos = pending.find('\n')) != std::string::npos) {
140  std::string line = pending.substr(0, pos);
141  pending.erase(0, pos + 1);
142  while (!line.empty() && (line.back() == '\n' || line.back() == '\r')) {
143  line.pop_back();
144  }
145  if (!line.empty() && cb_) {
146  cb_(line);
147  }
148  }
149  }
150  LineCallback cb_;
151  std::atomic<bool> running_{false};
152  std::thread thread_;
153 };
154 } // namespace
