23 // SaveProfiler — where save time actually goes.
24 //
25 // FASE L instrumentation: the autosave / /save / shutdown save pipeline
26 // (chunk snapshot -> save queue -> LevelDB worker -> WAL -> memtable ->
27 // table flush -> compaction) accumulates nanoseconds, call counts and
28 // byte counters into lock-free atomics, plus a few live gauges (queue
29 // depth, bytes queued, chunks pending). /status and the save diagnostics
30 // read the snapshot; nothing here influences save behavior or ordering.
31 //
32 // Overhead when enabled: two steady_clock reads plus a few atomic adds
33 // per save operation (per chunk, not per block) — noise next to the
34 // tens of microseconds a chunk snapshot costs. setEnabled(false) drops
35 // every probe to a single relaxed atomic load; production benchmarks
36 // run with it off.
37 //
38 // Threading: every counter is a relaxed atomic; producers run on the
39 // main thread (snapshot/enqueue/drain), the LevelDB worker (put/flush/
40 // compact) and the GC eviction path — none of them ever block on these.
41 #pragma once
42 
43 #include <atomic>
44 #include <chrono>
45 #include <cstdint>

49 // Nanosecond/call pairs, one per pipeline stage. Stages overlap on
50 // purpose (WorkerPut includes a table flush when the memtable crosses
51 // its threshold inside the batch) — read them as "what does each part
52 // cost", not as a partition.
53 enum class SaveStage : uint8_t {
54  Snapshot, // chunk snapshot + blob copy on the calling thread
55  Enqueue, // enqueueSave total, including blocked backpressure
56  EnqueueBlocked, // time the caller spent waiting for queue space
57  WorkerPut, // put_batch on the save worker (WAL + memtable)
58  EngineFlush, // memtable -> SSTable (+ fsync + WAL rotate + manifest)
59  EngineCompact, // full compaction (read every table, write merged)
60  DrainWait, // flushPendingSaves caller-side wait (the /save stall)
61  EngineFlushCall, // explicit engine flush (drain end / close)
62  FullFlushSweep, // ChunkManager::flushDirty total (mtx_ held)
63  BudgetFlushSweep,// ChunkManager::flushDirtyBudget total (mtx_ held)
64  DirtyCount, // dirtyCount() scan cost (polled by autosave/status)
65  Count
66 };

68 class SaveProfiler {
69  public:
70  struct Entry {
71  uint64_t ns = 0;
72  uint64_t calls = 0;
73  };
74  struct Snapshot {
75  Entry stages[(size_t)SaveStage::Count];
76  // Live gauges and byte counters (non-destructive cumulative).
77  uint64_t queue_depth = 0; // save jobs waiting right now
78  uint64_t queue_depth_max = 0; // high-water mark since boot
79  uint64_t queue_bytes = 0; // bytes held by queued jobs (RAM)
80  uint64_t queue_bytes_max = 0; // high-water mark since boot
81  uint64_t chunks_submitted = 0; // jobs accepted into the queue
82  uint64_t chunks_written = 0; // batches the worker wrote to the db
83  uint64_t write_failures = 0; // put_batch rejections
84  uint64_t wal_bytes = 0; // bytes appended to the WAL
85  uint64_t table_bytes = 0; // bytes written into SSTables
86  uint64_t compact_bytes = 0; // bytes read back by compactions
87  uint64_t flush_count = 0; // memtable -> table transitions
88  uint64_t compact_count = 0; // full-merge compactions
89  uint64_t enqueue_blocks = 0; // callers that hit a full queue
90  };
91 
92  // Probes are on by default (a few atomic adds per chunk save);
93  // production benchmarks switch them off.
94  static bool enabled() {
95  return enabled_.load(std::memory_order_relaxed);
96  }
97  static void setEnabled(bool on) {
98  enabled_.store(on, std::memory_order_relaxed);
99  }
100 
101  static void add(SaveStage s, uint64_t ns) {
102  if (!enabled())
103  return;
104  auto& e = stages_[(size_t)s];
105  e.ns.fetch_add(ns, std::memory_order_relaxed);
106  e.calls.fetch_add(1, std::memory_order_relaxed);
107  }
108 
109  // --- gauge updates (cheap, called alongside the stages above) ------
110  static void noteQueueDepth(size_t depth, size_t bytes) {
111  queue_depth_.store(depth, std::memory_order_relaxed);
112  queue_bytes_.store(bytes, std::memory_order_relaxed);
113  if (!enabled())
114  return;
115  // High-water marks: single CAS loop, contended only while deep.
116  uint64_t d = depth, db = queue_depth_max_.load(std::memory_order_relaxed);
117  while (d > db && !queue_depth_max_.compare_exchange_weak(
118  db, d, std::memory_order_relaxed)) {
119  }
120  uint64_t b = bytes, bb = queue_bytes_max_.load(std::memory_order_relaxed);
121  while (b > bb && !queue_bytes_max_.compare_exchange_weak(
122  bb, b, std::memory_order_relaxed)) {
123  }
124  }
125  // Callers that found the queue full and had to wait (backpressure
126  // events, counted separately from the gauge updates above).
127  static void noteEnqueueBlocked() {
128  enqueue_blocks_.fetch_add(1, std::memory_order_relaxed);
129  }
130  static void noteChunkSubmitted() {
131  chunks_submitted_.fetch_add(1, std::memory_order_relaxed);
132  }
133  static void noteChunkWritten(bool ok) {
134  (ok ? chunks_written_ : write_failures_).fetch_add(1, std::memory_order_relaxed);
135  }
136  static void noteWalBytes(uint64_t n) {
137  wal_bytes_.fetch_add(n, std::memory_order_relaxed);
138  }
139  static void noteTableBytes(uint64_t n) {
140  table_bytes_.fetch_add(n, std::memory_order_relaxed);
141  }
142  static void noteCompactBytes(uint64_t n) {
143  compact_bytes_.fetch_add(n, std::memory_order_relaxed);
144  }
145  static void noteFlush() {
146  flush_count_.fetch_add(1, std::memory_order_relaxed);
147  }
148  static void noteCompact() {
149  compact_count_.fetch_add(1, std::memory_order_relaxed);
150  }

152  // Non-destructive read for /status and the save diagnostics; the
153  // cumulative counters keep accumulating (diff two peeks to describe
154  // a window). Stage entries are cumulative here, unlike
155  // GenerationProfiler::drain — save windows are long, nobody wants
156  // them erased by a status poll.
157  static Snapshot peek() {
158  Snapshot out;
159  for (size_t i = 0; i < (size_t)SaveStage::Count; ++i) {
160  out.stages[i].ns = stages_[i].ns.load(std::memory_order_relaxed);
161  out.stages[i].calls = stages_[i].calls.load(std::memory_order_relaxed);
162  }
163  out.queue_depth = queue_depth_.load(std::memory_order_relaxed);
164  out.queue_depth_max = queue_depth_max_.load(std::memory_order_relaxed);
165  out.queue_bytes = queue_bytes_.load(std::memory_order_relaxed);
166  out.queue_bytes_max = queue_bytes_max_.load(std::memory_order_relaxed);
167  out.chunks_submitted = chunks_submitted_.load(std::memory_order_relaxed);
168  out.chunks_written = chunks_written_.load(std::memory_order_relaxed);
169  out.write_failures = write_failures_.load(std::memory_order_relaxed);
170  out.wal_bytes = wal_bytes_.load(std::memory_order_relaxed);
171  out.table_bytes = table_bytes_.load(std::memory_order_relaxed);
172  out.compact_bytes = compact_bytes_.load(std::memory_order_relaxed);
173  out.flush_count = flush_count_.load(std::memory_order_relaxed);
174  out.compact_count = compact_count_.load(std::memory_order_relaxed);
175  out.enqueue_blocks = enqueue_blocks_.load(std::memory_order_relaxed);
176  return out;
177  }

179  // Reset the high-water marks and cumulative counters (used by the
180  // save diagnostics to describe a fresh window). Queue depth/bytes
181  // gauges are live values, not reset.
182  static void resetWindow() {
183  for (size_t i = 0; i < (size_t)SaveStage::Count; ++i) {
184  stages_[i].ns.store(0, std::memory_order_relaxed);
185  stages_[i].calls.store(0, std::memory_order_relaxed);
186  }
187  queue_depth_max_.store(queue_depth_.load(std::memory_order_relaxed),
188  std::memory_order_relaxed);
189  queue_bytes_max_.store(queue_bytes_.load(std::memory_order_relaxed),
190  std::memory_order_relaxed);
191  chunks_submitted_.store(0, std::memory_order_relaxed);
192  chunks_written_.store(0, std::memory_order_relaxed);
193  write_failures_.store(0, std::memory_order_relaxed);
194  wal_bytes_.store(0, std::memory_order_relaxed);
195  table_bytes_.store(0, std::memory_order_relaxed);
196  compact_bytes_.store(0, std::memory_order_relaxed);
197  flush_count_.store(0, std::memory_order_relaxed);
198  compact_count_.store(0, std::memory_order_relaxed);
199  enqueue_blocks_.store(0, std::memory_order_relaxed);
200  }

202  // Times one scope into one stage. Cheap enough to sprinkle freely.
203  class Scope {
204  public:
205  explicit Scope(SaveStage s)
206  : stage_(s),
207  start_(SaveProfiler::enabled() ? std::chrono::steady_clock::now()
208  : TimePoint{}) {}
209  ~Scope() {
210  if (!SaveProfiler::enabled())
211  return;
212  const auto end = std::chrono::steady_clock::now();
213  SaveProfiler::add(
214  stage_,
215  (uint64_t)std::chrono::duration_cast<std::chrono::nanoseconds>(end - start_)
216  .count());
217  }
218 
219  private:
220  using TimePoint = std::chrono::steady_clock::time_point;
221  SaveStage stage_;
222  TimePoint start_;
223  };
